model Dispenser {
  id                     String   @id @default(cuid())
  stationId              String
  name                   String
  code                   String
  manufacturer           String?  // tatsuno, gilbarco, wayne, tokheim, bennett, petrotec, scheidt_bachmann, etc.
  model                  String?
  serialNumber           String?
  controllerType         String?  // pts, fusion, doms, hectronic, etc.
  forecourtControllerId  String?  // Links to ForecourtController - ERP never talks directly to pump
  communicationType      String?  // tcpip, rs232, rs485, serial, ethernet
  ipAddress              String?
  port                   Int?
  pumpLane               Int      @default(1)
  pumpAddress            Int?     // Logical address on the controller bus (1-32)
  protocol               String?  // ifsf, patrius, dfs, wayne-dart, gilbarco-g-site, proprietary
  nozzleCount            Int      @default(2)
  flowRate               Float?   // L/min max flow rate
  supportedFuels         String?  // JSON: ["petrol", "diesel", "premium", "hsd"]
  firmwareVersion        String?
  lastMeterReading       Float    @default(0)
  status                 String   @default("idle") // idle, fueling, stopped, error, locked, maintenance, offline, authorized
  isActive               Boolean  @default(true)
  lastCommunication      DateTime?
  installedDate          DateTime?
  warrantyExpiry         DateTime?
  configJson             String?  // Extended vendor-specific config
  createdAt              DateTime @default(now())
  updatedAt              DateTime @updatedAt

  station              Station              @relation(fields: [stationId], references: [id])
  forecourtController  ForecourtController? @relation(fields: [forecourtControllerId], references: [id])
  nozzles              Nozzle[]
  fuelings              Fueling[]
}
