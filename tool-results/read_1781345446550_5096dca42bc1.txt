   135→function LoadingSkeleton() {
   136→  return (
   137→    <div className="space-y-4">
   138→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">

   146→
   147→export default function TanksModule() {
   148→  const [tanks, setTanks] = useState<Tank[]>([])
   149→  const [deliveries, setDeliveries] = useState<Delivery[]>([])
   150→  const [reconciliations, setReconciliations] = useState<Reconciliation[]>([])
   151→  const [calibrations, setCalibrations] = useState<Calibration[]>([])
   152→  const [envAlerts, setEnvAlerts] = useState<EnvironmentalAlert[]>([])
   153→  const [vaporReadings, setVaporReadings] = useState<VaporReading[]>([])
   154→  const [loading, setLoading] = useState(true)
   155→  const [error, setError] = useState<string | null>(null)
   156→  const [activeTab, setActiveTab] = useState('monitor')
   157→  const [deliveryDialogOpen, setDeliveryDialogOpen] = useState(false)
   158→
   159→  useEffect(() => {
   160→    async function fetchData() {
   161→      try {
   162→        setLoading(true)
   163→        const [tanksRes] = await Promise.all([
   164→          fetch('/api/tanks'),
   165→        ])
   166→        if (!tanksRes.ok) throw new Error('Failed to load tanks')
   167→        const tanksData = await tanksRes.json()
   168→        setTanks(tanksData.tanks || [])
   169→
   170→        // Fetch deliveries
   171→        try {
   172→          const delRes = await fetch('/api/deliveries')
   173→          if (delRes.ok) {
   174→            const delData = await delRes.json()
   175→            setDeliveries(delData.deliveries || [])
   176→          }
   177→        } catch {}
   178→
   179→        // Fetch environmental
   180→        try {
   181→          const envRes = await fetch('/api/environmental')
   182→          if (envRes.ok) {
   183→            const envData = await envRes.json()
   184→            setEnvAlerts(envData.alerts || [])
   185→            setVaporReadings(envData.vaporReadings || [])
   186→          }
   187→        } catch {}
   188→
   189→        // Fetch calibrations
   190→        try {
   191→          const calRes = await fetch('/api/calibrations')
   192→          if (calRes.ok) {
   193→            const calData = await calRes.json()
   194→            setCalibrations(calData.calibrations || [])
   195→          }
   196→        } catch {}
   197→
   198→        // Fetch reconciliations
   199→        try {
   200→          const recRes = await fetch('/api/reconciliations')
   201→          if (recRes.ok) {
   202→            const recData = await recRes.json()
   203→            setReconciliations(recData.reconciliations || [])
   204→          }
   205→        } catch {}
   206→      } catch (err) {
   207→        setError(err instanceof Error ? err.message : 'Unknown error')
   208→      } finally {
   209→        setLoading(false)
   210→      }
   211→    }
   212→    fetchData()
   213→    const interval = setInterval(fetchData, 30000)
   214→    return () => clearInterval(interval)
   215→  }, [])
   216→
   217→  if (loading) return <LoadingSkeleton />
   218→  if (error) {
   219→    return (
   220→      <Card>
   221→        <CardContent className="p-6 text-center">
   222→          <AlertTriangle className="size-8 text-red-500 mx-auto mb-2" />
   223→          <p className="text-sm text-muted-foreground">{error}</p>
   224→          <Button variant="outline" size="sm" className="mt-2" onClick={() => window.location.reload()}>Retry</Button>
   225→        </CardContent>
   226→      </Card>
   227→    )
   228→  }
   229→
   230→  return (
   231→    <div className="space-y-4">
   232→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   233→        <TabsList>
   234→          <TabsTrigger value="monitor">Tank Monitor</TabsTrigger>
   235→          <TabsTrigger value="deliveries">Deliveries</TabsTrigger>
   236→          <TabsTrigger value="reconciliation">Reconciliation</TabsTrigger>
   237→          <TabsTrigger value="calibration">Calibration</TabsTrigger>
   238→          <TabsTrigger value="environmental">Environmental</TabsTrigger>
   239→        </TabsList>
   240→
   241→        {/* Tank Monitor */}
   242→        <TabsContent value="monitor" className="mt-4">
   243→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   244→            {tanks.map((tank) => (
   245→              <Card key={tank.id} className={
   246→                tank.status === 'critical' ? 'border-red-200 dark:border-red-900/50' :
   247→                tank.status === 'low' ? 'border-amber-200 dark:border-amber-900/50' : ''
   248→              }>
   249→                <CardHeader className="pb-2 p-4">
   250→                  <div className="flex items-center justify-between">
   251→                    <CardTitle className="text-sm font-semibold">{tank.name}</CardTitle>
   252→                    <Badge
   253→                      variant={tank.status === 'critical' ? 'destructive' : tank.status === 'low' ? 'secondary' : 'outline'}
   254→                      className="text-[10px] px-1.5 py-0"
   255→                    >
   256→                      {tank.status}
   257→                    </Badge>
   258→                  </div>
   259→                  <CardDescription className="text-[10px] capitalize">{tank.fuelType} • {tank.code}</CardDescription>
   260→                </CardHeader>
   261→                <CardContent className="p-4 pt-0 space-y-3">
   262→                  {/* Level Bar */}
   263→                  <div className="space-y-1">
   264→                    <div className="flex justify-between text-xs">
   265→                      <span>Level</span>
   266→                      <span className="font-bold">{tank.levelPercentage.toFixed(1)}%</span>
   267→                    </div>
   268→                    <div className="h-3 w-full rounded-full bg-muted overflow-hidden">
   269→                      <div
   270→                        className={`h-full rounded-full transition-all ${
   271→                          tank.status === 'critical' ? 'bg-red-500' :
   272→                          tank.status === 'low' ? 'bg-amber-500' : 'bg-emerald-500'
   273→                        }`}
   274→                        style={{ width: `${Math.min(tank.levelPercentage, 100)}%` }}
   275→                      />
   276→                    </div>
   277→                    <div className="flex justify-between text-[10px] text-muted-foreground">
   278→                      <span>{Math.round(tank.currentLevel).toLocaleString()} L</span>
   279→                      <span>{tank.capacity.toLocaleString()} L</span>
   280→                    </div>
   281→                  </div>

   283→                  {/* Metrics */}

   286→                      <Thermometer className="size-3 text-amber-500" />
   287→                      <span>{tank.temperature.toFixed(1)}°C</span>
   288→                    </div>
   289→                    <div className="flex items-center gap-1.5">
   290→                      <Droplets className="size-3 text-blue-500" />
   291→                      <span>Water: {tank.waterLevel}mm</span>
   292→                    </div>
   293→                    {tank.density && (
   294→                      <div className="flex items-center gap-1.5">
   295→                        <Gauge className="size-3 text-purple-500" />
   296→                        <span>ρ: {tank.density.toFixed(3)}</span>
   297→                      </div>
   298→                    )}
   299→                    <div className="flex items-center gap-1.5">
   300→                      <Activity className="size-3 text-slate-400" />
   301→                      <span>ATG: {tank.atgType || 'N/A'}</span>
   302→                    </div>
   303→                  </div>
   304→
   305→                  {/* Mini Chart */}
   306→                  {tank.recentReadings.length > 0 && (
   307→                    <div className="h-16 mt-1">
   308→                      <ResponsiveContainer width="100%" height="100%">
   309→                        <LineChart data={tank.recentReadings.slice().reverse().map((r) => ({
   310→                          time: new Date(r.readingAt).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: false }),

   312→                        }))}>

   316→                      </ResponsiveContainer>
   317→                    </div>
   318→                  )}
   319→                </CardContent>
   320→              </Card>
   321→            ))}
   322→          </div>
   323→        </TabsContent>
   324→
   325→        {/* Deliveries */}
   326→        <TabsContent value="deliveries" className="mt-4">
   327→          <div className="flex items-center justify-between mb-4">
   328→            <div>
   329→              <h3 className="text-sm font-semibold">Fuel Deliveries</h3>
   330→              <p className="text-xs text-muted-foreground">{deliveries.length} delivery records</p>
   331→            </div>
   332→            <Dialog open={deliveryDialogOpen} onOpenChange={setDeliveryDialogOpen}>
   333→              <DialogTrigger asChild>
   334→                <Button size="sm" className="gap-1">
   335→                  <Plus className="size-3" /> New Delivery
   336→                </Button>
   337→              </DialogTrigger>
   338→              <DialogContent>
   339→                <DialogHeader>
   340→                  <DialogTitle>Record New Delivery</DialogTitle>
   341→                </DialogHeader>
   342→                <div className="space-y-3">
   343→                  <div>
   344→                    <label className="text-xs font-medium">Supplier Name</label>
   345→                    <Input placeholder="Enter supplier name" className="mt-1" />
   346→                  </div>
   347→                  <div>
   348→                    <label className="text-xs font-medium">Product Type</label>
   349→                    <Select>
   350→                      <SelectTrigger className="mt-1"><SelectValue placeholder="Select fuel type" /></SelectTrigger>
   351→                      <SelectContent>
   352→                        <SelectItem value="petrol">Petrol</SelectItem>
   353→                        <SelectItem value="diesel">Diesel</SelectItem>
   354→                        <SelectItem value="premium">Premium</SelectItem>
   355→                        <SelectItem value="hsd">HSD</SelectItem>
   356→                      </SelectContent>
   357→                    </Select>
   358→                  </div>
   359→                  <div>
   360→                    <label className="text-xs font-medium">Expected Volume (L)</label>
   361→                    <Input type="number" placeholder="0" className="mt-1" />
   362→                  </div>
   363→                  <div>
   364→                    <label className="text-xs font-medium">Tanker Number</label>
   365→                    <Input placeholder="Enter tanker number" className="mt-1" />
   366→                  </div>
   367→                  <Button className="w-full" onClick={() => setDeliveryDialogOpen(false)}>
   368→                    Record Delivery
   369→                  </Button>
   370→                </div>
   371→              </DialogContent>
   372→            </Dialog>
   373→          </div>
   374→          <Card>
   375→            <CardContent className="p-0">
   376→              <ScrollArea className="max-h-96">
   377→                <Table>
   378→                  <TableHeader>
   379→                    <TableRow>
   380→                      <TableHead className="text-xs">Date</TableHead>
   381→                      <TableHead className="text-xs">Supplier</TableHead>
   382→                      <TableHead className="text-xs">Product</TableHead>
   383→                      <TableHead className="text-xs text-right">Expected</TableHead>
   384→                      <TableHead className="text-xs text-right">Actual</TableHead>
   385→                      <TableHead className="text-xs text-right">Variance</TableHead>
   386→                      <TableHead className="text-xs">Status</TableHead>
   387→                    </TableRow>
   388→                  </TableHeader>
   389→                  <TableBody>
   390→                    {deliveries.length === 0 ? (
   391→                      <TableRow>
   392→                        <TableCell colSpan={7} className="text-center text-xs text-muted-foreground py-8">
   393→                          No delivery records found
   394→                        </TableCell>
   395→                      </TableRow>
   396→                    ) : (
   397→                      deliveries.map((d) => (
   398→                        <TableRow key={d.id}>
   399→                          <TableCell className="text-xs font-mono">
   400→                            {new Date(d.deliveryDate).toLocaleDateString()}
   401→                          </TableCell>
   402→                          <TableCell className="text-xs">{d.supplierName}</TableCell>
   403→                          <TableCell className="text-xs capitalize">{d.productType}</TableCell>
   404→                          <TableCell className="text-xs text-right">{d.expectedVolume.toLocaleString()} L</TableCell>
   405→                          <TableCell className="text-xs text-right">{d.actualVolume.toLocaleString()} L</TableCell>
   406→                          <TableCell className="text-xs text-right">
   407→                            {d.variance !== null ? (
   408→                              <span className={d.variance > 0 ? 'text-red-500' : 'text-emerald-500'}>
   409→                                {d.variance > 0 ? '+' : ''}{d.variance} L
   410→                              </span>
   411→                            ) : '-'}
   412→                          </TableCell>
   413→                          <TableCell>
   414→                            <Badge
   415→                              variant={
   416→                                d.status === 'completed' ? 'outline' :
   417→                                d.status === 'discrepancy' ? 'destructive' : 'secondary'
   418→                              }
   419→                              className="text-[10px] px-1.5 py-0 capitalize"
   420→                            >
   421→                              {d.status}
   422→                            </Badge>
   423→                          </TableCell>
   424→                        </TableRow>
   425→                      ))
   426→                    )}
   427→                  </TableBody>
   428→                </Table>
   429→              </ScrollArea>
   430→            </CardContent>
   431→          </Card>
   432→        </TabsContent>
   433→
   434→        {/* Reconciliation */}
   435→        <TabsContent value="reconciliation" className="mt-4">
   436→          <Card>
   437→            <CardHeader className="pb-2">
   438→              <CardTitle className="text-sm font-semibold">Stock Reconciliation</CardTitle>
   439→              <CardDescription className="text-xs">{reconciliations.length} records</CardDescription>
   440→            </CardHeader>
   441→            <CardContent className="p-0">
   442→              <ScrollArea className="max-h-96">
   443→                <Table>
   444→                  <TableHeader>
   445→                    <TableRow>
   446→                      <TableHead className="text-xs">Date</TableHead>
   447→                      <TableHead className="text-xs text-right">Opening</TableHead>
   448→                      <TableHead className="text-xs text-right">Deliveries</TableHead>
   449→                      <TableHead className="text-xs text-right">Sales</TableHead>
   450→                      <TableHead className="text-xs text-right">Book Stock</TableHead>
   451→                      <TableHead className="text-xs text-right">Actual</TableHead>
   452→                      <TableHead className="text-xs text-right">Variance</TableHead>
   453→                      <TableHead className="text-xs">Status</TableHead>
   454→                    </TableRow>
   455→                  </TableHeader>
   456→                  <TableBody>
   457→                    {reconciliations.length === 0 ? (
   458→                      <TableRow>
   459→                        <TableCell colSpan={8} className="text-center text-xs text-muted-foreground py-8">
   460→                          No reconciliation records found
   461→                        </TableCell>
   462→                      </TableRow>
   463→                    ) : (
   464→                      reconciliations.map((r) => (
   465→                        <TableRow key={r.id}>
   466→                          <TableCell className="text-xs font-mono">
   467→                            {new Date(r.date).toLocaleDateString()}
   468→                          </TableCell>
   469→                          <TableCell className="text-xs text-right">{r.openingStock.toLocaleString()}</TableCell>
   470→                          <TableCell className="text-xs text-right">{r.deliveries.toLocaleString()}</TableCell>
   471→                          <TableCell className="text-xs text-right">{r.sales.toLocaleString()}</TableCell>
   472→                          <TableCell className="text-xs text-right">{r.bookStock.toLocaleString()}</TableCell>

   487→                              >
   488→                                {r.status}
   489→                              </Badge>
   490→                              {r.status === 'pending' && (
   491→                                <Button size="sm" variant="ghost" className="h-5 text-[9px]">Review</Button>
   492→                              )}
   493→                            </div>
   494→                          </TableCell>
   495→                        </TableRow>
   496→                      ))
   497→                    )}
   498→                  </TableBody>
   499→                </Table>
   500→              </ScrollArea>
   501→            </CardContent>
   502→          </Card>
   503→        </TabsContent>
   504→
   505→        {/* Calibration */}
   506→        <TabsContent value="calibration" className="mt-4">
   507→          <Card>
   508→            <CardHeader className="pb-2">
   509→              <CardTitle className="text-sm font-semibold">Calibration Records</CardTitle>
   510→              <CardDescription className="text-xs">{calibrations.length} records</CardDescription>
   511→            </CardHeader>
   512→            <CardContent className="p-0">
   513→              <ScrollArea className="max-h-96">
