   159→   159→  useEffect(() => {

   170→   170→        // Fetch deliveries

   174→   174→            const delData = await delRes.json()
   175→   175→            setDeliveries(delData.deliveries || [])

   178→   178→
   179→   179→        // Fetch environmental
   180→   180→        try {
   181→   181→          const envRes = await fetch('/api/environmental')

   183→   183→            const envData = await envRes.json()
   184→   184→            setEnvAlerts(envData.alerts || [])
   185→   185→            setVaporReadings(envData.vaporReadings || [])

   188→   188→
   189→   189→        // Fetch calibrations

   194→   194→            setCalibrations(calData.calibrations || [])

   198→   198→        // Fetch reconciliations

   202→   202→            const recData = await recRes.json()
   203→   203→            setReconciliations(recData.reconciliations || [])
   204→   204→          }
   205→   205→        } catch {}
   206→   206→      } catch (err) {
   207→   207→        setError(err instanceof Error ? err.message : 'Unknown error')
   208→   208→      } finally {
   209→   209→        setLoading(false)
   210→   210→      }
   211→   211→    }
   212→   212→    fetchData()

   214→   214→    return () => clearInterval(interval)
   215→   215→  }, [])
