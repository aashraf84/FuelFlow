   172→async function fetchApi(action: string, params?: Record<string, string>) {
   173→  const url = new URL(API, window.location.origin)
   174→  url.searchParams.set('action', action)
   175→  if (params) Object.entries(params).forEach(([k, v]) => { if (v) url.searchParams.set(k, v) })
   176→  const res = await fetch(url.toString())
   177→  if (!res.ok) throw new Error(`API error: ${res.status}`)
   178→  return res.json()
   179→}
   180→
   181→async function postApi(action: string, body: Record<string, unknown>) {
   182→  const res = await fetch(API, {
   183→    method: 'POST',
   184→    headers: { 'Content-Type': 'application/json' },
   185→    body: JSON.stringify({ action, ...body }),
   186→  })
   187→  if (!res.ok) throw new Error(`API error: ${res.status}`)
   188→  return res.json()
   189→}
   190→
   191→async function deleteApi(action: string, id: string) {
   192→  const url = new URL(API, window.location.origin)
   193→  url.searchParams.set('action', action)
   194→  url.searchParams.set('id', id)
   195→  const res = await fetch(url.toString(), { method: 'DELETE' })
   196→  if (!res.ok) throw new Error(`API error: ${res.status}`)
   197→  return res.json()
   198→}

   221→function formatTime(dateStr?: string | null) {
   222→  if (!dateStr) return '—'
   223→  const d = new Date(dateStr)
   224→  return d.toLocaleString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
   225→}
   226→
   227→function timeAgo(dateStr?: string | null) {
   228→  if (!dateStr) return 'Never'
   229→  const diff = Date.now() - new Date(dateStr).getTime()
   230→  if (diff < 60000) return 'Just now'
   231→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`
   232→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`
   233→  return `${Math.floor(diff / 86400000)}d ago`
   234→}
