   118→// ── Helpers ──────────────────────────────────────────────────
   119→
   120→function formatPKR(amount: number): string {
   121→  return `PKR ${amount.toLocaleString('en-PK')}`
   122→}
   123→
   124→function formatCompactPKR(amount: number): string {
   125→  if (amount >= 1000000) return `PKR ${(amount / 1000000).toFixed(1)}M`
   126→  if (amount >= 1000) return `PKR ${(amount / 1000).toFixed(0)}K`
   127→  return `PKR ${amount}`
   128→}
