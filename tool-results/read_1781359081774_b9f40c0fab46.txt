   365→// ── Tenant Databases Management Tab ──────────────────────
   366→
   367→function TenantDatabasesTab({ clients, onRefresh }: { clients: Client[]; onRefresh: () => void }) {
   368→  const [provisioning, setProvisioning] = useState<string | null>(null)
   369→  const [dropping, setDropping] = useState<string | null>(null)
   370→  const [migrating, setMigrating] = useState<string | null>(null)
   371→  const [error, setError] = useState<string | null>(null)
   372→  const [success, setSuccess] = useState<string | null>(null)
   373→
   374→  const handleProvision = async (clientId: string, companyName: string) => {
   375→    setProvisioning(clientId)
   376→    setError(null)
   377→    setSuccess(null)
   378→    try {
   379→      const res = await fetch('/api/owner/tenants', {
   380→        method: 'POST',
   381→        headers: { 'Content-Type': 'application/json' },
   382→        body: JSON.stringify({ clientId }),
   383→      })
   384→      const data = await res.json()
   385→      if (!res.ok) {
   386→        setError(data.error || 'Provisioning failed')
   387→      } else {
   388→        setSuccess(`Database for ${companyName} provisioned successfully (${data.tablesCreated || 0} tables created)`)
   389→        onRefresh()
   390→      }
   391→    } catch {
   392→      setError('Network error during provisioning')
   393→    } finally {
   394→      setProvisioning(null)
   395→    }
   396→  }
   397→
   398→  const handleDrop = async (clientId: string, companyName: string) => {
   399→    if (!confirm(`Are you sure you want to DROP the database for ${companyName}? This action is irreversible and will delete ALL data.`)) return
   400→    setDropping(clientId)
   401→    setError(null)
   402→    setSuccess(null)
   403→    try {
   404→      const res = await fetch(`/api/owner/tenants?clientId=${clientId}`, { method: 'DELETE' })
   405→      const data = await res.json()
   406→      if (!res.ok) {
   407→        setError(data.error || 'Failed to drop database')
   408→      } else {
   409→        setSuccess(`Database for ${companyName} dropped successfully`)
   410→        onRefresh()
   411→      }
   412→    } catch {
