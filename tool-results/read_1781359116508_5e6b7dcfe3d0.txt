   327→function filterAllowedFields(
   328→  data: Record<string, unknown>,
   329→  deviceTypeKey: string,
   330→  includeCommon: boolean = true
   331→): Record<string, unknown> {
   332→  const result: Record<string, unknown> = {}
   333→  const allowedFields = new Set<string>()
   334→
   335→  if (includeCommon) {
   336→    for (const f of COMMON_FIELDS) allowedFields.add(f)
   337→  }
   338→  for (const f of TYPE_SPECIFIC_FIELDS[deviceTypeKey] ?? []) allowedFields.add(f)
   339→
   340→  for (const [key, value] of Object.entries(data)) {
   341→    if (allowedFields.has(key) && value !== undefined) {
   342→      result[key] = value
   343→    }
   344→  }
   345→
   346→  return result
   347→}

   368→async function createCommandEntry(data: {
   369→  stationId: string
   370→  deviceType: string
   371→  deviceId: string
   372→  command: string
   373→  parameters: string | null
   374→  priority: string
   375→  maxRetries: number
   376→  retryCount: number
   377→  createdBy: string | null
   378→}): Promise<Record<string, unknown> | null> {
   379→  try {
   380→    const entry = await db.deviceCommandQueue.create({
   381→      data: {
   382→        stationId: data.stationId,
   383→        deviceType: data.deviceType,
   384→        deviceId: data.deviceId,
   385→        command: data.command,
   386→        parameters: data.parameters,
   387→        priority: data.priority,
   388→        status: 'queued',
   389→        maxRetries: data.maxRetries,
   390→        retryCount: data.retryCount,
   391→        createdBy: data.createdBy,
   392→      },
   393→    })
   394→    return entry as Record<string, unknown>
   395→  } catch (err) {
   396→    if (err instanceof Prisma.PrismaClientKnownRequestError && err.code === 'P2003') {
   397→      console.warn('Command queue creation skipped (FK constraint):', err.message)
   398→      return {
   399→        id: 'virtual-' + Date.now(),
   400→        ...data,
   401→        status: 'queued',
   402→        persisted: false,
   403→        reason: 'FK constraint - command stored in memory only',
   404→      }
   405→    }
   406→    throw err
   407→  }

   432→export async function GET(request: Request) {
   433→  try {
   434→    const { searchParams } = new URL(request.url)
   435→    const action = searchParams.get('action')
   436→
   437→    switch (action) {
   438→      // ────────────────────────────────────────────
   439→      // CATALOG - Search the HardwareCatalog
   440→      // ────────────────────────────────────────────
   441→      case 'catalog': {

   443→        if (id) {
   444→          // Single catalog item

   446→            where: { id },
   447→          })
   448→          if (!item) {
   449→            return errorResponse('Catalog item not found', 404)
   450→          }

   452→        }
   453→
   454→        // Search with filters
   455→        const category = searchParams.get('category')
   456→        const manufacturer = searchParams.get('manufacturer')
   457→        const q = searchParams.get('q')
   458→        const page = parseInt(searchParams.get('page') || '1')
   459→        const limit = parseInt(searchParams.get('limit') || '50')
   460→        const skip = (page - 1) * limit
   461→
   462→        const where: Record<string, unknown> = { isActive: true }
   463→        if (category) where.category = category
   464→        if (manufacturer) where.manufacturer = { contains: manufacturer }
   465→        if (q) {
   466→          where.OR = [
   467→            { modelName: { contains: q } },
   468→            { manufacturer: { contains: q } },
   469→            { description: { contains: q } },
   470→            { modelCode: { contains: q } },
   471→          ]
   472→        }
   473→
   474→        const [items, total] = await Promise.all([
   475→          db.hardwareCatalog.findMany({
   476→            where,
   477→            orderBy: [
   478→              { popularity: 'desc' },
   479→              { manufacturer: 'asc' },
   480→              { modelName: 'asc' },
   481→            ],
   482→            skip,
   483→            take: limit,
   484→          }),
   485→          db.hardwareCatalog.count({ where }),
   486→        ])
   487→
   488→        return NextResponse.json({
   489→          items,
   490→          pagination: { page, limit, total, pages: Math.ceil(total / limit) },
   491→        })
   492→      }
   493→
   494→      // ────────────────────────────────────────────
   495→      // CONTROLLERS - List all ForecourtControllers
   496→      // ────────────────────────────────────────────
   497→      case 'controllers': {
   498→        const id = searchParams.get('id')
   499→        if (id) {
   500→          // Single controller with dispensers, config, recent events
   501→          const controller = await db.forecourtController.findUnique({
   502→            where: { id },
   503→            include: {
   504→              station: { select: { id: true, name: true, code: true, city: true } },
   505→              configs: { orderBy: { configKey: 'asc' } },
   506→              eventLogs: {
   507→                orderBy: { createdAt: 'desc' },
   508→                take: 20,
   509→              },
   510→            },
   511→          })
   512→          if (!controller) {
   513→            return errorResponse('Forecourt controller not found', 404)
   514→          }
   515→
   516→          // Fetch dispensers separately (avoids Prisma relation include issue)
   517→          const controllerDispensers = await db.dispenser.findMany({
   518→            where: { forecourtControllerId: id, isActive: true },
   519→            include: {
   520→              nozzles: {
   521→                where: { isActive: true },
   522→                orderBy: { nozzleNumber: 'asc' },
   523→              },
   524→            },
   525→            orderBy: { pumpAddress: 'asc' },
   526→          })
   527→
   528→          return NextResponse.json({ controller, dispensers: controllerDispensers })
   529→        }
   530→
   531→        // List all controllers
   532→        const stationId = searchParams.get('stationId')
   533→        const status = searchParams.get('status')
   534→        const controllerType = searchParams.get('controllerType')
   535→
   536→        const where: Record<string, unknown> = { isActive: true }
   537→        if (stationId) where.stationId = stationId
   538→        if (status) where.status = status
   539→        if (controllerType) where.controllerType = controllerType
   540→
   541→        const controllers = await db.forecourtController.findMany({
   542→          where,
   543→          include: {
   544→            station: { select: { id: true, name: true, code: true } },
   545→          },
   546→          orderBy: { name: 'asc' },
   547→        })
   548→
   549→        // Fetch all dispensers linked to these controllers in one query
   550→        const controllerIds = controllers.map(c => c.id)
   551→        const allDispensers = await db.dispenser.findMany({
   552→          where: {
   553→            forecourtControllerId: { in: controllerIds },
   554→            isActive: true,
   555→          },
   556→          select: {
   557→            id: true,

   587→        const offlineControllers = controllers.filter(c => c.status === 'offline').length
   588→        const errorControllers = controllers.filter(c => c.status === 'error').length
   589→
   590→        return NextResponse.json({

   592→          stats: {
   593→            total: totalControllers,
   594→            online: onlineControllers,
   595→            offline: offlineControllers,
   596→            error: errorControllers,

   637→          // Get configs for dispenser

   646→        // List all dispensers

   648→        const status = searchParams.get('status')
   649→        const manufacturer = searchParams.get('manufacturer')
   650→        const controllerId = searchParams.get('controllerId')

   652→        const where: Record<string, unknown> = { isActive: true }

   654→        if (status) where.status = status

   895→        return NextResponse.json({

  1092→      case 'commands':
  1093→      case 'events': {
  1094→        const isEvents = action === 'events'

  1096→        const severity = searchParams.get('severity')

  1118→            db.deviceCommandQueue.findMany({ where, orderBy: [{ priority: 'desc' }, { createdAt: 'asc' }], skip, take: limit }),
  1119→            db.deviceCommandQueue.count({ where }),

  1145→          const dev = await info.getModel().findUnique({ where: { id } })

  1147→          device = dev as Record<string, unknown>
  1148→        }
  1149→
  1150→        const isReachable = ['online', 'idle', 'fueling', 'running', 'normal', 'recording'].includes(device.status as string)
  1151→        return NextResponse.json({
  1152→          success: true,
  1153→          testResult: {
  1154→            deviceId: id, deviceType,
  1155→            deviceName: device.name,
  1156→            status: device.status,
  1157→            ipAddress: device.ipAddress,
  1158→            port: device.port,
  1159→            communicationType: device.communicationType,
  1160→            reachable: !!isReachable,
  1161→            latency: isReachable ? Math.floor(Math.random() * 30 + 5) : 0,
  1162→            message: isReachable ? `Device "${device.name}" appears reachable (status: ${device.status})` : `Device "${device.name}" appears unreachable (status: ${device.status})`,
  1163→          },
  1164→        })
  1165→      }
  1166→
  1167→      default:
  1168→        return errorResponse(
  1169→          `Unknown action: "${action}". Valid GET actions: catalog, controllers, dispensers, atgs, payment_terminals, generators, cameras, fire_alarms, utility_meters, weather_stations, tanker_trackers, car_washes, air_compressors, vacuum_stations, controller_profiles, dispenser_profiles, atg_profiles, overview, event_logs, events, command_queue, commands, configs, search, test_connection`,
  1170→          400
  1171→        )
  1172→    }
  1173→  } catch (error) {
  1174→    console.error('Hardware API GET error:', error)
  1175→    return NextResponse.json(
  1176→      { error: 'Internal server error', details: error instanceof Error ? error.message : String(error) },
  1177→      { status: 500 }
  1178→    )
  1179→  }
  1180→}
