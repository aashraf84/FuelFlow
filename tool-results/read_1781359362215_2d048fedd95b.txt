   180→// ── Helpers ──────────────────────────────────────────────────
   181→
   182→function getLevelColor(pct: number): string {
   183→  if (pct > 50) return '#34c759'
   184→  if (pct >= 25) return '#ff9f0a'
   185→  return '#ff3b30'
   186→}
   187→
   188→function getLevelBgClass(pct: number): string {
   189→  if (pct > 50) return 'bg-[#34c759]'
   190→  if (pct >= 25) return 'bg-[#ff9f0a]'
   191→  return 'bg-[#ff3b30]'
   192→}
   193→
   194→function getStatusClass(status: TankData['status']): string {
   195→  if (status === 'online') return 'macos-status-online'
   196→  if (status === 'warning') return 'macos-status-warning'
   197→  return 'macos-status-error'
   198→}
   199→
   200→function formatPKR(amount: number): string {
   201→  return `PKR ${amount.toLocaleString('en-PK')}`
   202→}
   203→
   204→function formatTime(iso: string): string {
   205→  return new Date(iso).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: true })
   206→}
   207→
   208→function formatDate(iso: string): string {
   209→  return new Date(iso).toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   210→}

   278→function AlertItem({ alert, index }: { alert: TankAlert; index: number }) {
   279→  const severityIcon = {
   280→    critical: <XCircle className="size-4 text-[#ff3b30]" />,
   281→    warning: <AlertCircle className="size-4 text-[#ff9f0a]" />,
   282→    info: <Info className="size-4 text-[#007aff]" />,
   283→  }
   284→  const severityBg = {
   285→    critical: 'bg-[#ff3b30]/5 border-[#ff3b30]/10',
   286→    warning: 'bg-[#ff9f0a]/5 border-[#ff9f0a]/10',
   287→    info: 'bg-[#007aff]/5 border-[#007aff]/10',
   288→  }
   289→  return (
   290→    <motion.div
   291→      initial={{ opacity: 0, x: -8 }}
   292→      animate={{ opacity: 1, x: 0 }}
   293→      transition={{ duration: 0.25, delay: index * 0.05 }}
   294→      className={`macos-group-item flex items-start gap-3 border ${severityBg[alert.severity]} rounded-lg`}
   295→    >
   296→      <div className="mt-0.5 shrink-0">{severityIcon[alert.severity]}</div>
   297→      <div className="flex-1 min-w-0">
   298→        <div className="flex items-center justify-between gap-2">
   299→          <p className="text-[13px] font-medium text-foreground">{alert.tank}</p>
   300→          <span className="text-[11px] text-muted-foreground shrink-0">{formatTime(alert.timestamp)}</span>
   301→        </div>
   302→        <p className="text-[12px] text-muted-foreground mt-0.5 leading-relaxed">{alert.message}</p>
   303→      </div>
   304→    </motion.div>
   305→  )
   306→}
