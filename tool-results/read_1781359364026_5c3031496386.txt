   169→// ── Helpers ──────────────────────────────────────────────
   170→
   171→function formatPKR(amount: number): string {
   172→  if (amount >= 1000000) return `PKR ${(amount / 1000000).toFixed(2)}M`
   173→  if (amount >= 1000) return `PKR ${(amount / 1000).toFixed(0)}K`
   174→  return `PKR ${amount.toLocaleString()}`
   175→}
