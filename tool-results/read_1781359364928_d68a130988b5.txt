   104→

   109→        {[
   110→          { label: 'Fleet Accounts', value: fleetAccounts.length.toString(), icon: Truck, accent: false },

   113→          { label: 'Outstanding', value: pkr(totalOutstanding), icon: Wallet, accent: true },
   114→        ].map((stat, i) => (

   116→            key={stat.label}
   117→            initial={{ opacity: 0, y: 8 }}
   118→            animate={{ opacity: 1, y: 0 }}

   120→            className="macos-card p-4"

   123→              <div>

   126→                  {stat.value}
   127→                </p>
   128→              </div>
   129→              <div className={`size-8 rounded-lg flex items-center justify-center ${stat.accent ? 'bg-amber-500/10' : 'bg-black/[0.03] dark:bg-white/[0.05]'}`}>
   130→                <stat.icon className={`size-4 ${stat.accent ? 'text-amber-500' : 'text-muted-foreground'}`} strokeWidth={1.5} />

   132→            </div>

   135→      </div>

   137→      {/* ── Tab Bar ── */}

   154→            </button>

   157→      </div>
   158→
   159→      {/* ── Tab Content ── */}
