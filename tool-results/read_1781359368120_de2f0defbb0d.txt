   182→   182→function getLevelColor(pct: number): string {
   183→   183→  if (pct > 50) return '#34c759'
   184→   184→  if (pct >= 25) return '#ff9f0a'
   185→   185→  return '#ff3b30'
   186→   186→}
   187→   187→
   188→   188→function getLevelBgClass(pct: number): string {
   189→   189→  if (pct > 50) return 'bg-[#34c759]'
   190→   190→  if (pct >= 25) return 'bg-[#ff9f0a]'
   191→   191→  return 'bg-[#ff3b30]'
   192→   192→}
   193→   193→
   194→   194→function getStatusClass(status: TankData['status']): string {
   195→   195→  if (status === 'online') return 'macos-status-online'
   196→   196→  if (status === 'warning') return 'macos-status-warning'
   197→   197→  return 'macos-status-error'
   198→   198→}
