   132→   132→// HELPER: Filter data to only allowed fields
   133→   133→// =====================================================
   134→   134→
   135→   135→function filterAllowedFields(
   136→   136→  data: Record<string, unknown>,
   137→   137→  deviceTypeKey: string,
   138→   138→  includeCommon: boolean = true
   139→   139→): Record<string, unknown> {
   140→   140→  const result: Record<string, unknown> = {}
   141→   141→  const allowedFields = new Set<string>()
   142→   142→
   143→   143→  if (includeCommon) {
   144→   144→    for (const f of COMMON_FIELDS) allowedFields.add(f)
   145→   145→  }
   146→   146→  for (const f of TYPE_SPECIFIC_FIELDS[deviceTypeKey] ?? []) allowedFields.add(f)
   147→   147→
   148→   148→  for (const [key, value] of Object.entries(data)) {
   149→   149→    if (allowedFields.has(key) && value !== undefined) {
   150→   150→      result[key] = value
   151→   151→    }
   152→   152→  }
   153→   153→
   154→   154→  return result
   155→   155→}

   168→   168→// =====================================================
   169→   169→// HELPER: Safely log a device event (non-blocking)
   170→   170→// The DeviceEventLog model has polymorphic FK relations to all
   171→   171→// device types on the deviceId field. SQLite enforces ALL FK
   172→   172→// constraints simultaneously, so inserting a record with a deviceId
   173→   173→// that only exists in one device table causes FK violations on
   174→   174→// the others. This helper silently catches these violations.
   175→   175→// =====================================================
   176→   176→
   177→   177→async function logDeviceEvent(data: {
   178→   178→  stationId: string
   179→   179→  deviceType: string
   180→   180→  deviceId: string
   181→   181→  eventType: string
   182→   182→  severity: string
   183→   183→  message: string
   184→   184→  dataJson?: string
   185→   185→}): Promise<void> {
   186→   186→  try {
   187→   187→    await db.deviceEventLog.create({ data })
   188→   188→  } catch (err) {
   189→   189→    // FK constraint violations are expected due to polymorphic relations.
   190→   190→    // Log but don't throw — event logging is non-critical.
   191→   191→    console.warn(
   192→   192→      'Device event log creation skipped (FK constraint):',
   193→   193→      err instanceof Prisma.PrismaClientKnownRequestError ? err.code : err instanceof Error ? err.message : String(err)
   194→   194→    )
   195→   195→  }
   196→   196→}
   197→   197→

   199→   199→// HELPER: Safely create a command queue entry
   200→   200→// Same FK constraint issue as event logs. Falls back to
   201→   201→// creating the record with a non-matching deviceId if needed.
   202→   202→// =====================================================
   203→   203→
   204→   204→async function createCommandEntry(data: {
   205→   205→  stationId: string
   206→   206→  deviceType: string
   207→   207→  deviceId: string
   208→   208→  command: string
   209→   209→  parameters: string | null
   210→   210→  priority: string
   211→   211→  maxRetries: number
   212→   212→  retryCount: number
   213→   213→  createdBy: string | null
   214→   214→}): Promise<{ id: string; [key: string]: unknown } | null> {
   215→   215→  try {
   216→   216→    const entry = await db.deviceCommandQueue.create({
   217→   217→      data: {
   218→   218→        stationId: data.stationId,
   219→   219→        deviceType: data.deviceType,
   220→   220→        deviceId: data.deviceId,
   221→   221→        command: data.command,
   222→   222→        parameters: data.parameters,
   223→   223→        priority: data.priority,
   224→   224→        status: 'queued',
   225→   225→        maxRetries: data.maxRetries,
   226→   226→        retryCount: data.retryCount,
   227→   227→        createdBy: data.createdBy,
   228→   228→      },
   229→   229→    })
   230→   230→    return entry as { id: string; [key: string]: unknown }
   231→   231→  } catch (err) {
   232→   232→    // FK constraint violation due to polymorphic relations
   233→   233→    if (err instanceof Prisma.PrismaClientKnownRequestError && err.code === 'P2003') {
   234→   234→      console.warn('Command queue creation skipped (FK constraint):', err.message)
   235→   235→      // Return a virtual command entry that wasn't persisted
   236→   236→      return {
   237→   237→        id: 'virtual-' + Date.now(),
   238→   238→        stationId: data.stationId,
   239→   239→        deviceType: data.deviceType,
   240→   240→        deviceId: data.deviceId,
   241→   241→        command: data.command,
   242→   242→        parameters: data.parameters,
   243→   243→        priority: data.priority,
   244→   244→        status: 'queued',
   245→   245→        maxRetries: data.maxRetries,
   246→   246→        retryCount: data.retryCount,
   247→   247→        createdBy: data.createdBy,
   248→   248→        persisted: false,
   249→   249→        reason: 'FK constraint - command stored in memory only',
   250→   250→      }
   251→   251→    }
   252→   252→    throw err
   253→   253→  }
   254→   254→}

   323→   323→async function safeDeleteRelated(deviceType: string, deviceId: string): Promise<void> {
   324→   324→  const operations = [

   328→   328→  ]
   329→   329→
   330→   330→  for (const op of operations) {
   331→   331→    try {
   332→   332→      await op()
   333→   333→    } catch (err) {
   334→   334→      console.warn('Related record deletion warning:', err instanceof Error ? err.message : String(err))
   335→   335→    }
   336→   336→  }

   339→   339→// =====================================================
   340→   340→// GET /api/devices?type=...
   341→   341→// =====================================================
   342→   342→
   343→   343→export async function GET(request: Request) {
   344→   344→  try {
   345→   345→    const { searchParams } = new URL(request.url)
   346→   346→    const type = searchParams.get('type')
   347→   347→
   348→   348→    if (!type) {
   349→   349→      return errorResponse(
   350→   350→        'Missing required query param "type". Valid values: ' +
   351→   351→          ALL_DEVICE_TYPE_KEYS.join(', ') +
   352→   352→          ', event_logs, command_queue, configs, overview',
   353→   353→        400
   354→   354→      )
   355→   355→    }
   356→   356→
   357→   357→    // -------------------------------------------------
   358→   358→    // 1) OVERVIEW - summary counts of all device types
   359→   359→    // -------------------------------------------------
   360→   360→    if (type === 'overview') {
   361→   361→      const stationId = await getFirstStationId()
   362→   362→      if (!stationId) {
   363→   363→        return errorResponse('No station found', 404)
   364→   364→      }
   365→   365→
   366→   366→      const counts: Record<string, number> = {}
   367→   367→      const statusBreakdown: Record<string, Record<string, number>> = {}
   368→   368→
   369→   369→      for (const deviceType of DEVICE_TYPES) {
   370→   370→        const model = deviceType.getModel()
   371→   371→        const count = await model.count({
   372→   372→          where: { stationId },
   373→   373→        })
   374→   374→        counts[deviceType.key] = count
   375→   375→
   376→   376→        // Get status breakdown using groupBy
   377→   377→        try {
   378→   378→          const grouped = await model.groupBy({
   379→   379→            by: ['status'],
   380→   380→            where: { stationId },
   381→   381→            _count: { status: true },
   382→   382→          })
   383→   383→          statusBreakdown[deviceType.key] = {}
   384→   384→          for (const g of grouped) {
   385→   385→            statusBreakdown[deviceType.key][String(g.status)] =
   386→   386→              Number((g as Record<string, unknown>)._count?.status ?? 0)
   387→   387→          }
   388→   388→        } catch {
   389→   389→          statusBreakdown[deviceType.key] = {}
   390→   390→        }
   391→   391→      }
   392→   392→
   393→   393→      // Additional counts for event logs and command queue
   394→   394→      const eventLogCount = await db.deviceEventLog.count({
   395→   395→        where: { stationId },
   396→   396→      })
   397→   397→      const commandQueueCount = await db.deviceCommandQueue.count({
   398→   398→        where: { stationId },
   399→   399→      })
   400→   400→
   401→   401→      // Recent critical events
   402→   402→      const recentCriticalEvents = await db.deviceEventLog.findMany({
   403→   403→        where: { stationId, severity: { in: ['error', 'critical'] } },
   404→   404→        orderBy: { createdAt: 'desc' },
   405→   405→        take: 5,
   406→   406→      })
   407→   407→
   408→   408→      // Pending commands count
   409→   409→      const pendingCommands = await db.deviceCommandQueue.count({
   410→   410→        where: { stationId, status: 'queued' },
   411→   411→      })
   412→   412→
   413→   413→      return NextResponse.json({
   414→   414→        overview: {
   415→   415→          stationId,
   416→   416→          deviceCounts: counts,
   417→   417→          statusBreakdown,
   418→   418→          totalDevices: Object.values(counts).reduce((sum, c) => sum + c, 0),
   419→   419→          eventLogCount,
   420→   420→          commandQueueCount,
   421→   421→          pendingCommands,
   422→   422→          recentCriticalEvents,
   423→   423→        },
   424→   424→      })
   425→   425→    }
   426→   426→
   427→   427→    // -------------------------------------------------
   428→   428→    // 2) EVENT_LOGS - recent DeviceEventLog entries
   429→   429→    // -------------------------------------------------
   430→   430→    if (type === 'event_logs') {
   431→   431→      const stationId = await getFirstStationId()
   432→   432→      if (!stationId) {
   433→   433→        return errorResponse('No station found', 404)
   434→   434→      }
   435→   435→
   436→   436→      const limit = Math.min(Math.max(parseInt(searchParams.get('limit') || '50'), 1), 500)
   437→   437→      const offset = Math.max(parseInt(searchParams.get('offset') || '0'), 0)
   438→   438→      const deviceTypeFilter = searchParams.get('deviceType')
   439→   439→      const severity = searchParams.get('severity')
   440→   440→      const eventType = searchParams.get('eventType')
   441→   441→      const deviceId = searchParams.get('deviceId')
   442→   442→
   443→   443→      const where: Record<string, unknown> = { stationId }
   444→   444→      if (deviceTypeFilter) where.deviceType = deviceTypeFilter
   445→   445→      if (severity) where.severity = severity
   446→   446→      if (eventType) where.eventType = eventType
   447→   447→      if (deviceId) where.deviceId = deviceId
   448→   448→
   449→   449→      const [eventLogs, total] = await Promise.all([
   450→   450→        db.deviceEventLog.findMany({
   451→   451→          where,
   452→   452→          orderBy: { createdAt: 'desc' },
   453→   453→          take: limit,
   454→   454→          skip: offset,
   455→   455→        }),
   456→   456→        db.deviceEventLog.count({ where }),
   457→   457→      ])
   458→   458→
   459→   459→      return NextResponse.json({
   460→   460→        eventLogs,
   461→   461→        pagination: { total, limit, offset, hasMore: offset + limit < total },
   462→   462→      })
   463→   463→    }
   464→   464→
   465→   465→    // -------------------------------------------------
   466→   466→    // 3) COMMAND_QUEUE - DeviceCommandQueue entries
   467→   467→    // -------------------------------------------------
   468→   468→    if (type === 'command_queue') {
   469→   469→      const stationId = await getFirstStationId()
   470→   470→      if (!stationId) {
   471→   471→        return errorResponse('No station found', 404)
   472→   472→      }
   473→   473→
   474→   474→      const limit = Math.min(Math.max(parseInt(searchParams.get('limit') || '50'), 1), 500)
   475→   475→      const offset = Math.max(parseInt(searchParams.get('offset') || '0'), 0)
   476→   476→      const deviceTypeFilter = searchParams.get('deviceType')
   477→   477→      const statusFilter = searchParams.get('status')
   478→   478→      const priority = searchParams.get('priority')
   479→   479→      const deviceId = searchParams.get('deviceId')
   480→   480→
   481→   481→      const where: Record<string, unknown> = { stationId }
   482→   482→      if (deviceTypeFilter) where.deviceType = deviceTypeFilter
   483→   483→      if (statusFilter) where.status = statusFilter
   484→   484→      if (priority) where.priority = priority
   485→   485→      if (deviceId) where.deviceId = deviceId
   486→   486→
   487→   487→      const [commands, total] = await Promise.all([
   488→   488→        db.deviceCommandQueue.findMany({
   489→   489→          where,
   490→   490→          orderBy: [{ priority: 'desc' }, { createdAt: 'asc' }],
   491→   491→          take: limit,
   492→   492→          skip: offset,
   493→   493→        }),
   494→   494→        db.deviceCommandQueue.count({ where }),
   495→   495→      ])
   496→   496→
   497→   497→      return NextResponse.json({
   498→   498→        commands,
   499→   499→        pagination: { total, limit, offset, hasMore: offset + limit < total },
   500→   500→      })
   501→   501→    }
   502→   502→
   503→   503→    // -------------------------------------------------
   504→   504→    // 4) CONFIGS - DeviceConfig for a specific device
   505→   505→    // -------------------------------------------------
   506→   506→    if (type === 'configs') {
   507→   507→      const deviceTypeParam = searchParams.get('deviceType')
   508→   508→      const deviceId = searchParams.get('deviceId')
   509→   509→
   510→   510→      if (!deviceTypeParam || !deviceId) {
   511→   511→        return errorResponse('Missing required params: deviceType and deviceId', 400)
   512→   512→      }
   513→   513→
   514→   514→      const configs = await db.deviceConfig.findMany({
   515→   515→        where: { deviceType: deviceTypeParam, deviceId },
   516→   516→        orderBy: { configKey: 'asc' },
   517→   517→      })
   518→   518→
   519→   519→      return NextResponse.json({ configs, deviceType: deviceTypeParam, deviceId })
   520→   520→    }
   521→   521→
   522→   522→    // -------------------------------------------------
   523→   523→    // 5) DEVICE TYPE - fetch all devices of a specific type
   524→   524→    // -------------------------------------------------
   525→   525→    const deviceInfo = getDeviceInfo(type)
   526→   526→    if (!deviceInfo) {
   527→   527→      return errorResponse(
   528→   528→        `Invalid device type "${type}". Valid types: ${ALL_DEVICE_TYPE_KEYS.join(', ')}, event_logs, command_queue, configs, overview`,
   529→   529→        400
   530→   530→      )
   531→   531→    }
   532→   532→
   533→   533→    const stationId = await getFirstStationId()
   534→   534→    if (!stationId) {
   535→   535→      return errorResponse('No station found', 404)
   536→   536→    }
   537→   537→
   538→   538→    const limit = Math.min(Math.max(parseInt(searchParams.get('limit') || '100'), 1), 500)
   539→   539→    const offset = Math.max(parseInt(searchParams.get('offset') || '0'), 0)
   540→   540→    const statusFilter = searchParams.get('status')
   541→   541→    const isActiveFilter = searchParams.get('isActive')
   542→   542→
   543→   543→    const where: Record<string, unknown> = {
   544→   544→      stationId,
   545→   545→      ...(statusFilter ? { status: statusFilter } : {}),
   546→   546→      ...(isActiveFilter !== null ? { isActive: isActiveFilter === 'true' } : {}),
   547→   547→    }
   548→   548→
   549→   549→    const include = {
   550→   550→      eventLogs: {
   551→   551→        orderBy: { createdAt: 'desc' as const },
   552→   552→        take: 5,
   553→   553→      },
   554→   554→      commandQueue: {
   555→   555→        where: { status: 'queued' },
   556→   556→        orderBy: [{ priority: 'desc' as const }, { createdAt: 'asc' as const }],
   557→   557→        take: 5,
   558→   558→      },
   559→   559→      configs: {
   560→   560→        where: { isEditable: true },
   561→   561→        orderBy: { configKey: 'asc' as const },
   562→   562→      },
   563→   563→    }
   564→   564→
   565→   565→    const model = deviceInfo.getModel()
   566→   566→    const [devices, total] = await Promise.all([
   567→   567→      model.findMany({
   568→   568→        where,
   569→   569→        include,
   570→   570→        orderBy: { createdAt: 'desc' },
   571→   571→        take: limit,
   572→   572→        skip: offset,
   573→   573→      }),
   574→   574→      model.count({ where }),
   575→   575→    ])
   576→   576→
   577→   577→    return NextResponse.json({
   578→   578→      devices,
   579→   579→      deviceType: type,
   580→   580→      label: deviceInfo.label,
   581→   581→      pagination: { total, limit, offset, hasMore: offset + limit < total },
   582→   582→    })
   583→   583→  } catch (error) {
   584→   584→    console.error('Devices GET error:', error)
   585→   585→    return errorResponse(
   586→   586→      'Failed to fetch device data',
   587→   587→      500,
   588→   588→      error instanceof Error ? error.message : undefined
   589→   589→    )
   590→   590→  }
   591→   591→}
