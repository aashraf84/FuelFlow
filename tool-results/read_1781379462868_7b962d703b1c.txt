   173→   173→function timeAgo(date: Date | string | null): string {
   174→   174→  if (!date) return 'N/A'
   175→   175→  const d = typeof date === 'string' ? new Date(date) : date
   176→   176→  const diff = Date.now() - d.getTime()
   177→   177→  if (diff < 60000) return 'just now'
   178→   178→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`
   179→   179→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`
   180→   180→  return `${Math.floor(diff / 86400000)}d ago`
   181→   181→}
