   171→
   172→export default function SecurityCenterModule() {
   173→  const [activeTab, setActiveTab] = useState('overview')
   174→  const [auditFilter, setAuditFilter] = useState('all')
   175→
   176→  // Data state
   177→  const [overview, setOverview] = useState<SecurityOverview | null>(null)
   178→  const [sessions, setSessions] = useState<SecuritySessionData[]>([])
   179→  const [permissions, setPermissions] = useState<RBACPermissionData[]>([])
   180→  const [securityAlerts, setSecurityAlerts] = useState<SecurityAlertData[]>([])
   181→  const [encryptionKeys, setEncryptionKeys] = useState<EncryptionKeyData[]>([])
   182→  const [auditLogs, setAuditLogs] = useState<AuditLogData[]>([])
   183→
   184→  // Loading & error state
   185→  const [loading, setLoading] = useState(true)
   186→  const [error, setError] = useState<string | null>(null)
   187→  const [actionLoading, setActionLoading] = useState<string | null>(null)
   188→
   189→  const fetchData = useCallback(async () => {
   190→    setLoading(true)
   191→    setError(null)
   192→    try {
   193→      const [overviewRes, sessionsRes, rbacRes, alertsRes, encryptionRes, auditRes] = await Promise.all([
   194→        fetch('/api/security'),
   195→        fetch('/api/security?section=sessions'),
   196→        fetch('/api/security?section=rbac'),
   197→        fetch('/api/security?section=alerts'),
   198→        fetch('/api/security?section=encryption'),
   199→        fetch('/api/audit?limit=50'),
   200→      ])
   201→
   202→      if (!overviewRes.ok) throw new Error('Failed to fetch security data')
   203→
   204→      const overviewData = await overviewRes.json()
   205→      const sessionsData = await sessionsRes.json()
   206→      const rbacData = await rbacRes.json()
   207→      const alertsData = await alertsRes.json()
   208→      const encryptionData = await encryptionRes.json()
   209→      const auditData = await auditRes.json()
   210→
   211→      setOverview(overviewData.overview || null)
   212→      setSessions(sessionsData.sessions || [])
   213→      setPermissions(rbacData.permissions || [])
   214→      setSecurityAlerts(alertsData.alerts || [])
   215→      setEncryptionKeys(encryptionData.keys || [])
   216→      setAuditLogs(auditData.logs || [])
   217→    } catch (err) {
   218→      setError(err instanceof Error ? err.message : 'Unknown error')
   219→    } finally {
   220→      setLoading(false)
   221→    }
   222→  }, [])
   223→
   224→  useEffect(() => {
   225→    fetchData()
   226→  }, [fetchData])
   227→
   228→  const handleRevokeSession = async (sessionId: string) => {
   229→    setActionLoading(sessionId)
   230→    try {
   231→      const res = await fetch('/api/security', {
   232→        method: 'POST',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({ action: 'revoke_session', sessionId }),
   235→      })
   236→      if (res.ok) {
   237→        setSessions(prev => prev.filter(s => s.id !== sessionId))
   238→      }
   239→    } catch (err) {
   240→      console.error('Revoke session error:', err)
   241→    } finally {
   242→      setActionLoading(null)
   243→    }
   244→  }
   245→
   246→  const handleResolveAlert = async (alertId: string) => {
   247→    setActionLoading(alertId)
   248→    try {
   249→      const res = await fetch('/api/security', {
   250→        method: 'POST',
   251→        headers: { 'Content-Type': 'application/json' },
   252→        body: JSON.stringify({ action: 'resolve_alert', alertId }),
   253→      })
   254→      if (res.ok) {
   255→        setSecurityAlerts(prev => prev.map(a => a.id === alertId ? { ...a, isResolved: true, resolvedAt: new Date().toISOString() } : a))
   256→      }
   257→    } catch (err) {
   258→      console.error('Resolve alert error:', err)
   259→    } finally {
   260→      setActionLoading(null)
   261→    }
   262→  }
   263→
   264→  // Determine threat level based on unresolved alerts
   265→  const threatLevel = (() => {
   266→    const unresolvedCritical = securityAlerts.filter(a => !a.isResolved && (a.severity === 'critical' || a.severity === 'error')).length
   267→    const unresolvedWarning = securityAlerts.filter(a => !a.isResolved && a.severity === 'warning').length
   268→    if (unresolvedCritical > 0) return 'high'
   269→    if (unresolvedWarning > 2) return 'medium'
   270→    return 'low'
   271→  })()
   272→
   273→  const threat = THREAT_MAP[threatLevel]
   274→
   275→  const filteredLogs = auditFilter === 'all'
   276→    ? auditLogs
   277→    : auditLogs.filter(l => {
   278→        if (auditFilter === 'critical') return l.action.toLowerCase().includes('fail') || l.action.toLowerCase().includes('delete')
   279→        if (auditFilter === 'warning') return l.action.toLowerCase().includes('update') || l.action.toLowerCase().includes('warning')
   280→        return true
   281→      })
   282→
   283→  // Build RBAC matrix from permissions
   284→  const rbacMatrix = (() => {
   285→    const roleMap = new Map<string, Map<string, string>>()
   286→    permissions.forEach(p => {
   287→      if (!roleMap.has(p.role)) roleMap.set(p.role, new Map())
   288→      const modules = roleMap.get(p.role)!
   289→      const current = modules.get(p.module)
   290→      if (p.isAllowed) {
   291→        if (p.action === 'delete' || p.action === 'approve' || p.action === 'configure') {
   292→          modules.set(p.module, 'full')
   293→        } else if (p.action === 'write' && current !== 'full') {
   294→          modules.set(p.module, 'edit')
   295→        } else if (p.action === 'read' && !current) {
   296→          modules.set(p.module, 'view')
   297→        }
   298→      } else if (!current) {
   299→        modules.set(p.module, 'none')
   300→      }
   301→    })
   302→    return Array.from(roleMap.entries()).map(([role, modules]) => ({
   303→      role,
   304→      modules: Object.fromEntries(modules),
   305→    }))
   306→  })()
   307→
   308→  // Compute compliance score based on resolved alerts and active keys
   309→  const complianceScore = (() => {
   310→    if (!overview) return 0
   311→    const total = securityAlerts.length || 1
   312→    const resolved = securityAlerts.filter(a => a.isResolved).length
   313→    const hasActiveKeys = overview.activeEncryptionKeys > 0
   314→    const base = Math.round((resolved / total) * 100)
   315→    return hasActiveKeys ? Math.min(100, base + 10) : base
   316→  })()
   317→
   318→  if (error) {
   319→    return (
   320→      <div className="macos-card p-8 text-center">
   321→        <AlertTriangle className="size-8 text-red-500 mx-auto mb-3" />
   322→        <p className="text-[13px] font-semibold text-red-500 mb-1">Failed to load security data</p>
   323→        <p className="text-[12px] text-muted-foreground mb-3">{error}</p>
   324→        <button onClick={fetchData} className="macos-btn gap-1.5">
   325→          <RefreshCw className="size-3.5" /> Retry
   326→        </button>
   327→      </div>
   328→    )
   329→  }
   330→
   331→  return (
   332→    <div className="space-y-4" style={{ fontStyle: 'normal !important' }}>
   333→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   334→        <div className="overflow-x-auto pb-1">
   335→          <TabsList className="h-8 w-max gap-0.5 bg-muted/50">
   336→            <TabsTrigger value="overview" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Overview</TabsTrigger>
   337→            <TabsTrigger value="audit" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Audit Log</TabsTrigger>
   338→            <TabsTrigger value="sessions" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Sessions</TabsTrigger>
   339→            <TabsTrigger value="rbac" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">RBAC</TabsTrigger>
   340→            <TabsTrigger value="alerts" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Alerts</TabsTrigger>
   341→            <TabsTrigger value="encryption" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Encryption</TabsTrigger>
   342→          </TabsList>
   343→        </div>
   344→
   345→        {/* ── Overview ── */}
   346→        <TabsContent value="overview" className="mt-4 space-y-4">
   347→          {loading ? (
   348→            <div className="space-y-4">
   349→              <div className="macos-card p-5 animate-pulse">
   350→                <div className="flex items-center gap-4">
   351→                  <div className="size-14 rounded-2xl bg-muted/40" />
   352→                  <div className="flex-1 space-y-2">
   353→                    <div className="h-4 rounded bg-muted/40 w-48" />
   354→                    <div className="h-3 rounded bg-muted/40 w-64" />
   355→                  </div>
   356→                  <div className="size-14 rounded bg-muted/40" />
   357→                </div>
   358→              </div>
   359→              <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   360→                {Array.from({ length: 4 }).map((_, i) => (
   361→                  <div key={i} className="macos-card p-4 animate-pulse">
   362→                    <div className="h-3 rounded bg-muted/40 w-24 mb-2" />
   363→                    <div className="h-8 rounded bg-muted/40 w-12" />
   364→                  </div>
   365→                ))}
   366→              </div>
   367→            </div>
   368→          ) : (
   369→            <>
   370→              {/* Threat Level Banner */}
   371→              <div className="macos-card p-5">
   372→                <div className="flex items-center gap-4">
   373→                  <div className="size-14 rounded-2xl bg-emerald-500/10 flex items-center justify-center shrink-0">
   374→                    <Shield className="size-7 text-emerald-500" />
   375→                  </div>
   376→                  <div className="flex-1">
   377→                    <div className="flex items-center gap-2 mb-1">
   378→                      <h2 className="text-[15px] font-semibold">Security Status</h2>
   379→                      <div className="flex items-center gap-1.5">
   380→                        <div className={`macos-status-dot ${threat.dot}`} />
   381→                        <span className={`text-[11px] font-semibold ${threat.color}`}>Threat Level: {threat.label}</span>
   382→                      </div>
   383→                    </div>
   384→                    <p className="text-[12px] text-muted-foreground">
   385→                      {overview?.unresolvedAlerts ? `${overview.unresolvedAlerts} unresolved alerts` : 'All security services operating normally. No active threats detected.'}
   386→                    </p>
   387→                  </div>
   388→                  <div className="text-center">
   389→                    <div className="text-3xl font-bold text-emerald-600 dark:text-emerald-400">{complianceScore}</div>
   390→                    <div className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold">Compliance</div>
   391→                  </div>
   392→                </div>
   393→              </div>
   394→
   395→              {/* Stats Grid */}
   396→              <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   397→                {[
   398→                  { label: 'Active Sessions', value: overview?.activeSessions ?? 0, icon: Monitor, accent: '#34c759' },
   399→                  { label: 'Unresolved Alerts', value: overview?.unresolvedAlerts ?? 0, icon: Ban, accent: '#ff3b30' },
   400→                  { label: 'Last Backup', value: overview?.lastBackup ? new Date(overview.lastBackup.completedAt || '').toLocaleDateString('en-PK', { month: 'short', day: 'numeric' }) : 'Never', icon: HardDrive, accent: '#ff9f0a', isText: true },
   401→                  { label: 'Encryption Keys', value: overview?.activeEncryptionKeys ?? 0, icon: Key, accent: '#e8930c' },
   402→                ].map((stat) => (
   403→                  <div key={stat.label} className="macos-card p-4">
   404→                    <div className="flex items-center justify-between">
   405→                      <div>
   406→                        <p className="text-[11px] uppercase tracking-wider text-muted-foreground font-semibold">{stat.label}</p>
   407→                        <p className={`text-2xl font-bold mt-0.5 ${stat.isText ? 'text-[16px]' : ''}`} style={{ color: stat.accent }}>{stat.value}</p>
   408→                      </div>
   409→                      <div className="size-9 rounded-[10px] flex items-center justify-center" style={{ backgroundColor: `${stat.accent}12` }}>
   410→                        <stat.icon className="size-4" style={{ color: stat.accent }} />
   411→                      </div>
   412→                    </div>
   413→                  </div>
   414→                ))}
   415→              </div>
   416→
   417→              {/* Recent Alerts Preview */}
   418→              <div>
   419→                <div className="macos-section-header">Recent Alerts</div>
   420→                <div className="macos-group mt-1">
   421→                  {securityAlerts.length === 0 ? (
   422→                    <div className="macos-group-item text-center py-6 text-[13px] text-muted-foreground">No security alerts</div>
   423→                  ) : (
   424→                    securityAlerts.slice(0, 3).map(alert => {
   425→                      const severity = SEVERITY_MAP[alert.severity] || alert.severity
   426→                      const cfg = ALERT_TYPE_MAP[severity] || ALERT_TYPE_MAP['info']
   427→                      return (
   428→                        <div key={alert.id} className="macos-group-item flex items-center gap-3">
   429→                          <div className={`macos-status-dot ${cfg.dot}`} />
   430→                          <div className="flex-1 min-w-0">
   431→                            <div className="flex items-center gap-2">
   432→                              <span className="text-[13px] font-medium">{alert.title}</span>
   433→                              {alert.isResolved && (
   434→                                <Badge variant="outline" className="text-[9px] px-1.5 py-0 bg-emerald-500/10 text-emerald-600 border-emerald-500/20">Resolved</Badge>
   435→                              )}
   436→                            </div>
   437→                            <p className="text-[11px] text-muted-foreground truncate">{alert.description || alert.alertType}</p>
   438→                          </div>
   439→                          <span className="text-[11px] text-muted-foreground whitespace-nowrap">
   440→                            {new Date(alert.createdAt).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })}
   441→                          </span>
   442→                        </div>
   443→                      )
   444→                    })
   445→                  )}
   446→                </div>
   447→              </div>
   448→            </>
   449→          )}
   450→        </TabsContent>
   451→
   452→        {/* ── Audit Log ── */}
   453→        <TabsContent value="audit" className="mt-4 space-y-3">
   454→          <div className="flex items-center gap-3">
   455→            <div className="relative flex-1 max-w-xs">
   456→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   457→              <input className="macos-search pl-8 py-1.5 text-[12px]" placeholder="Search audit logs..." />
   458→            </div>
   459→            <div className="flex items-center gap-1.5">
   460→              {['all', 'info', 'warning', 'critical'].map(f => (
   461→                <button
   462→                  key={f}
   463→                  onClick={() => setAuditFilter(f)}
   464→                  className={`text-[11px] px-2.5 py-1 rounded-md font-medium transition-colors capitalize ${
   465→                    auditFilter === f
   466→                      ? 'bg-teal-500/15 text-teal-600 dark:text-teal-400'
   467→                      : 'text-muted-foreground hover:bg-black/[0.03] dark:hover:bg-white/[0.04]'
   468→                  }`}
   469→                >
   470→                  {f}
   471→                </button>
   472→              ))}
   473→            </div>
   474→          </div>
   475→
   476→          <div className="macos-card overflow-hidden">
   477→            <div className="max-h-[420px] overflow-y-auto">
   478→              <table className="macos-table">
   479→                <thead>
   480→                  <tr>
   481→                    <th>Time</th>
   482→                    <th>User</th>
   483→                    <th>Action</th>
   484→                    <th>Entity</th>
