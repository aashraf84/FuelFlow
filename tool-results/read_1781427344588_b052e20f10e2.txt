   143→// ── Helpers ──────────────────────────────────────────────
   144→
   145→function formatPKR(amount: number): string {
   146→  if (amount >= 1000000) return `PKR ${(amount / 1000000).toFixed(2)}M`
   147→  if (amount >= 1000) return `PKR ${(amount / 1000).toFixed(0)}K`
   148→  return `PKR ${amount.toLocaleString()}`
   149→}

   151→const fadeUp = {

   153→  animate: { opacity: 1, y: 0 },
   154→  transition: { duration: 0.3, ease: [0.25, 0.1, 0.25, 1] },
   155→}
   156→
   157→function Skeleton({ className }: { className?: string }) {
   158→  return <div className={`animate-pulse rounded-md bg-muted ${className ?? ''}`} />
   159→}

   161→// ── Component ────────────────────────────────────────────
   162→
   163→export default function EVCNGModule() {
   164→  const [activeTab, setActiveTab] = useState<'ev' | 'sessions' | 'cng' | 'revenue'>('ev')
   165→
   166→  // Data states
   167→  const [evChargers, setEvChargers] = useState<EVChargerAPI[]>([])
   168→  const [chargingSessions, setChargingSessions] = useState<ChargingSessionAPI[]>([])

   173→
   174→  // Loading / error
   175→  const [loading, setLoading] = useState(true)
   176→  const [error, setError] = useState<string | null>(null)
   177→
   178→  const fetchData = useCallback(async () => {
   179→    setLoading(true)
   180→    setError(null)
   181→    try {
   182→      const [evCngRes, sessionsRes] = await Promise.allSettled([
   183→        fetch('/api/ev-cng'),
   184→        fetch('/api/charging-sessions'),
   185→      ])
   186→
   187→      if (evCngRes.status === 'fulfilled') {
   188→        const data = await evCngRes.value.json()
   189→        setEvChargers(data.evChargers || [])
   190→        setCngDispensers(data.cngDispensers || [])
   191→        setChargingSessions(data.chargingSessions || [])
   192→        if (data.summary) {
   193→          setEvSummary(data.summary.ev || null)
   194→          setCngSummary(data.summary.cng || null)
   195→          setChargingSummary(data.summary.charging || null)
   196→        }
   197→      }
   198→
   199→      if (sessionsRes.status === 'fulfilled') {
   200→        const data = await sessionsRes.value.json()
   201→        // Only use if ev-cng didn't return sessions, or merge
   202→        if (evCngRes.status !== 'fulfilled' || (data.sessions && data.sessions.length > 0)) {
   203→          setChargingSessions(data.sessions || [])
   204→        }
   205→      }
   206→    } catch (err) {
   207→      console.error('EVCNG fetchData error:', err)
   208→      setError(err instanceof Error ? err.message : 'Failed to load EV/CNG data')
   209→    } finally {
   210→      setLoading(false)
   211→    }
   212→  }, [])
   213→
   214→  useEffect(() => {
   215→    fetchData()
   216→  }, [fetchData])
   217→
   218→  // Compute revenue data from charging sessions
   219→  const revenueData = (() => {
   220→    if (!chargingSessions.length) return []

   222→    for (const s of chargingSessions) {
   223→      const date = new Date(s.startedAt)
   224→      const key = date.toLocaleString('en', { month: 'short' })
   225→      if (!monthMap[key]) monthMap[key] = { ev: 0, cng: 0 }
   226→      monthMap[key].ev += s.totalAmount || 0
   227→    }
   228→    return Object.entries(monthMap).map(([month, data]) => ({
   229→      month,
   230→      ev: Math.round(data.ev),
   231→      cng: Math.round(data.cng),
   232→    }))
   233→  })()
   234→
   235→  const totalEVCRevenue = evSummary?.totalRevenue || revenueData.reduce((s, d) => s + d.ev, 0)
   236→  const totalCNGRevenue = revenueData.reduce((s, d) => s + d.cng, 0)
   237→
   238→  const tabs = [
   239→    { key: 'ev' as const, label: 'EV Chargers' },
   240→    { key: 'sessions' as const, label: 'Sessions' },
   241→    { key: 'cng' as const, label: 'CNG' },
   242→    { key: 'revenue' as const, label: 'Revenue' },
   243→  ]
   244→
   245→  // ── Loading state ──
   246→  if (loading) {
   247→    return (
   248→      <div className="space-y-5" style={{ fontStretch: 'normal' }}>
   249→        <div className="grid grid-cols-3 sm:grid-cols-6 gap-3">
   250→          {Array.from({ length: 6 }).map((_, i) => (
   251→            <div key={i} className="macos-card p-3 text-center">
   252→              <Skeleton className="h-6 w-8 mx-auto mb-1" />
   253→              <Skeleton className="h-3 w-12 mx-auto" />
   254→            </div>
   255→          ))}
   256→        </div>
   257→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   258→          {Array.from({ length: 6 }).map((_, i) => (
   259→            <div key={i} className="macos-card p-4">
   260→              <Skeleton className="h-4 w-24 mb-3" />
   261→              <Skeleton className="h-3 w-32 mb-2" />
   262→              <Skeleton className="h-3 w-20" />
   263→            </div>
   264→          ))}
   265→        </div>
   266→        <div className="macos-card p-4 flex items-center justify-center gap-2 text-muted-foreground">
   267→          <Loader2 className="size-4 animate-spin" />
   268→          <span className="text-[13px]">Loading EV/CNG data from database…</span>
   269→        </div>
   270→      </div>
   271→    )
   272→  }
   273→
   274→  // ── Error state ──
   275→  if (error) {
   276→    return (

   279→        <p className="text-[14px] font-semibold mb-1">Failed to Load EV/CNG Data</p>
   280→        <p className="text-[12px] text-muted-foreground mb-4">{error}</p>
   281→        <button onClick={fetchData} className="macos-btn text-[12px] px-4 py-1.5 flex items-center gap-2 mx-auto">
   282→          <RefreshCw className="size-3.5" /> Retry
   283→        </button>
   284→      </div>
   285→    )
   286→  }
   287→
   288→  return (
   289→    <div className="space-y-5" style={{ fontStretch: 'normal' }}>
   290→      {/* Tab Bar */}
   291→      <div className="flex items-center gap-1 rounded-xl bg-black/[0.03] dark:bg-white/[0.04] p-1 w-fit overflow-x-auto">
   292→        {tabs.map((tab) => (
   293→          <button
   294→            key={tab.key}
   295→            onClick={() => setActiveTab(tab.key)}
   296→            className={`px-4 py-1.5 rounded-lg text-[13px] font-medium transition-all duration-150 whitespace-nowrap ${
   297→              activeTab === tab.key
   298→                ? 'bg-card text-foreground shadow-sm'
   299→                : 'text-muted-foreground hover:text-foreground'
   300→            }`}
   301→          >
   302→            {tab.label}
   303→          </button>
   304→        ))}
   305→      </div>
   306→
   307→      <AnimatePresence mode="wait">
   308→        {/* ── EV Chargers Tab ── */}
   309→        {activeTab === 'ev' && (
   310→          <motion.div key="ev" {...fadeUp} className="space-y-5">
   311→            {/* EV Summary Row */}
   312→            <div className="grid grid-cols-3 sm:grid-cols-6 gap-3">
   313→              {[
   314→                { label: 'Total', value: evSummary?.total ?? evChargers.length, color: '' },
   315→                { label: 'Available', value: evSummary?.available ?? evChargers.filter(c => c.status === 'available').length, color: 'text-emerald-500' },
   316→                { label: 'Charging', value: evSummary?.charging ?? evChargers.filter(c => c.status === 'charging').length, color: 'text-teal-500' },
   317→                { label: 'Occupied', value: evSummary?.occupied ?? evChargers.filter(c => c.status === 'occupied').length, color: 'text-orange-500' },
   318→                { label: 'Offline', value: evSummary?.offline ?? evChargers.filter(c => c.status === 'offline').length, color: 'text-slate-400' },
   319→                { label: 'Error', value: evSummary?.error ?? evChargers.filter(c => c.status === 'error').length, color: 'text-red-500' },
   320→              ].map((item) => (
   321→                <div key={item.label} className="macos-card p-3 text-center">
   322→                  <p className={`text-[18px] font-bold ${item.color}`}>{item.value}</p>
   323→                  <p className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider">{item.label}</p>
   324→                </div>
   325→              ))}
   326→            </div>
   327→
   328→            {/* Charger Grid */}
   329→            {evChargers.length > 0 ? (
   330→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   331→                {evChargers.map((charger, i) => {
   332→                  const cfg = chargerStatusConfig[charger.status] || chargerStatusConfig.offline
   333→                  const activeSession = charger.recentSessions?.find(s => s.status === 'active')
   334→                  return (
   335→                    <motion.div
   336→                      key={charger.id}
   337→                      initial={{ opacity: 0, y: 8 }}
   338→                      animate={{ opacity: 1, y: 0 }}
   339→                      transition={{ delay: i * 0.04, duration: 0.25 }}
   340→                      className="macos-card p-4"
   341→                    >
   342→                      <div className="flex items-center justify-between mb-3">
   343→                        <div className="flex items-center gap-2">
   344→                          <Zap className={`size-4 ${cfg.color} ${charger.status === 'charging' ? 'animate-pulse' : ''}`} />
   345→                          <h4 className="text-[13px] font-semibold">{charger.name}</h4>
   346→                        </div>
   347→                        <div className="flex items-center gap-1.5">
   348→                          <div className={`macos-status-dot ${cfg.dot}`} />
   349→                          <span className={`text-[11px] font-medium ${cfg.color}`}>{cfg.label}</span>
   350→                        </div>
   351→                      </div>
   352→
   353→                      <div className="grid grid-cols-2 gap-2 text-[12px] mb-3">
   354→                        <div className="flex items-center gap-1.5 text-muted-foreground">
   355→                          <Plug className="size-3" />
   356→                          <span>{charger.connectorType}</span>
   357→                        </div>
   358→                        <div className="flex items-center gap-1.5 text-muted-foreground">
   359→                          <Battery className="size-3" />
   360→                          <span>{charger.maxPowerKw} kW</span>

   362→                        <div className="flex items-center gap-1.5 text-muted-foreground col-span-2">
   363→                          <Cpu className="size-3" />
   364→                          <span>{charger.station?.name || 'Unknown Station'}</span>
   365→                        </div>
   366→                      </div>
   367→
   368→                      {/* Active Session */}
   369→                      {activeSession && (
   370→                        <div className="rounded-xl bg-teal-500/5 border border-teal-500/10 p-3 space-y-2">
   371→                          <div className="flex items-center justify-between">
   372→                            <span className="text-[11px] font-semibold text-teal-600 dark:text-teal-400 uppercase tracking-wider">Active Session</span>
   373→                            <span className="text-[11px] text-muted-foreground">
   374→                              {new Date(activeSession.startedAt).toLocaleTimeString('en', { hour: '2-digit', minute: '2-digit' })}

   376→                          </div>
   377→                          <div className="grid grid-cols-2 gap-x-4 gap-y-1.5 text-[12px]">
   378→                            <div className="flex justify-between">
   379→                              <span className="text-muted-foreground">Vehicle</span>
   380→                              <span className="font-mono font-medium">{activeSession.vehiclePlate || 'N/A'}</span>
   381→                            </div>
   382→                            <div className="flex justify-between">
   383→                              <span className="text-muted-foreground">Energy</span>
   384→                              <span className="font-medium">{activeSession.energyKwh.toFixed(1)} kWh</span>
   385→                            </div>
   386→                            <div className="flex justify-between">
   387→                              <span className="text-muted-foreground">SoC</span>
   388→                              <span className="font-medium">
   389→                                {activeSession.socStart ?? '?'}% → {activeSession.socCurrent ?? '?'}%
   390→                              </span>
   391→                            </div>
   392→                            <div className="flex justify-between">
   393→                              <span className="text-muted-foreground">Amount</span>
   394→                              <span className="font-semibold text-teal-600 dark:text-teal-400">{formatPKR(activeSession.totalAmount)}</span>

   397→                          {/* SoC Progress */}
   398→                          {activeSession.socCurrent != null && (
   399→                            <div className="space-y-1">
   400→                              <div className="h-1.5 w-full rounded-full bg-muted overflow-hidden">
   401→                                <div
   402→                                  className="h-full rounded-full bg-teal-500 transition-all"
   403→                                  style={{ width: `${activeSession.socCurrent}%` }}
   404→                                />
   405→                              </div>
   406→                            </div>
   407→                          )}
   408→                        </div>
   409→                      )}
   410→                    </motion.div>
   411→                  )
   412→                })}
   413→              </div>
   414→            ) : (
   415→              <div className="macos-card p-6 text-center text-[13px] text-muted-foreground">
   416→                No EV chargers configured
   417→              </div>
   418→            )}
   419→          </motion.div>
   420→        )}
   421→
   422→        {/* ── Sessions Tab ── */}
   423→        {activeTab === 'sessions' && (
   424→          <motion.div key="sessions" {...fadeUp} className="space-y-4">
   425→            <div className="flex items-center justify-between">
   426→              <div className="flex items-center gap-2">
   427→                <Clock className="size-4 text-teal-500" />
   428→                <h3 className="text-[13px] font-semibold">Charging Sessions</h3>
   429→                <Badge variant="outline" className="text-[11px] px-2 py-0.5">
   430→                  {chargingSessions.filter(s => s.status === 'active').length} active
   431→                </Badge>
   432→              </div>
   433→              <button onClick={fetchData} className="macos-btn-secondary text-[11px] px-2.5 py-1 flex items-center gap-1">
   434→                <RefreshCw className="size-3" /> Refresh
   435→              </button>
   436→            </div>
   437→
   438→            <div className="macos-card overflow-hidden">
   439→              <ScrollArea className="max-h-[500px]">
   440→                <table className="macos-table">
   441→                  <thead>
   442→                    <tr>
   443→                      <th>Charger</th>
   444→                      <th>Vehicle</th>
   445→                      <th className="text-right">Energy (kWh)</th>
   446→                      <th className="text-right">Duration</th>
   447→                      <th className="text-right">Amount</th>
   448→                      <th>Status</th>
   449→                      <th>Started</th>
   450→                    </tr>
   451→                  </thead>
   452→                  <tbody>
   453→                    {chargingSessions.length > 0 ? chargingSessions.map((session) => (
   454→                      <tr key={session.id}>
   455→                        <td className="text-[13px] font-medium">{session.evCharger?.name || session.evChargerId}</td>
   456→                        <td className="text-[13px] font-mono">{session.vehiclePlate || 'N/A'}</td>
   457→                        <td className="text-right text-[13px] tabular-nums">{session.energyKwh.toFixed(1)}</td>

   463→                            <span className={`text-[11px] font-medium capitalize ${
   464→                              session.status === 'active' ? 'text-teal-500' :
   465→                              session.status === 'completed' ? 'text-emerald-500' :
   466→                              'text-muted-foreground'
   467→                            }`}>
   468→                              {session.status}
   469→                            </span>
   470→                          </div>
   471→                        </td>
   472→                        <td className="text-[12px] text-muted-foreground">
   473→                          {new Date(session.startedAt).toLocaleTimeString('en', { hour: '2-digit', minute: '2-digit' })}
   474→                        </td>
   475→                      </tr>

   479→                      </tr>
   480→                    )}
   481→                  </tbody>
   482→                </table>
   483→              </ScrollArea>
   484→            </div>
   485→          </motion.div>
   486→        )}
   487→
   488→        {/* ── CNG Tab ── */}
   489→        {activeTab === 'cng' && (
   490→          <motion.div key="cng" {...fadeUp} className="space-y-5">
   491→            {/* CNG Summary */}
   492→            <div className="grid grid-cols-3 sm:grid-cols-5 gap-3">
