   117→function getProtocolIcon(transport: string | null): React.ElementType {
   118→  if (!transport) return Network
   119→  const lower = transport.toLowerCase()
   120→  for (const [key, icon] of Object.entries(PROTOCOL_ICONS)) {
   121→    if (lower.includes(key)) return icon
   122→  }
   123→  return Network
   124→}

   128→function timeAgo(date: Date | string | null): string {
   129→  if (!date) return 'N/A'
   130→  const d = typeof date === 'string' ? new Date(date) : date
   131→  const diff = Date.now() - d.getTime()
   132→  if (diff < 60000) return 'just now'
   133→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`
   134→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`
   135→  return `${Math.floor(diff / 86400000)}d ago`
   136→}
   137→
   138→function formatTimestamp(date: Date | string | null): string {
   139→  if (!date) return 'N/A'
   140→  const d = typeof date === 'string' ? new Date(date) : date
   141→  return d.toLocaleTimeString('en', { hour: '2-digit', minute: '2-digit', second: '2-digit' })
   142→}

   638→          {/* Connection Health */}
   639→          <div className="macos-section-header">Connection Health</div>
   640→          {connections.length > 0 ? (
