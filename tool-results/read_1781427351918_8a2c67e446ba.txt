   145→   145→function formatPKR(amount: number): string {
   146→   146→  if (amount >= 1000000) return `PKR ${(amount / 1000000).toFixed(2)}M`
   147→   147→  if (amount >= 1000) return `PKR ${(amount / 1000).toFixed(0)}K`
   148→   148→  return `PKR ${amount.toLocaleString()}`
   149→   149→}

   190→   190→        setCngDispensers(data.cngDispensers || [])

   211→   211→    }
   212→   212→  }, [])

   226→   226→      monthMap[key].ev += s.totalAmount || 0
   227→   227→    }

   229→   229→      month,
   230→   230→      ev: Math.round(data.ev),

   232→   232→    }))
   233→   233→  })()

   236→   236→  const totalCNGRevenue = revenueData.reduce((s, d) => s + d.cng, 0)

   240→   240→    { key: 'sessions' as const, label: 'Sessions' },
   241→   241→    { key: 'cng' as const, label: 'CNG' },
   242→   242→    { key: 'revenue' as const, label: 'Revenue' },
   243→   243→  ]
   244→   244→

   249→   249→        <div className="grid grid-cols-3 sm:grid-cols-6 gap-3">
   250→   250→          {Array.from({ length: 6 }).map((_, i) => (
   251→   251→            <div key={i} className="macos-card p-3 text-center">
   252→   252→              <Skeleton className="h-6 w-8 mx-auto mb-1" />
   253→   253→              <Skeleton className="h-3 w-12 mx-auto" />
   254→   254→            </div>
   255→   255→          ))}
   256→   256→        </div>
   257→   257→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   258→   258→          {Array.from({ length: 6 }).map((_, i) => (
   259→   259→            <div key={i} className="macos-card p-4">

   267→   267→          <Loader2 className="size-4 animate-spin" />

   269→   269→        </div>
   270→   270→      </div>
   271→   271→    )
   272→   272→  }
   273→   273→
   274→   274→  // ── Error state ──
   275→   275→  if (error) {
   276→   276→    return (
   277→   277→      <div className="macos-card p-6 text-center" style={{ fontStretch: 'normal' }}>
