   101→   101→// ── Helpers ──────────────────────────────────────────────────
   102→   102→
   103→   103→function formatPKR(amount: number): string {
   104→   104→  return `PKR ${amount.toLocaleString('en-PK')}`
   105→   105→}
   106→   106→
   107→   107→function formatCompactPKR(amount: number): string {
   108→   108→  if (amount >= 1000000) return `PKR ${(amount / 1000000).toFixed(1)}M`
   109→   109→  if (amount >= 1000) return `PKR ${(amount / 1000).toFixed(0)}K`
   110→   110→  return `PKR ${amount}`
   111→   111→}

   113→   113→const FUEL_COLORS: Record<string, string> = {
   114→   114→  petrol: '#e8930c',
   115→   115→  pmg: '#e8930c',
   116→   116→  diesel: '#34c759',
   117→   117→  hsd: '#34c759',
   118→   118→  hobc: '#ff9f0a',
   119→   119→  premium: '#ff6b35',
   120→   120→  xl: '#ff6b35',
   121→   121→  default: '#e8930c',
   122→   122→}
   123→   123→
   124→   124→function getFuelColor(fuelType: string): string {
   125→   125→  const lower = fuelType.toLowerCase()
   126→   126→  for (const [key, color] of Object.entries(FUEL_COLORS)) {
   127→   127→    if (lower.includes(key)) return color
   128→   128→  }
   129→   129→  return FUEL_COLORS.default
   130→   130→}
