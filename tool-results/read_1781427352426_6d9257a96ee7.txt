   220→   220→function timeAgo(date: Date | string | null): string {
   221→   221→  if (!date) return 'N/A'
   222→   222→  const d = typeof date === 'string' ? new Date(date) : date
   223→   223→  const diff = Date.now() - d.getTime()
   224→   224→  if (diff < 60000) return 'just now'
   225→   225→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`
   226→   226→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`
   227→   227→  return `${Math.floor(diff / 86400000)}d ago`
   228→   228→}
