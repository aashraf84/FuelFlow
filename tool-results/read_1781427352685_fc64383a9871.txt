   172→   172→export default function SecurityCenterModule() {
   173→   173→  const [activeTab, setActiveTab] = useState('overview')
   174→   174→  const [auditFilter, setAuditFilter] = useState('all')
   175→   175→
   176→   176→  // Data state
   177→   177→  const [overview, setOverview] = useState<SecurityOverview | null>(null)
   178→   178→  const [sessions, setSessions] = useState<SecuritySessionData[]>([])
   179→   179→  const [permissions, setPermissions] = useState<RBACPermissionData[]>([])
   180→   180→  const [securityAlerts, setSecurityAlerts] = useState<SecurityAlertData[]>([])
   181→   181→  const [encryptionKeys, setEncryptionKeys] = useState<EncryptionKeyData[]>([])
   182→   182→  const [auditLogs, setAuditLogs] = useState<AuditLogData[]>([])
   183→   183→
   184→   184→  // Loading & error state
   185→   185→  const [loading, setLoading] = useState(true)
   186→   186→  const [error, setError] = useState<string | null>(null)
   187→   187→  const [actionLoading, setActionLoading] = useState<string | null>(null)
   188→   188→
   189→   189→  const fetchData = useCallback(async () => {
   190→   190→    setLoading(true)
   191→   191→    setError(null)
   192→   192→    try {
   193→   193→      const [overviewRes, sessionsRes, rbacRes, alertsRes, encryptionRes, auditRes] = await Promise.all([
   194→   194→        fetch('/api/security'),
   195→   195→        fetch('/api/security?section=sessions'),
   196→   196→        fetch('/api/security?section=rbac'),
   197→   197→        fetch('/api/security?section=alerts'),
   198→   198→        fetch('/api/security?section=encryption'),
   199→   199→        fetch('/api/audit?limit=50'),
   200→   200→      ])
   201→   201→
   202→   202→      if (!overviewRes.ok) throw new Error('Failed to fetch security data')
   203→   203→
   204→   204→      const overviewData = await overviewRes.json()
   205→   205→      const sessionsData = await sessionsRes.json()
   206→   206→      const rbacData = await rbacRes.json()
   207→   207→      const alertsData = await alertsRes.json()
   208→   208→      const encryptionData = await encryptionRes.json()
   209→   209→      const auditData = await auditRes.json()
   210→   210→
   211→   211→      setOverview(overviewData.overview || null)
   212→   212→      setSessions(sessionsData.sessions || [])
   213→   213→      setPermissions(rbacData.permissions || [])
   214→   214→      setSecurityAlerts(alertsData.alerts || [])
   215→   215→      setEncryptionKeys(encryptionData.keys || [])
   216→   216→      setAuditLogs(auditData.logs || [])
   217→   217→    } catch (err) {
   218→   218→      setError(err instanceof Error ? err.message : 'Unknown error')
   219→   219→    } finally {
   220→   220→      setLoading(false)
   221→   221→    }
   222→   222→  }, [])
   223→   223→
   224→   224→  useEffect(() => {
   225→   225→    fetchData()
   226→   226→  }, [fetchData])
   227→   227→
   228→   228→  const handleRevokeSession = async (sessionId: string) => {
   229→   229→    setActionLoading(sessionId)
   230→   230→    try {
   231→   231→      const res = await fetch('/api/security', {
   232→   232→        method: 'POST',
   233→   233→        headers: { 'Content-Type': 'application/json' },
   234→   234→        body: JSON.stringify({ action: 'revoke_session', sessionId }),
   235→   235→      })
   236→   236→      if (res.ok) {
   237→   237→        setSessions(prev => prev.filter(s => s.id !== sessionId))
   238→   238→      }
   239→   239→    } catch (err) {
   240→   240→      console.error('Revoke session error:', err)
   241→   241→    } finally {
   242→   242→      setActionLoading(null)
   243→   243→    }
   244→   244→  }
   245→   245→
   246→   246→  const handleResolveAlert = async (alertId: string) => {
   247→   247→    setActionLoading(alertId)
   248→   248→    try {
   249→   249→      const res = await fetch('/api/security', {
   250→   250→        method: 'POST',
   251→   251→        headers: { 'Content-Type': 'application/json' },
   252→   252→        body: JSON.stringify({ action: 'resolve_alert', alertId }),
   253→   253→      })
   254→   254→      if (res.ok) {
   255→   255→        setSecurityAlerts(prev => prev.map(a => a.id === alertId ? { ...a, isResolved: true, resolvedAt: new Date().toISOString() } : a))
   256→   256→      }
   257→   257→    } catch (err) {
   258→   258→      console.error('Resolve alert error:', err)
   259→   259→    } finally {
   260→   260→      setActionLoading(null)
   261→   261→    }
   262→   262→  }
   263→   263→
   264→   264→  // Determine threat level based on unresolved alerts
   265→   265→  const threatLevel = (() => {
   266→   266→    const unresolvedCritical = securityAlerts.filter(a => !a.isResolved && (a.severity === 'critical' || a.severity === 'error')).length
   267→   267→    const unresolvedWarning = securityAlerts.filter(a => !a.isResolved && a.severity === 'warning').length
   268→   268→    if (unresolvedCritical > 0) return 'high'
   269→   269→    if (unresolvedWarning > 2) return 'medium'
   270→   270→    return 'low'
   271→   271→  })()
   272→   272→
   273→   273→  const threat = THREAT_MAP[threatLevel]
   274→   274→
   275→   275→  const filteredLogs = auditFilter === 'all'
   276→   276→    ? auditLogs
   277→   277→    : auditLogs.filter(l => {
   278→   278→        if (auditFilter === 'critical') return l.action.toLowerCase().includes('fail') || l.action.toLowerCase().includes('delete')
   279→   279→        if (auditFilter === 'warning') return l.action.toLowerCase().includes('update') || l.action.toLowerCase().includes('warning')
   280→   280→        return true
   281→   281→      })
   282→   282→
   283→   283→  // Build RBAC matrix from permissions
   284→   284→  const rbacMatrix = (() => {
   285→   285→    const roleMap = new Map<string, Map<string, string>>()
   286→   286→    permissions.forEach(p => {
   287→   287→      if (!roleMap.has(p.role)) roleMap.set(p.role, new Map())
   288→   288→      const modules = roleMap.get(p.role)!
   289→   289→      const current = modules.get(p.module)
   290→   290→      if (p.isAllowed) {
   291→   291→        if (p.action === 'delete' || p.action === 'approve' || p.action === 'configure') {
   292→   292→          modules.set(p.module, 'full')
   293→   293→        } else if (p.action === 'write' && current !== 'full') {
   294→   294→          modules.set(p.module, 'edit')
   295→   295→        } else if (p.action === 'read' && !current) {
   296→   296→          modules.set(p.module, 'view')
   297→   297→        }
   298→   298→      } else if (!current) {
   299→   299→        modules.set(p.module, 'none')
   300→   300→      }
   301→   301→    })
   302→   302→    return Array.from(roleMap.entries()).map(([role, modules]) => ({
   303→   303→      role,
   304→   304→      modules: Object.fromEntries(modules),
   305→   305→    }))
   306→   306→  })()
   307→   307→
   308→   308→  // Compute compliance score based on resolved alerts and active keys
   309→   309→  const complianceScore = (() => {
   310→   310→    if (!overview) return 0
   311→   311→    const total = securityAlerts.length || 1
   312→   312→    const resolved = securityAlerts.filter(a => a.isResolved).length
   313→   313→    const hasActiveKeys = overview.activeEncryptionKeys > 0
   314→   314→    const base = Math.round((resolved / total) * 100)
   315→   315→    return hasActiveKeys ? Math.min(100, base + 10) : base
   316→   316→  })()
   317→   317→
   318→   318→  if (error) {
   319→   319→    return (
   320→   320→      <div className="macos-card p-8 text-center">
   321→   321→        <AlertTriangle className="size-8 text-red-500 mx-auto mb-3" />
   322→   322→        <p className="text-[13px] font-semibold text-red-500 mb-1">Failed to load security data</p>
   323→   323→        <p className="text-[12px] text-muted-foreground mb-3">{error}</p>
   324→   324→        <button onClick={fetchData} className="macos-btn gap-1.5">
   325→   325→          <RefreshCw className="size-3.5" /> Retry
   326→   326→        </button>
   327→   327→      </div>
   328→   328→    )
   329→   329→  }
   330→   330→
   331→   331→  return (
   332→   332→    <div className="space-y-4" style={{ fontStyle: 'normal !important' }}>
   333→   333→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   334→   334→        <div className="overflow-x-auto pb-1">
   335→   335→          <TabsList className="h-8 w-max gap-0.5 bg-muted/50">
   336→   336→            <TabsTrigger value="overview" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Overview</TabsTrigger>
   337→   337→            <TabsTrigger value="audit" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Audit Log</TabsTrigger>
   338→   338→            <TabsTrigger value="sessions" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Sessions</TabsTrigger>
   339→   339→            <TabsTrigger value="rbac" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">RBAC</TabsTrigger>
   340→   340→            <TabsTrigger value="alerts" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Alerts</TabsTrigger>
   341→   341→            <TabsTrigger value="encryption" className="text-[11px] px-3 h-6 data-[state=active]:bg-teal-500/15 data-[state=active]:text-teal-600">Encryption</TabsTrigger>
   342→   342→          </TabsList>
   343→   343→        </div>
   344→   344→
   345→   345→        {/* ── Overview ── */}
   346→   346→        <TabsContent value="overview" className="mt-4 space-y-4">
   347→   347→          {loading ? (
   348→   348→            <div className="space-y-4">
   349→   349→              <div className="macos-card p-5 animate-pulse">
   350→   350→                <div className="flex items-center gap-4">
   351→   351→                  <div className="size-14 rounded-2xl bg-muted/40" />
   352→   352→                  <div className="flex-1 space-y-2">
   353→   353→                    <div className="h-4 rounded bg-muted/40 w-48" />
   354→   354→                    <div className="h-3 rounded bg-muted/40 w-64" />
   355→   355→                  </div>
   356→   356→                  <div className="size-14 rounded bg-muted/40" />
   357→   357→                </div>
   358→   358→              </div>
   359→   359→              <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   360→   360→                {Array.from({ length: 4 }).map((_, i) => (
   361→   361→                  <div key={i} className="macos-card p-4 animate-pulse">
   362→   362→                    <div className="h-3 rounded bg-muted/40 w-24 mb-2" />
   363→   363→                    <div className="h-8 rounded bg-muted/40 w-12" />
   364→   364→                  </div>
   365→   365→                ))}
   366→   366→              </div>
   367→   367→            </div>
   368→   368→          ) : (
   369→   369→            <>
   370→   370→              {/* Threat Level Banner */}
   371→   371→              <div className="macos-card p-5">
   372→   372→                <div className="flex items-center gap-4">
   373→   373→                  <div className="size-14 rounded-2xl bg-emerald-500/10 flex items-center justify-center shrink-0">
   374→   374→                    <Shield className="size-7 text-emerald-500" />
   375→   375→                  </div>
   376→   376→                  <div className="flex-1">
   377→   377→                    <div className="flex items-center gap-2 mb-1">
   378→   378→                      <h2 className="text-[15px] font-semibold">Security Status</h2>
   379→   379→                      <div className="flex items-center gap-1.5">
   380→   380→                        <div className={`macos-status-dot ${threat.dot}`} />
   381→   381→                        <span className={`text-[11px] font-semibold ${threat.color}`}>Threat Level: {threat.label}</span>
   382→   382→                      </div>

   384→   384→                    <p className="text-[12px] text-muted-foreground">
   385→   385→                      {overview?.unresolvedAlerts ? `${overview.unresolvedAlerts} unresolved alerts` : 'All security services operating normally. No active threats detected.'}
   386→   386→                    </p>
   387→   387→                  </div>
   388→   388→                  <div className="text-center">
   389→   389→                    <div className="text-3xl font-bold text-emerald-600 dark:text-emerald-400">{complianceScore}</div>
   390→   390→                    <div className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold">Compliance</div>

   394→   394→
   395→   395→              {/* Stats Grid */}
   396→   396→              <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   397→   397→                {[
   398→   398→                  { label: 'Active Sessions', value: overview?.activeSessions ?? 0, icon: Monitor, accent: '#34c759' },

   400→   400→                  { label: 'Last Backup', value: overview?.lastBackup ? new Date(overview.lastBackup.completedAt || '').toLocaleDateString('en-PK', { month: 'short', day: 'numeric' }) : 'Never', icon: HardDrive, accent: '#ff9f0a', isText: true },
   401→   401→                  { label: 'Encryption Keys', value: overview?.activeEncryptionKeys ?? 0, icon: Key, accent: '#e8930c' },
   402→   402→                ].map((stat) => (

   408→   408→                      </div>

   410→   410→                        <stat.icon className="size-4" style={{ color: stat.accent }} />
