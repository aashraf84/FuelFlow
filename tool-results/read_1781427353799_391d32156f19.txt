   121→   121→function getHttpStatusColor(code: number) {
   122→   122→  if (code >= 200 && code < 300) return 'text-emerald-600 dark:text-emerald-400'
   123→   123→  if (code >= 400 && code < 500) return 'text-teal-600 dark:text-teal-400'
   124→   124→  return 'text-red-600 dark:text-red-400'
   125→   125→}
   126→   126→
   127→   127→function timeAgo(dateStr: string | null): string {
   128→   128→  if (!dateStr) return 'Never'
   129→   129→  const diff = Date.now() - new Date(dateStr).getTime()
   130→   130→  const mins = Math.floor(diff / 60000)
   131→   131→  if (mins < 1) return 'Just now'
   132→   132→  if (mins < 60) return `${mins} min ago`
   133→   133→  const hrs = Math.floor(mins / 60)
   134→   134→  if (hrs < 24) return `${hrs} hr ago`
   135→   135→  const days = Math.floor(hrs / 24)
   136→   136→  return `${days} day${days > 1 ? 's' : ''} ago`
   137→   137→}
