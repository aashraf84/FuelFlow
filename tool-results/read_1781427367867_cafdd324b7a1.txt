   137→export default function EmployeesModule() {
   138→  const [activeTab, setActiveTab] = useState<TabKey>('employees')
   139→  const [employees, setEmployees] = useState<EmployeeData[]>([])
   140→  const [leaveRequests, setLeaveRequests] = useState<LeaveRequestData[]>([])
   141→  const [payrollSummary, setPayrollSummary] = useState<PayrollSummary | null>(null)
   142→  const [loading, setLoading] = useState(true)
   143→  const [error, setError] = useState<string | null>(null)
   144→  const { selectedStation } = useAppStore()
   145→
   146→  const fetchData = useCallback(async () => {
   147→    setLoading(true)
   148→    setError(null)
   149→    try {
   150→      const stationParam = selectedStation && selectedStation !== 'all' ? `?stationId=${selectedStation}` : ''
   151→      const [empRes, leaveRes, payrollRes] = await Promise.all([
   152→        fetch(`/api/employees${stationParam}`),
   153→        fetch(`/api/leave${stationParam}`),
   154→        fetch(`/api/payroll${stationParam}`),
   155→      ])
   156→
   157→      if (!empRes.ok || !leaveRes.ok || !payrollRes.ok) {
   158→        throw new Error('Failed to fetch employee data')
   159→      }
   160→
   161→      const empData = await empRes.json()
   162→      const leaveData = await leaveRes.json()
   163→      const payrollData = await payrollRes.json()
   164→
   165→      setEmployees(empData.employees || [])
   166→      setLeaveRequests(leaveData.requests || [])
   167→      setPayrollSummary(payrollData.summary || null)
   168→    } catch (err) {
   169→      setError(err instanceof Error ? err.message : 'Unknown error')
   170→    } finally {
   171→      setLoading(false)
   172→    }
   173→  }, [selectedStation])
   174→
   175→  useEffect(() => {
   176→    fetchData()
   177→  }, [fetchData])
   178→
   179→  const totalEmp = employees.length
   180→  const onShift = employees.filter(e => e.status === 'active' || e.status === 'on-shift' || e.status === 'new').length
   181→  const onLeave = employees.filter(e => e.status === 'on-leave').length
   182→  const newEmp = employees.filter(e => e.status === 'new').length
   183→
   184→  if (error) {
   185→    return (
   186→      <div className="macos-card p-8 text-center">
   187→        <AlertCircle className="size-8 text-red-500 mx-auto mb-3" />
   188→        <p className="text-[13px] font-semibold text-red-500 mb-1">Failed to load employee data</p>
   189→        <p className="text-[12px] text-muted-foreground mb-3">{error}</p>
   190→        <button onClick={fetchData} className="macos-btn gap-1.5">
   191→          <RefreshCw className="size-3.5" /> Retry
   192→        </button>
   193→      </div>
   194→    )
   195→  }
   196→
   197→  return (
   198→    <div className="space-y-5" style={{ fontStyle: 'normal !important' }}>
   199→      {/* ── Summary Stats ── */}
   200→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   201→        {loading ? (
   202→          Array.from({ length: 4 }).map((_, i) => <SkeletonCard key={i} />)
   203→        ) : (
   204→          [
   205→            { label: 'Total Employees', value: totalEmp.toString(), icon: Users, accent: false },
   206→            { label: 'On Shift', value: onShift.toString(), icon: Clock, accent: false },
   207→            { label: 'On Leave', value: onLeave.toString(), icon: Coffee, accent: false },
   208→            { label: 'New Hires', value: newEmp.toString(), icon: UserPlus, accent: true },
   209→          ].map((stat, i) => (
   210→            <motion.div
   211→              key={stat.label}
   212→              initial={{ opacity: 0, y: 8 }}
   213→              animate={{ opacity: 1, y: 0 }}
   214→              transition={{ delay: i * 0.05, duration: 0.3 }}
   215→              className="macos-card p-4"
   216→            >
   217→              <div className="flex items-start justify-between">
   218→                <div>
   219→                  <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">{stat.label}</p>
   220→                  <p className={`text-[20px] font-bold mt-1 ${stat.accent ? 'fuel-text-gradient' : 'text-foreground'}`}>
   221→                    {stat.value}
   222→                  </p>
   223→                </div>
   224→                <div className={`size-8 rounded-lg flex items-center justify-center ${stat.accent ? 'bg-teal-500/10' : 'bg-black/[0.03] dark:bg-white/[0.05]'}`}>
   225→                  <stat.icon className={`size-4 ${stat.accent ? 'text-teal-500' : 'text-muted-foreground'}`} strokeWidth={1.5} />
   226→                </div>
   227→              </div>
   228→            </motion.div>
   229→          ))
   230→        )}
   231→      </div>
   232→
   233→      {/* ── Tab Bar ── */}
   234→      <div className="flex items-center gap-1 overflow-x-auto pb-1 -mb-1">
   235→        {tabs.map((tab) => {
   236→          const isActive = activeTab === tab.key
   237→          const Icon = tab.icon
   238→          return (
   239→            <button
   240→              key={tab.key}
   241→              onClick={() => setActiveTab(tab.key)}
   242→              className={`flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[13px] font-medium transition-all duration-150 whitespace-nowrap ${
   243→                isActive
   244→                  ? 'bg-teal-500/10 text-teal-600 dark:text-teal-400'
   245→                  : 'text-muted-foreground hover:bg-black/[0.03] dark:hover:bg-white/[0.05] hover:text-foreground'
   246→              }`}
   247→            >
   248→              <Icon className="size-3.5" strokeWidth={isActive ? 2 : 1.5} />
   249→              {tab.label}
   250→            </button>
   251→          )
   252→        })}
   253→      </div>
   254→
   255→      {/* ── Tab Content ── */}
   256→      <AnimatePresence mode="wait">
   257→        <motion.div
   258→          key={activeTab}
   259→          initial={{ opacity: 0, y: 6 }}
   260→          animate={{ opacity: 1, y: 0 }}
   261→          exit={{ opacity: 0, y: -4 }}
   262→          transition={{ duration: 0.2, ease: [0.25, 0.1, 0.25, 1] }}
   263→        >
   264→          {activeTab === 'employees' && <EmployeesTab employees={employees} loading={loading} onRefresh={fetchData} />}
   265→          {activeTab === 'attendance' && <AttendanceTab employees={employees} loading={loading} />}
   266→          {activeTab === 'payroll' && <PayrollTab employees={employees} payrollSummary={payrollSummary} loading={loading} />}
   267→          {activeTab === 'leave' && <LeaveTab leaveRequests={leaveRequests} loading={loading} onRefresh={fetchData} />}
   268→        </motion.div>
   269→      </AnimatePresence>
   270→    </div>
   271→  )
   272→}

   376→function AttendanceTab({ employees, loading }: { employees: EmployeeData[]; loading: boolean }) {
   377→  const present = employees.filter(e => e.todayAttendance && e.todayAttendance.checkIn).length
   378→  const onLeave = employees.filter(e => e.status === 'on-leave').length
   379→  const offDuty = employees.filter(e => !e.todayAttendance || !e.todayAttendance.checkIn).length
   380→  const late = employees.filter(e => e.todayAttendance && e.todayAttendance.status === 'late').length
   381→  const total = employees.length || 1
   382→  const rate = total > 0 ? Math.round((present / total) * 100) : 0
   383→
   384→  const attStats = [
   385→    { label: 'Present', value: present, total, color: 'text-emerald-600 dark:text-emerald-400', bg: 'bg-emerald-500/10', dot: 'macos-status-online' },
   386→    { label: 'On Leave', value: onLeave, total, color: 'text-teal-500', bg: 'bg-teal-500/10', dot: 'macos-status-warning' },
   387→    { label: 'Off Duty', value: offDuty, total, color: 'text-muted-foreground', bg: 'bg-black/[0.03] dark:bg-white/[0.05]', dot: 'macos-status-offline' },
   388→    { label: 'Late Check-in', value: late, total, color: 'text-red-500', bg: 'bg-red-500/10', dot: 'macos-status-error' },
   389→  ]
   390→
   391→  const activeEmployees = employees.filter(e => e.status === 'active' || e.status === 'on-shift' || e.status === 'new')
   392→
   393→  return (
   394→    <div className="space-y-4">
   395→      {/* Attendance Stats */}
   396→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   397→        {loading ? (
   398→          Array.from({ length: 4 }).map((_, i) => <SkeletonCard key={i} />)
   399→        ) : (
   400→          attStats.map((s, i) => (
   401→            <motion.div
   402→              key={s.label}
   403→              initial={{ opacity: 0, y: 8 }}
   404→              animate={{ opacity: 1, y: 0 }}
   405→              transition={{ delay: i * 0.05, duration: 0.3 }}
   406→              className="macos-card p-4"
   407→            >
   408→              <div className="flex items-center gap-2 mb-2">
   409→                <div className={`macos-status-dot ${s.dot}`} />
   410→                <span className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">{s.label}</span>
   411→              </div>
   412→              <p className={`text-[24px] font-bold ${s.color}`}>{s.value}</p>
   413→              <p className="text-[11px] text-muted-foreground">{Math.round((s.value / s.total) * 100)}% of workforce</p>
   414→            </motion.div>
   415→          ))
   416→        )}
   417→      </div>
   418→
   419→      {/* Attendance Rate Card */}
   420→      <div className="macos-card p-5">
   421→        <div className="flex items-center justify-between mb-4">
   422→          <div>
   423→            <h3 className="text-[13px] font-semibold">Attendance Rate</h3>
   424→            <p className="text-[11px] text-muted-foreground mt-0.5">Current shift overview</p>
   425→          </div>
   426→          <div className="text-right">
   427→            <p className="text-[28px] font-bold fuel-text-gradient">{rate}%</p>
   428→            <div className="flex items-center gap-1 justify-end mt-0.5">
   429→              <TrendingUp className="size-3 text-emerald-500" />
   430→              <span className="text-[11px] font-medium text-emerald-500">Live data</span>
   431→            </div>
   432→          </div>
   433→        </div>
   434→        <div className="w-full h-2 rounded-full bg-black/[0.04] dark:bg-white/[0.06] overflow-hidden">
   435→          <motion.div
   436→            initial={{ width: 0 }}
   437→            animate={{ width: `${rate}%` }}
   438→            transition={{ duration: 0.8, ease: 'easeOut' }}
   439→            className="h-full rounded-full fuel-gradient"
   440→          />
   441→        </div>
   442→      </div>
   443→
   444→      {/* Who's On Shift */}
   445→      <div className="macos-card overflow-hidden">
   446→        <div className="px-4 py-3 border-b border-border">
   447→          <h3 className="text-[13px] font-semibold">Currently Active</h3>
   448→        </div>
   449→        {loading ? (
   450→          <div className="p-4 space-y-3">
   451→            {Array.from({ length: 3 }).map((_, i) => (
   452→              <div key={i} className="h-10 rounded-lg bg-muted/30 animate-pulse" />
   453→            ))}
   454→          </div>
   455→        ) : activeEmployees.length === 0 ? (
   456→          <div className="p-8 text-center text-[13px] text-muted-foreground">No active employees</div>
   457→        ) : (
   458→          <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-0">
   459→            {activeEmployees.map((emp) => {
   460→              const role = roleConfig[emp.role] || roleConfig['attendant']
   461→              return (
   462→                <div key={emp.id} className="macos-group-item !py-3 !px-4 flex items-center gap-2.5">
   463→                  <div className={`size-9 rounded-lg flex items-center justify-center ${role.bg} shrink-0`}>
   464→                    <span className={`text-[11px] font-bold ${role.color}`}>
   465→                      {emp.name.split(' ').map(n => n[0]).join('').slice(0, 2)}
   466→                    </span>
   467→                  </div>
   468→                  <div className="min-w-0">
   469→                    <p className="text-[13px] font-semibold truncate">{emp.name}</p>
   470→                    <div className="flex items-center gap-1.5">

   612→function LeaveTab({ leaveRequests, loading, onRefresh }: { leaveRequests: LeaveRequestData[]; loading: boolean; onRefresh: () => void }) {
   613→  const [actionLoading, setActionLoading] = useState<string | null>(null)
   614→
   615→  const pending = leaveRequests.filter(l => l.status === 'pending').length
   616→  const approved = leaveRequests.filter(l => l.status === 'approved').length
   617→  const rejected = leaveRequests.filter(l => l.status === 'rejected').length
   618→
   619→  const handleLeaveAction = async (id: string, status: 'approved' | 'rejected') => {
   620→    setActionLoading(id)
   621→    try {
   622→      const res = await fetch('/api/leave', {
   623→        method: 'PUT',
   624→        headers: { 'Content-Type': 'application/json' },
   625→        body: JSON.stringify({ id, status }),
   626→      })
   627→      if (res.ok) {
   628→        onRefresh()
   629→      }
   630→    } catch (err) {
   631→      console.error('Leave action error:', err)
   632→    } finally {
   633→      setActionLoading(null)
   634→    }
   635→  }
   636→
   637→  const formatDate = (dateStr: string) => {
   638→    try {
   639→      return new Date(dateStr).toLocaleDateString('en-PK', { year: 'numeric', month: 'short', day: 'numeric' })
   640→    } catch {
   641→      return dateStr
   642→    }
   643→  }
   644→
   645→  const getDays = (start: string, end: string) => {
   646→    try {
   647→      const s = new Date(start)
   648→      const e = new Date(end)
   649→      return Math.max(1, Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)) + 1)
   650→    } catch {
   651→      return 1
   652→    }
   653→  }
   654→
   655→  return (
   656→    <div className="space-y-4">
   657→      {/* Leave Stats */}
   658→      <div className="grid grid-cols-3 gap-3">
   659→        <div className="macos-card p-4">
   660→          <div className="flex items-center gap-2 mb-1">
   661→            <div className="macos-status-dot macos-status-warning" />
   662→            <span className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Pending</span>
   663→          </div>
   664→          <p className="text-[20px] font-bold text-teal-500">{pending}</p>
   665→        </div>
   666→        <div className="macos-card p-4">
   667→          <div className="flex items-center gap-2 mb-1">
   668→            <div className="macos-status-dot macos-status-online" />
   669→            <span className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Approved</span>
   670→          </div>
   671→          <p className="text-[20px] font-bold text-emerald-600 dark:text-emerald-400">{approved}</p>
   672→        </div>
   673→        <div className="macos-card p-4">
   674→          <div className="flex items-center gap-2 mb-1">
   675→            <div className="macos-status-dot macos-status-error" />
   676→            <span className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Rejected</span>
   677→          </div>
   678→          <p className="text-[20px] font-bold text-red-500">{rejected}</p>
   679→        </div>
   680→      </div>
   681→
   682→      {/* Leave Requests List */}
   683→      <div className="macos-card overflow-hidden">
   684→        <div className="flex items-center justify-between px-4 py-3 border-b border-border">
   685→          <h3 className="text-[13px] font-semibold">Leave Requests</h3>
   686→          <span className="text-[11px] text-muted-foreground">{leaveRequests.length} total requests</span>
   687→        </div>
   688→        <div className="overflow-x-auto">
   689→          <table className="macos-table">
   690→            <thead>
   691→              <tr>
   692→                <th>Employee</th>
   693→                <th>Role</th>
   694→                <th>Type</th>
   695→                <th>From</th>
   696→                <th>To</th>
   697→                <th className="text-center">Days</th>
   698→                <th>Reason</th>
   699→                <th>Status</th>
   700→                <th>Actions</th>
   701→              </tr>
   702→            </thead>
   703→            <tbody>
   704→              {loading ? (
   705→                Array.from({ length: 4 }).map((_, i) => <SkeletonRow key={i} cols={9} />)
   706→              ) : leaveRequests.length === 0 ? (
   707→                <tr>
   708→                  <td colSpan={9} className="text-center py-8 text-[13px] text-muted-foreground">
