    95→function getLevelColor(pct: number): string {
    96→  if (pct > 50) return '#34c759'
    97→  if (pct >= 25) return '#ff9f0a'
    98→  return '#ff3b30'
    99→}
   100→
   101→function getLevelBgClass(pct: number): string {
   102→  if (pct > 50) return 'bg-[#34c759]'
   103→  if (pct >= 25) return 'bg-[#ff9f0a]'
   104→  return 'bg-[#ff3b30]'
   105→}
   106→
   107→function getTankStatusClass(status: string): string {
   108→  if (status === 'normal' || status === 'online') return 'macos-status-online'
   109→  if (status === 'low' || status === 'warning') return 'macos-status-warning'
   110→  return 'macos-status-error'
   111→}
   112→
   113→function formatPKR(amount: number): string {
   114→  return `PKR ${amount.toLocaleString('en-PK')}`
   115→}
   116→
   117→function formatTime(iso: string): string {
   118→  return new Date(iso).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: true })
   119→}
   120→
   121→function formatDate(iso: string): string {
   122→  return new Date(iso).toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   123→}

   125→function generateAlertsFromTanks(tanks: TankData[]): TankAlert[] {
   126→  const alerts: TankAlert[] = []
   127→  tanks.forEach((tank, i) => {
   128→    if (tank.levelPercentage < 15) {
   129→      alerts.push({
   130→        id: `alert-low-${i}`,
   131→        type: 'low-level',
   132→        severity: 'critical',
   133→        tank: tank.name,
   134→        message: `${tank.fuelType} level critically low at ${tank.levelPercentage}% — below reorder point`,
   135→        timestamp: new Date().toISOString(),
   136→      })
   137→    } else if (tank.levelPercentage < 25) {
   138→      alerts.push({
   139→        id: `alert-lowwarn-${i}`,
   140→        type: 'low-level',
   141→        severity: 'warning',
   142→        tank: tank.name,
   143→        message: `${tank.fuelType} level at ${tank.levelPercentage}% — approaching reorder point`,
   144→        timestamp: new Date().toISOString(),
   145→      })
   146→    }
   147→    if (tank.waterLevel > 1.0) {
   148→      alerts.push({
   149→        id: `alert-water-${i}`,
   150→        type: 'high-water',
   151→        severity: 'warning',
   152→        tank: tank.name,
   153→        message: `Water level ${tank.waterLevel}cm exceeds 1.0cm threshold — schedule draining`,
   154→        timestamp: new Date().toISOString(),
   155→      })
   156→    }
   157→    if (tank.temperature > 30) {
   158→      alerts.push({
   159→        id: `alert-temp-${i}`,
   160→        type: 'temperature',
   161→        severity: 'warning',
   162→        tank: tank.name,
   163→        message: `${tank.fuelType} temperature ${tank.temperature}°C above 30°C monitoring threshold`,
   164→        timestamp: new Date().toISOString(),
   165→      })
   166→    }
   167→  })
   168→  return alerts
   169→}

   270→function AlertItem({ alert, index }: { alert: TankAlert; index: number }) {
   271→  const severityIcon = {
   272→    critical: <XCircle className="size-4 text-[#ff3b30]" />,
   273→    warning: <AlertCircle className="size-4 text-[#ff9f0a]" />,
   274→    info: <Info className="size-4 text-[#007aff]" />,
   275→  }
   276→  const severityBg = {
   277→    critical: 'bg-[#ff3b30]/5 border-[#ff3b30]/10',
   278→    warning: 'bg-[#ff9f0a]/5 border-[#ff9f0a]/10',
   279→    info: 'bg-[#007aff]/5 border-[#007aff]/10',
   280→  }
   281→  return (
   282→    <motion.div
   283→      initial={{ opacity: 0, x: -8 }}
   284→      animate={{ opacity: 1, x: 0 }}
   285→      transition={{ duration: 0.25, delay: index * 0.05 }}
   286→      className={`macos-group-item flex items-start gap-3 border ${severityBg[alert.severity]} rounded-lg`}
   287→    >
   288→      <div className="mt-0.5 shrink-0">{severityIcon[alert.severity]}</div>
   289→      <div className="flex-1 min-w-0">
   290→        <div className="flex items-center justify-between gap-2">
   291→          <p className="text-[13px] font-medium text-foreground">{alert.tank}</p>
   292→          <span className="text-[11px] text-muted-foreground shrink-0">{formatTime(alert.timestamp)}</span>
   293→        </div>
   294→        <p className="text-[12px] text-muted-foreground mt-0.5 leading-relaxed">{alert.message}</p>
   295→      </div>
   296→    </motion.div>
   297→  )
   298→}

   302→export default function TanksModule() {
   303→  const [activeTab, setActiveTab] = useState<'overview' | 'readings' | 'deliveries' | 'reconciliation'>('overview')
   304→  const { selectedStation } = useAppStore()
   305→
   306→  // Data state
   307→  const [tanks, setTanks] = useState<TankData[]>([])
   308→  const [deliveries, setDeliveries] = useState<DeliveryData[]>([])
   309→  const [reconciliations, setReconciliations] = useState<ReconciliationData[]>([])
   310→  const [loading, setLoading] = useState(true)
   311→  const [error, setError] = useState<string | null>(null)
   312→
   313→  const fetchData = useCallback(async () => {
   314→    setLoading(true)
   315→    setError(null)
   316→    try {
   317→      const stationParam = selectedStation && selectedStation !== 'all' ? `?stationId=${selectedStation}` : ''
   318→
   319→      const [tanksRes, deliveriesRes, recRes] = await Promise.all([
   320→        fetch(`/api/tanks${stationParam ? `?stationId=${selectedStation}` : ''}`),
   321→        fetch(`/api/deliveries${stationParam ? `?stationId=${selectedStation}` : ''}`),
   322→        fetch(`/api/reconciliations${stationParam ? `?stationId=${selectedStation}` : ''}`),
   323→      ])
   324→
   325→      if (!tanksRes.ok) throw new Error('Failed to fetch tanks data')
   326→
   327→      const tanksData = await tanksRes.json()
   328→      const deliveriesData = deliveriesRes.ok ? await deliveriesRes.json() : { deliveries: [] }
   329→      const recData = recRes.ok ? await recRes.json() : { reconciliations: [] }
   330→
   331→      setTanks(tanksData.tanks || [])
   332→      setDeliveries(deliveriesData.deliveries || [])
   333→      setReconciliations(recData.reconciliations || [])
   334→    } catch (err) {
   335→      console.error('TanksModule fetch error:', err)
   336→      setError('Unable to load tanks data. Please try again.')
   337→    } finally {
   338→      setLoading(false)
   339→    }
   340→  }, [selectedStation])
   341→
   342→  useEffect(() => {
   343→    fetchData()
   344→  }, [fetchData])
   345→
   346→  // Derived data
   347→  const alerts = generateAlertsFromTanks(tanks)
   348→  const onlineTanks = tanks.filter(t => t.status === 'normal' || t.status === 'online').length
   349→
   350→  // Build level history data from tank readings
   351→  const levelHistoryData = (() => {
   352→    if (tanks.length === 0) return []
   353→    // Create chart data from tank current levels
   354→    const data: Record<string, unknown>[] = []
   355→    const hourLabels = ['08:00', '09:00', '10:00', '11:00', '12:00', '13:00', '14:00']
   356→    hourLabels.forEach((time, idx) => {
   357→      const point: Record<string, unknown> = { time }
   358→      tanks.forEach((tank, tIdx) => {
   359→        const key = `T${tIdx + 1}`
   360→        // Simulate declining levels throughout the day based on current level
   361→        const decline = idx * (tank.levelPercentage / 14)
   362→        point[key] = Math.max(Math.round(tank.levelPercentage + (tank.levelPercentage * 0.15) - decline), 5)
   363→      })
   364→      data.push(point)
   365→    })
   366→    return data
   367→  })()
   368→
   369→  const tabs = [
   370→    { key: 'overview' as const, label: 'Overview' },
   371→    { key: 'readings' as const, label: 'Readings' },
   372→    { key: 'deliveries' as const, label: 'Deliveries' },
   373→    { key: 'reconciliation' as const, label: 'Reconciliation' },
   374→  ]
   375→
   376→  if (error) {
   377→    return <ErrorBanner message={error} onRetry={fetchData} />
   378→  }
   379→
   380→  return (
   381→    <div className="space-y-5" style={{ fontStyle: 'normal !important' }}>
   382→      {/* ── Header ── */}
   383→      <div className="flex items-center justify-between">
   384→        <div>
   385→          <h2 className="text-[13px] font-semibold text-foreground">Tanks &amp; Wet Stock</h2>
   386→          <p className="text-[11px] text-muted-foreground mt-0.5">Real-time tank monitoring and inventory management</p>
   387→        </div>
   388→        <div className="flex items-center gap-2">
   389→          <div className="flex items-center gap-1.5 rounded-lg bg-card px-3 py-1.5 shadow-[0_0_0_0.5px_var(--border)]">
   390→            <div className={`macos-status-dot ${onlineTanks === tanks.length && tanks.length > 0 ? 'macos-status-online' : 'macos-status-warning'}`} />
   391→            <span className="text-[11px] font-medium text-muted-foreground">
   392→              {loading ? 'Loading...' : `${onlineTanks} Tanks Online`}
   393→            </span>
   394→          </div>
   395→          <button onClick={fetchData} className="macos-btn-secondary gap-1.5" title="Refresh data">
   396→            <RefreshCw className={`size-3 ${loading ? 'animate-spin' : ''}`} />
   397→          </button>
   398→          <button className="macos-btn gap-1.5">
   399→            <ClipboardCheck className="size-3.5" />
   400→            New Reading
   401→          </button>
   402→        </div>
   403→      </div>
   404→
   405→      {/* ── Tab Bar ── */}
   406→      <div className="flex items-center gap-1 rounded-lg bg-black/[0.03] dark:bg-white/[0.04] p-0.5 w-fit">
   407→        {tabs.map((tab) => (
   408→          <button
   409→            key={tab.key}
   410→            onClick={() => setActiveTab(tab.key)}
   411→            className={`relative rounded-md px-3 py-1.5 text-[12px] font-medium transition-colors ${
   412→              activeTab === tab.key ? 'text-foreground' : 'text-muted-foreground hover:text-foreground'
   413→            }`}
   414→          >
   415→            {activeTab === tab.key && (
   416→              <motion.div
   417→                layoutId="tanks-tab-indicator"
   418→                className="absolute inset-0 rounded-md bg-card shadow-[0_0_0_0.5px_var(--border)]"
   419→                transition={{ duration: 0.2, ease: [0.25, 0.1, 0.25, 1] }}
   420→              />
   421→            )}
   422→            <span className="relative z-10">{tab.label}</span>
   423→          </button>
   424→        ))}
   425→      </div>
   426→
   427→      {/* ── Tab Content ── */}
   428→      <AnimatePresence mode="wait">
   429→        {activeTab === 'overview' && (
   430→          <motion.div
   431→            key="overview"
   432→            initial={{ opacity: 0, y: 6 }}
   433→            animate={{ opacity: 1, y: 0 }}
   434→            exit={{ opacity: 0, y: -4 }}
   435→            transition={{ duration: 0.2 }}
   436→            className="space-y-5"
   437→          >
   438→            {loading ? (
   439→              <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-4">
   440→                {Array.from({ length: 4 }).map((_, i) => <SkeletonCard key={i} />)}
   441→              </div>
   442→            ) : tanks.length === 0 ? (
   443→              <div className="macos-card p-8 text-center">
   444→                <Droplets className="size-8 text-muted-foreground mx-auto mb-3" />
   445→                <p className="text-[13px] font-medium text-foreground">No tanks found</p>
   446→                <p className="text-[11px] text-muted-foreground mt-1">Tanks will appear here once configured.</p>
   447→              </div>
   448→            ) : (
   449→              <>
   450→                {/* Tank Overview Cards */}
   451→                <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-4">
   452→                  {tanks.map((tank, i) => (
   453→                    <TankOverviewCard key={tank.id} tank={tank} index={i} />
   454→                  ))}
   455→                </div>
   456→
   457→                {/* Level History Chart + Alerts */}
   458→                <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   459→                  {/* Level Trend Chart */}
   460→                  <div className="macos-card p-4 lg:col-span-2">
   461→                    <div className="flex items-center justify-between mb-4">
   462→                      <div>
   463→                        <p className="text-[13px] font-semibold text-foreground">Level Trend</p>
   464→                        <p className="text-[11px] text-muted-foreground">Today&apos;s tank level progression</p>
   465→                      </div>
   466→                      <div className="flex items-center gap-3">

   468→                          <div key={i} className="flex items-center gap-1">
   469→                            <div
   470→                              className="size-2 rounded-full"
   471→                              style={{ backgroundColor: ['#e8930c', '#34c759', '#ff3b30', '#ff9f0a'][i % 4] }}

   474→                          </div>

   476→                      </div>
   477→                    </div>
   478→                    {levelHistoryData.length > 0 ? (
   479→                      <div className="h-56">
   480→                        <ResponsiveContainer width="100%" height="100%">
   481→                          <AreaChart data={levelHistoryData} margin={{ top: 4, right: 4, left: -20, bottom: 0 }}>
   482→                            <defs>
   483→                              <linearGradient id="gradT1" x1="0" y1="0" x2="0" y2="1">
   484→                                <stop offset="5%" stopColor="#e8930c" stopOpacity={0.15} />
   485→                                <stop offset="95%" stopColor="#e8930c" stopOpacity={0} />
   486→                              </linearGradient>
   487→                              <linearGradient id="gradT2" x1="0" y1="0" x2="0" y2="1">
   488→                                <stop offset="5%" stopColor="#34c759" stopOpacity={0.15} />
   489→                                <stop offset="95%" stopColor="#34c759" stopOpacity={0} />
   490→                              </linearGradient>
   491→                            </defs>
   492→                            <XAxis dataKey="time" tick={{ fontSize: 11, fill: 'var(--muted-foreground)' }} axisLine={false} tickLine={false} />

   494→                            <Tooltip
   495→                              contentStyle={{ fontSize: 12, borderRadius: 10, border: '0.5px solid var(--border)', boxShadow: '0 2px 8px rgba(0,0,0,0.08)', background: 'var(--card)' }}
   496→                              labelStyle={{ fontWeight: 600, fontSize: 11 }}
   497→                            />
   498→                            {tanks.length >= 1 && <Area type="monotone" dataKey="T1" stroke="#e8930c" strokeWidth={2} fill="url(#gradT1)" />}
