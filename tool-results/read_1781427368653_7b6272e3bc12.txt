   254→function HardwareSettings() {

   258→  const [loading, setLoading] = useState(true)
   259→  const [error, setError] = useState<string | null>(null)
   260→
   261→  const fetchData = useCallback(async () => {
   262→    setLoading(true)
   263→    setError(null)
   264→    try {
   265→      const [ctrlRes, protoRes] = await Promise.all([
   266→        fetch('/api/hardware?action=controllers'),
   267→        fetch('/api/comms?section=protocols'),
   268→      ])
   269→      if (!ctrlRes.ok) throw new Error('Failed to load hardware data')
   270→      const ctrlData = await ctrlRes.json()
   271→      const protoData = await protoRes.json()

   273→      const allControllers: Record<string, unknown>[] = ctrlData.controllers ?? []
   274→      const allDispensers: Record<string, unknown>[] = []

   276→        const ctrlDispensers = (ctrl.dispensers ?? []) as Record<string, unknown>[]
   277→        for (const d of ctrlDispensers) {
   278→          allDispensers.push({
   279→            ...d,
   280→            controllerName: ctrl.name,
   281→          })
   282→        }
   283→      }
   284→
   285→      setControllers(allControllers)
   286→      setDispensers(allDispensers)
   287→      setProtocols(protoData.protocols ?? [])
   288→    } catch (err) {
   289→      setError(err instanceof Error ? err.message : 'Unknown error')

   291→      setLoading(false)
   292→    }
