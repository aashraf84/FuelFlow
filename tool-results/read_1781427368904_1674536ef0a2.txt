    85→// ── Helpers ──────────────────────────────────────
    86→
    87→function formatPKR(amount: number): string {
    88→  if (amount >= 1000000) return `PKR ${(amount / 1000000).toFixed(2)}M`
    89→  if (amount >= 1000) return `PKR ${(amount / 1000).toFixed(0)}K`
    90→  return `PKR ${amount.toLocaleString()}`
    91→}
    92→
    93→function formatVol(litres: number): string {
    94→  if (litres >= 1000) return `${(litres / 1000).toFixed(0)}K L`
    95→  return `${litres} L`
    96→}

   132→          revenue: 0,

   135→          alerts: [],
   136→        })

   138→

   140→      // For now, distribute the dashboard totals evenly as a fallback
   141→      if (dashboardData?.kpis && enrichedStations.length > 0) {
   142→        const revPerStation = Math.round(dashboardData.kpis.revenue.today / enrichedStations.length)
   143→        const volPerStation = Math.round(dashboardData.kpis.fuelSold.value / enrichedStations.length)
   144→        const txPerStation = Math.round(
   145→          (dashboardData.recentTransactions?.length || 0) / enrichedStations.length
   146→        )

   149→          s.volume = volPerStation
   150→          s.transactions = txPerStation

   153→            s.alerts = [{ severity: 'critical', message: 'Fuel level critically low' }]

   155→            s.alerts = [{ severity: 'warning', message: 'Fuel level below 50%' }]
   156→          }
   157→        })

   160→      setStations(enrichedStations)
   161→      setKpis(dashboardData?.kpis || null)

   164→    } finally {
   165→      setLoading(false)
   166→    }
   167→  }, [])
   168→
   169→  useEffect(() => {
   170→    fetchData()
   171→  }, [fetchData])
   172→
   173→  // Loading state
   174→  if (loading) {
   175→    return (
   176→      <div className="space-y-5">
   177→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   178→          {[1, 2, 3, 4].map((i) => (
   179→            <div key={i} className="macos-card p-4 animate-pulse">
   180→              <div className="h-3 bg-muted rounded w-20 mb-2" />
   181→              <div className="h-6 bg-muted rounded w-28 mb-1" />
   182→              <div className="h-3 bg-muted rounded w-16" />
   183→            </div>
   184→          ))}
   185→        </div>
   186→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   187→          {[1, 2, 3].map((i) => (
   188→            <div key={i} className="macos-card p-4 animate-pulse">
   189→              <div className="h-4 bg-muted rounded w-32 mb-3" />
   190→              <div className="h-3 bg-muted rounded w-48 mb-3" />
   191→              <div className="grid grid-cols-3 gap-2">
   192→                <div className="h-10 bg-muted rounded" />
   193→                <div className="h-10 bg-muted rounded" />
   194→                <div className="h-10 bg-muted rounded" />
   195→              </div>
   196→            </div>
   197→          ))}
   198→        </div>
   199→      </div>
   200→    )
   201→  }
   202→
   203→  // Error state

   205→    return (
   206→      <div className="macos-card p-12 flex flex-col items-center justify-center text-muted-foreground">
   207→        <AlertTriangle className="size-10 mb-3 text-teal-500" />
   208→        <p className="text-[13px] font-medium">Failed to load station data</p>

   212→        </button>

   216→

   225→    { label: 'Transactions', value: totalTransactions.toLocaleString(), icon: ShoppingCart, change: kpis ? `${kpis.activePumps?.value || 0}/${kpis.activePumps?.total || 0} pumps` : 'No data', trend: 'up' as const, color: 'text-orange-600 dark:text-orange-400', bg: 'bg-orange-500/10' },
   226→    { label: 'Active Alerts', value: totalAlerts.toString(), icon: AlertTriangle, change: kpis ? `${kpis.alerts?.critical || 0} critical` : 'No data', trend: (kpis?.alerts?.critical || 0) > 0 ? 'down' as const : 'up' as const, color: 'text-red-500 dark:text-red-400', bg: 'bg-red-500/10' },

   239→    revenue: s.revenue || 0,

   242→
   243→  return (

   259→        {aggregateCards.map((card, i) => (

   265→          >
   266→            <div className="flex items-start justify-between">
   267→              <div>
   268→                <p className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">{card.label}</p>
   269→                <p className="text-[20px] font-bold mt-1 tracking-tight">{card.value}</p>
   270→                <div className="flex items-center gap-1 mt-1">
   271→                  {card.trend === 'up' ? (
   272→                    <ArrowUpRight className="size-3 text-emerald-500" />
   273→                  ) : (
   274→                    <AlertTriangle className="size-3 text-red-500" />
   275→                  )}
   276→                  <span className={`text-[12px] font-medium ${card.trend === 'up' ? 'text-emerald-500' : 'text-red-500'}`}>
   277→                    {card.change}
   278→                  </span>
   279→                </div>
   280→              </div>
   281→              <div className={`flex size-9 items-center justify-center rounded-xl ${card.bg}`}>
   282→                <card.icon className={`size-[18px] ${card.color}`} />
   283→              </div>
   284→            </div>
   285→          </motion.div>
   286→        ))}
   287→      </motion.div>
   288→
   289→      {/* View Switcher */}
   290→      <div className="flex items-center gap-1 rounded-xl bg-black/[0.03] dark:bg-white/[0.04] p-1 w-fit">
   291→        {[
   292→          { key: 'grid' as const, label: 'Stations' },
   293→          { key: 'compare' as const, label: 'Comparison' },
   294→          { key: 'detail' as const, label: 'Detail View' },
   295→        ].map((tab) => (
   296→          <button
   297→            key={tab.key}
   298→            onClick={() => { setActiveView(tab.key); if (tab.key !== 'detail') setSelectedStation(null) }}
   299→            className={`px-4 py-1.5 rounded-lg text-[13px] font-medium transition-all duration-150 ${
   300→              activeView === tab.key
   301→                ? 'bg-card text-foreground shadow-sm'
   302→                : 'text-muted-foreground hover:text-foreground'
   303→            }`}
   304→          >
   305→            {tab.label}
   306→          </button>
   307→        ))}
   308→      </div>
   309→
   310→      {/* ── Stations Grid ── */}
   311→      <AnimatePresence mode="wait">
   312→        {activeView === 'grid' && (
   313→          <motion.div key="grid" {...fadeUp} className="space-y-6">
   314→            {stations.length === 0 ? (
   315→              <div className="macos-card p-12 flex flex-col items-center justify-center text-muted-foreground">
   316→                <Building2 className="size-10 mb-3 opacity-20" />
   317→                <p className="text-[13px] font-medium">No stations found</p>
   318→                <p className="text-[11px]">Stations will appear here once they are added to the system</p>
   319→              </div>
   320→            ) : (
   321→              Object.entries(byRegion).map(([region, regionStations]) => (
   322→                <div key={region}>
   323→                  <div className="macos-section-header flex items-center gap-2 px-0">
   324→                    <Globe className="size-3 text-teal-500" />
   325→                    {region}
   326→                    <span className="text-[10px] text-muted-foreground">({regionStations.length} stations)</span>
   327→                  </div>
   328→                  <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4 mt-2">
   329→                    {regionStations.map((station) => {
   330→                      const tc = typeConfig[station.stationType] || typeConfig.fuel
   331→                      return (
   332→                        <motion.div
   333→                          key={station.id}
   334→                          whileHover={{ y: -2 }}
   335→                          transition={{ duration: 0.15 }}
   336→                          className="macos-card p-4 cursor-pointer"
   337→                          onClick={() => { setSelectedStation(station); setActiveView('detail') }}
   338→                        >
   339→                          <div className="flex items-center justify-between mb-3">
   340→                            <div className="flex items-center gap-2 min-w-0">
   341→                              <Building2 className="size-4 text-teal-500 shrink-0" />
   342→                              <h4 className="text-[13px] font-semibold truncate">{station.name}</h4>
   343→                            </div>
   344→                            <span className={`macos-pill ${tc.bg} ${tc.color} shrink-0`}>{tc.label}</span>
   345→                          </div>
   346→                          <div className="flex items-center gap-1 text-[11px] text-muted-foreground mb-3">
   347→                            <MapPin className="size-3" />
   348→                            <span className="truncate">{station.address || station.city || 'No address'}</span>
   349→                          </div>
   350→                          <div className="grid grid-cols-3 gap-2 mb-3">
   351→                            <div className="text-center">
   352→                              <p className="text-[15px] font-bold">{station.dispenserCount}</p>
   353→                              <p className="text-[10px] text-muted-foreground">Pumps</p>
   354→                            </div>
   355→                            <div className="text-center">
   356→                              <p className="text-[15px] font-bold">{station.tankCount || '—'}</p>
   357→                              <p className="text-[10px] text-muted-foreground">Tanks</p>
   358→                            </div>
   359→                            <div className="text-center">
   360→                              <p className="text-[15px] font-bold">{station.employeeCount}</p>
   361→                              <p className="text-[10px] text-muted-foreground">Staff</p>
   362→                            </div>
   363→                          </div>
   364→                          {/* Fuel Level Bar */}
   365→                          {station.fuelLevelPercentage > 0 && (
   366→                            <div className="space-y-1 mb-2">
   367→                              <div className="flex items-center justify-between text-[11px]">
   368→                                <span className="text-muted-foreground">Fuel Level</span>
   369→                                <span className="font-medium">{station.fuelLevelPercentage}%</span>
   370→                              </div>
   371→                              <div className="h-1.5 w-full rounded-full bg-muted overflow-hidden">
   372→                                <div
   373→                                  className={`h-full rounded-full transition-all ${
   374→                                    station.fuelLevelPercentage < 25 ? 'bg-red-500' :
   375→                                    station.fuelLevelPercentage < 50 ? 'bg-teal-500' : 'bg-emerald-500'
   376→                                  }`}
   377→                                  style={{ width: `${Math.min(station.fuelLevelPercentage, 100)}%` }}
   378→                                />
   379→                              </div>
   380→                            </div>
   381→                          )}
   382→                          {/* Alerts Indicator */}
   383→                          {station.alerts && station.alerts.length > 0 && (
   384→                            <div className="flex items-center gap-1.5 mt-2 pt-2 border-t border-border/50">
   385→                              <AlertTriangle className={`size-3 ${station.alerts.some(a => a.severity === 'critical') ? 'text-red-500' : 'text-teal-500'}`} />

   389→                        </motion.div>

   391→                    })}
   392→                  </div>
   393→                </div>
   394→              ))
   395→            )}
   396→          </motion.div>
   397→        )}
   398→
   399→        {/* ── Comparison View ── */}
   400→        {activeView === 'compare' && (
   401→          <motion.div key="compare" {...fadeUp} className="space-y-5">
   402→            {stations.length === 0 ? (
   403→              <div className="macos-card p-12 flex flex-col items-center justify-center text-muted-foreground">
   404→                <BarChart3 className="size-10 mb-3 opacity-20" />
   405→                <p className="text-[13px] font-medium">No stations to compare</p>
   406→                <p className="text-[11px]">Add stations to see comparison data</p>

   408→            ) : (
   409→              <>
   410→                {/* Chart */}
   411→                <div className="macos-card p-5">
   412→                  <div className="flex items-center gap-2 mb-4">
   413→                    <BarChart3 className="size-4 text-teal-500" />
   414→                    <h3 className="text-[13px] font-semibold">Revenue Comparison</h3>

   416→                  <div className="h-64">

   418→                      <BarChart data={comparisonData} margin={{ top: 4, right: 4, bottom: 0, left: 0 }}>
   419→                        <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />

   422→                        <Tooltip
   423→                          contentStyle={{ fontSize: 12, borderRadius: 10, border: '0.5px solid var(--border)', boxShadow: '0 4px 12px rgba(0,0,0,0.08)' }}
   424→                          formatter={(value: number) => [formatPKR(value), 'Revenue']}

   426→                        <Bar dataKey="revenue" fill="#e8930c" radius={[6, 6, 0, 0]} />
   427→                      </BarChart>
   428→                    </ResponsiveContainer>

   430→                </div>
   431→
   432→                {/* Comparison Table */}
   433→                <div className="macos-card overflow-hidden">
   434→                  <ScrollArea className="max-h-[400px]">

   436→                      <thead>
   437→                        <tr>
   438→                          <th>Station</th>
   439→                          <th>Type</th>
   440→                          <th>City</th>
   441→                          <th className="text-right">Revenue</th>
   442→                          <th className="text-right">Volume</th>
   443→                          <th className="text-right">Transactions</th>
   444→                          <th className="text-right">Pumps</th>
