   173→export default function FleetModule() {
   174→  const [activeTab, setActiveTab] = useState<TabKey>('accounts')
   175→  const { selectedStation } = useAppStore()
   176→
   177→  // Data state
   178→  const [fleetAccounts, setFleetAccounts] = useState<FleetAccount[]>([])
   179→  const [summary, setSummary] = useState<FleetSummary | null>(null)
   180→  const [rfidReaders, setRfidReaders] = useState<RFIDReader[]>([])
   181→  const [loading, setLoading] = useState(true)
   182→  const [error, setError] = useState<string | null>(null)
   183→
   184→  const fetchData = useCallback(async () => {
   185→    setLoading(true)
   186→    setError(null)
   187→    try {
   188→      const [fleetRes, rfidRes] = await Promise.all([
   189→        fetch('/api/fleet?includeDetails=true'),
   190→        fetch('/api/rfid-readers'),
   191→      ])
   192→
   193→      if (!fleetRes.ok || !rfidRes.ok) {
   194→        throw new Error('Failed to fetch fleet data')
   195→      }
   196→
   197→      const fleetData = await fleetRes.json()
   198→      const rfidData = await rfidRes.json()
   199→
   200→      setFleetAccounts(fleetData.fleetAccounts || [])
   201→      setSummary(fleetData.summary || null)
   202→      setRfidReaders(rfidData.readers || [])
   203→    } catch (err) {
   204→      console.error('FleetModule fetch error:', err)
   205→      setError('Unable to load fleet data. Please try again.')
   206→    } finally {
   207→      setLoading(false)
   208→    }
   209→  }, [])
   210→
   211→  useEffect(() => {
   212→    fetchData()
   213→  }, [fetchData])
   214→
   215→  // Derived values
   216→  const allVehicles = fleetAccounts.flatMap(a => a.vehicles || [])
   217→  const allFuelCards = fleetAccounts.flatMap(a => a.fuelCards || [])
   218→  const totalVehicles = summary?.totalVehicles ?? allVehicles.length
   219→  const totalCards = summary?.totalFuelCards ?? allFuelCards.length
   220→  const totalOutstanding = summary?.totalCurrentBalance ?? fleetAccounts.reduce((s, a) => s + a.currentBalance, 0)
   221→
   222→  if (error) {
   223→    return <ErrorBanner message={error} onRetry={fetchData} />
   224→  }
   225→
   226→  return (
   227→    <div className="space-y-5" style={{ fontStyle: 'normal !important' }}>
   228→      {/* ── Summary Stats ── */}
   229→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   230→        {[
   231→          { label: 'Fleet Accounts', value: (summary?.totalAccounts ?? fleetAccounts.length).toString(), icon: Truck, accent: false },
   232→          { label: 'Total Vehicles', value: totalVehicles.toString(), icon: Car, accent: false },
   233→          { label: 'Active Cards', value: totalCards.toString(), icon: CreditCard, accent: false },
   234→          { label: 'Outstanding', value: pkr(totalOutstanding), icon: Wallet, accent: true },
   235→        ].map((stat, i) => (
   236→          <motion.div
   237→            key={stat.label}
   238→            initial={{ opacity: 0, y: 8 }}
   239→            animate={{ opacity: 1, y: 0 }}
   240→            transition={{ delay: i * 0.05, duration: 0.3 }}
   241→            className="macos-card p-4"
   242→          >
   243→            <div className="flex items-start justify-between">
   244→              <div>
   245→                <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">{stat.label}</p>
   246→                <p className={`text-[20px] font-bold mt-1 ${stat.accent ? 'fuel-text-gradient' : 'text-foreground'}`}>
   247→                  {loading ? <span className="inline-block w-16 h-5 rounded bg-muted/30 animate-pulse" /> : stat.value}
   248→                </p>
   249→              </div>
   250→              <div className={`size-8 rounded-lg flex items-center justify-center ${stat.accent ? 'bg-teal-500/10' : 'bg-black/[0.03] dark:bg-white/[0.05]'}`}>
   251→                <stat.icon className={`size-4 ${stat.accent ? 'text-teal-500' : 'text-muted-foreground'}`} strokeWidth={1.5} />
   252→              </div>
   253→            </div>
   254→          </motion.div>
   255→        ))}
   256→      </div>
   257→
   258→      {/* ── Tab Bar ── */}
   259→      <div className="flex items-center gap-1 overflow-x-auto pb-1 -mb-1">
   260→        {tabs.map((tab) => {
   261→          const isActive = activeTab === tab.key
   262→          const Icon = tab.icon
   263→          return (
   264→            <button
   265→              key={tab.key}
   266→              onClick={() => setActiveTab(tab.key)}
   267→              className={`flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[13px] font-medium transition-all duration-150 whitespace-nowrap ${
   268→                isActive
   269→                  ? 'bg-teal-500/10 text-teal-600 dark:text-teal-400'
   270→                  : 'text-muted-foreground hover:bg-black/[0.03] dark:hover:bg-white/[0.05] hover:text-foreground'
   271→              }`}
   272→            >
   273→              <Icon className="size-3.5" strokeWidth={isActive ? 2 : 1.5} />
   274→              {tab.label}
   275→            </button>
   276→          )
   277→        })}
   278→        <button
   279→          onClick={fetchData}
   280→          className="ml-auto flex items-center gap-1 rounded-lg px-2 py-1.5 text-[12px] text-muted-foreground hover:bg-black/[0.03] dark:hover:bg-white/[0.05] transition-colors"
   281→          title="Refresh data"
   282→        >
   283→          <RefreshCw className={`size-3.5 ${loading ? 'animate-spin' : ''}`} />
   284→        </button>
   285→      </div>
   286→
   287→      {/* ── Tab Content ── */}
   288→      <AnimatePresence mode="wait">
   289→        <motion.div
   290→          key={activeTab}
   291→          initial={{ opacity: 0, y: 6 }}
   292→          animate={{ opacity: 1, y: 0 }}
   293→          exit={{ opacity: 0, y: -4 }}
   294→          transition={{ duration: 0.2, ease: [0.25, 0.1, 0.25, 1] }}
   295→        >
   296→          {loading ? (
   297→            <SkeletonTable rows={6} />
   298→          ) : (
   299→            <>
   300→              {activeTab === 'accounts' && <FleetAccountsTab accounts={fleetAccounts} onRefresh={fetchData} />}
   301→              {activeTab === 'vehicles' && <VehiclesTab vehicles={allVehicles} />}
   302→              {activeTab === 'cards' && <FuelCardsTab cards={allFuelCards} />}
   303→              {activeTab === 'rfid' && <RFIDReadersTab readers={rfidReaders} />}
   304→              {activeTab === 'transactions' && <TransactionsTab accounts={fleetAccounts} />}
   305→            </>
   306→          )}
   307→        </motion.div>
   308→      </AnimatePresence>
   309→    </div>
   310→  )
   311→}

   684→interface TransactionRow {
   685→  id: string
   686→  time: string
   687→  company: string
   688→  plate: string
   689→  pump: string
   690→  product: string
   691→  litres: number
   692→  amount: number

   696→function TransactionsTab({ accounts }: { accounts: FleetAccount[] }) {
   697→  const [transactions, setTransactions] = useState<TransactionRow[]>([])
   698→  const [loading, setLoading] = useState(true)
   699→  const { selectedStation } = useAppStore()
   700→
   701→  useEffect(() => {
   702→    async function fetchTransactions() {
   703→      try {
   704→        const params = new URLSearchParams()
   705→        if (selectedStation && selectedStation !== 'all') params.set('stationId', selectedStation)
   706→        params.set('limit', '20')
   707→        const res = await fetch(`/api/fuelings?${params.toString()}`)
   708→        if (res.ok) {
   709→          const data = await res.json()
   710→          const txRows: TransactionRow[] = (data.fuelings || []).map((f: Record<string, unknown>) => ({
   711→            id: f.id?.toString().slice(0, 8) || '',
   712→            time: f.startedAt ? new Date(f.startedAt as string).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' }) : '',
   713→            company: (f.fleetAccount as Record<string, string>)?.companyName || 'Walk-in',
   714→            plate: (f.vehiclePlate as string) || '—',
   715→            pump: (f.dispenser as Record<string, string>)?.name || '—',
   716→            product: (f.fuelType as string) || '—',
   717→            litres: (f.volume as number) || 0,
   718→            amount: (f.totalAmount as number) || 0,
   719→            card: (f.rfidTag as string) || 'Cash',
   720→          }))
   721→          setTransactions(txRows)
   722→        }
   723→      } catch (err) {
   724→        console.error('Transactions fetch error:', err)
   725→      } finally {
   726→        setLoading(false)
   727→      }
   728→    }
   729→    fetchTransactions()
   730→  }, [selectedStation])
   731→
   732→  const totalAmount = transactions.reduce((s, t) => s + t.amount, 0)

   749→  return (
   750→    <div className="space-y-4">
   751→      {/* Quick Stats */}
   752→      <div className="grid grid-cols-3 gap-3">
   753→        <div className="macos-card p-4">
   754→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Recent Transactions</p>
