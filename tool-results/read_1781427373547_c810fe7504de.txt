   157→function mapApiDispenserStatus(status: string): DispenserStatus {
   158→  if (status === 'dispensing' || status === 'active') return 'dispensing'
   159→  if (status === 'error' || status === 'fault') return 'error'
   160→  if (status === 'offline' || status === 'disconnected') return 'offline'
   161→  return 'idle'
   162→}
   163→
   164→function mapApiNozzleStatus(status: string): NozzleStatus {
   165→  if (status === 'dispensing' || status === 'active') return 'dispensing'
   166→  if (status === 'error' || status === 'fault') return 'error'
   167→  if (status === 'offline' || status === 'disconnected') return 'offline'
   168→  return 'idle'
   169→}

   171→function timeAgo(dateStr: string): string {
   172→  if (!dateStr) return '—'
   173→  const now = new Date()
   174→  const date = new Date(dateStr)
   175→  const diffSec = Math.floor((now.getTime() - date.getTime()) / 1000)
   176→  if (diffSec < 60) return `${diffSec} sec ago`
   177→  if (diffSec < 3600) return `${Math.floor(diffSec / 60)} min ago`
   178→  return `${Math.floor(diffSec / 3600)} hr ago`
   179→}
   180→
   181→function formatTime(iso: string): string {
   182→  return new Date(iso).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false })
   183→}

   779→export default function ForecourtModule() {
   780→  const [selectedDispenserId, setSelectedDispenserId] = useState<string | null>(null)
   781→  const { selectedStation } = useAppStore()
   782→
   783→  // Data state
   784→  const [dispensers, setDispensers] = useState<Dispenser[]>([])
   785→  const [commandQueue, setCommandQueue] = useState<CommandQueueEntry[]>([])
   786→  const [loading, setLoading] = useState(true)
   787→  const [error, setError] = useState<string | null>(null)
   788→
   789→  const fetchData = useCallback(async () => {
   790→    setLoading(true)
   791→    setError(null)
   792→    try {
   793→      const stationParam = selectedStation && selectedStation !== 'all' ? `?stationId=${selectedStation}` : ''
   794→
   795→      const [dispRes, eventsRes] = await Promise.all([
   796→        fetch(`/api/dispensers${stationParam ? `?stationId=${selectedStation}` : ''}`),
   797→        fetch('/api/nozzle-events'),
   798→      ])
   799→
   800→      if (!dispRes.ok) throw new Error('Failed to fetch dispensers')
   801→
   802→      const dispData = await dispRes.json()
   803→      const eventsData = eventsRes.ok ? await eventsRes.json() : { events: [] }
   804→
   805→      // Map API dispensers to UI dispensers
   806→      const mappedDispensers: Dispenser[] = (dispData.dispensers || []).map((d: Record<string, unknown>, idx: number) => {
   807→        const apiStatus = mapApiDispenserStatus(d.status as string || 'idle')
   808→        const nozzles = (d.nozzles || []) as Record<string, unknown>[]
   809→        // Price defaults based on fuel type (PKR per litre)
   810→        const priceMap: Record<string, number> = {
   811→          petrol: 272.50, diesel: 285.00, hobc: 302.00, premium: 272.50, xl: 295.00,
   812→        }
   813→        return {
   814→          id: d.id as string,
   815→          name: (d.name as string) || `Dispenser ${idx + 1}`,
   816→          code: (d.code as string) || `DSP-${String(idx + 1).padStart(3, '0')}`,
   817→          status: apiStatus,
   818→          pumpLane: (d.pumpLane as number) || Math.floor(idx / 2) + 1,
   819→          nozzles: nozzles.map((n: Record<string, unknown>, nIdx: number) => {
   820→            const fuelType = (n.fuelType as string) || 'petrol'
   821→            const pricePerLitre = (n.pricePerLitre as number) || priceMap[fuelType.toLowerCase()] || 272.50
   822→            const meterReading = (n.meterReading as number) || 0
   823→            const previousReading = (n.previousReading as number) || 0
   824→            const lastTxnVol = meterReading - previousReading
   825→            return {
   826→              id: n.id as string,
   827→              nozzleNumber: (n.nozzleNumber as number) || nIdx + 1,
   828→              fuelType: fuelType,
   829→              fuelGrade: (n.fuelGrade as string) || fuelType.charAt(0).toUpperCase() + fuelType.slice(1),
   830→              pricePerLitre,
   831→              status: mapApiNozzleStatus((n.status as string) || 'idle'),
   832→              totalVolume: (n.totalizer as number) || (n.totalVolume as number) || meterReading,
   833→              lastTransactionVolume: lastTxnVol > 0 ? lastTxnVol : 0,
   834→              lastTransactionAmount: lastTxnVol > 0 ? Math.round(lastTxnVol * pricePerLitre) : 0,
   835→            }
   836→          }),
   837→          lastCommunication: (d.lastCommunication as string) || (d.updatedAt ? timeAgo(d.updatedAt as string) : '—'),
   838→        }
   839→      })
   840→
   841→      setDispensers(mappedDispensers)
   842→
   843→      // Build command queue from nozzle events
   844→      const apiEvents = eventsData.events || []
   845→      const cmdQueue: CommandQueueEntry[] = apiEvents
   846→        .slice(0, 10)
   847→        .map((evt: Record<string, unknown>, i: number) => {
   848→          const eventType = evt.eventType as string || 'unknown'
   849→          const nozzleInfo = evt.nozzle as Record<string, unknown> | undefined
   850→          const dispenserInfo = nozzleInfo?.dispenser as Record<string, unknown> | undefined
   851→          let command = eventType.toUpperCase().replace(/_/g, ' ')
   852→          let cmdStatus: CommandStatus = evt.isAuthorized ? 'acknowledged' : 'failed'
   853→          if (eventType === 'nozzle_up') command = 'AUTHORIZE'
   854→          else if (eventType === 'nozzle_down') command = 'STOP'
   855→          else if (eventType === 'fueling_start') command = 'AUTHORIZE'
   856→          else if (eventType === 'fueling_end') command = 'STOP'
   857→
   858→          return {
   859→            id: evt.id as string || `cmd-${i}`,
   860→            command,
   861→            target: dispenserInfo ? `${dispenserInfo.name} N${nozzleInfo?.nozzleNumber || '?'}` : `Nozzle ${evt.nozzleId}`,
   862→            status: cmdStatus,
   863→            timestamp: evt.createdAt ? formatTime(evt.createdAt as string) : '—',
   864→          }
   865→        })
   866→
   867→      setCommandQueue(cmdQueue)
   868→    } catch (err) {
   869→      console.error('ForecourtModule fetch error:', err)
   870→      setError('Unable to load forecourt data. Please try again.')
   871→    } finally {
   872→      setLoading(false)
   873→    }
   874→  }, [selectedStation])
   875→

   877→    fetchData()
   878→  }, [fetchData])
   879→
   880→  const selectedDispenser = dispensers.find((d) => d.id === selectedDispenserId) || null
   881→
   882→  // Summary stats
   883→  const summary = {
   884→    idle: dispensers.filter((d) => d.status === 'idle').length,
   885→    dispensing: dispensers.filter((d) => d.status === 'dispensing').length,
   886→    error: dispensers.filter((d) => d.status === 'error').length,
   887→    offline: dispensers.filter((d) => d.status === 'offline').length,
   888→  }
   889→
   890→  if (error) {
   891→    return <ErrorBanner message={error} onRetry={fetchData} />

   893→

   898→        initial={{ opacity: 0, y: -4 }}
   899→        animate={{ opacity: 1, y: 0 }}
   900→        transition={{ duration: 0.3, ease: [0.25, 0.1, 0.25, 1] }}
   901→        className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3"
   902→      >
   903→        <div className="flex items-center gap-3">
   904→          <div className="flex size-9 items-center justify-center rounded-xl bg-[var(--macos-accent)]/10">
   905→            <Fuel className="size-5 text-[var(--macos-accent)]" />
   906→          </div>
   907→          <div>
   908→            <h2 className="text-[15px] font-semibold text-foreground tracking-tight">Forecourt Control</h2>
   909→            <p className="text-[11px] text-muted-foreground">Real-time dispenser monitoring &amp; control</p>
   910→          </div>
   911→        </div>
   912→
   913→        <div className="flex items-center gap-3">
   914→          {/* Live Status */}
   915→          <div className="flex items-center gap-2 px-3 py-1.5 rounded-lg bg-emerald-500/5 border border-emerald-500/15">
   916→            <div className="macos-status-dot macos-status-online" />
   917→            <span className="text-[11px] font-semibold text-emerald-600 dark:text-emerald-400">LIVE</span>
