   120→// ── Helpers ──────────────────────────────────────────────
   121→
   122→function formatPKR(amount: number): string {
   123→  if (Math.abs(amount) >= 1000000) return `PKR ${(amount / 1000000).toFixed(2)}M`
   124→  if (Math.abs(amount) >= 1000) return `PKR ${(amount / 1000).toFixed(0)}K`
   125→  return `PKR ${amount.toLocaleString()}`
   126→}
   127→
   128→function formatFull(amount: number): string {
   129→  return `PKR ${amount.toLocaleString()}`
   130→}

   162→        fetch('/api/finance-advanced?type=journal_entries'),
   163→        fetch('/api/finance-advanced?type=receivables'),
   164→        fetch('/api/finance-advanced?type=payables'),
   165→        fetch('/api/finance-advanced?type=tax_configs'),
   166→      ])
   167→
   168→      if (!financeRes.ok) throw new Error('Failed to load finance data')
   169→
   170→      const financeJson = await financeRes.json()
   171→      setFinanceData(financeJson)
   172→
   173→      if (glRes.ok) {
   174→        const glJson = await glRes.json()
   175→        setGLAccounts(glJson.glAccounts || [])
   176→      }
   177→      if (jeRes.ok) {
   178→        const jeJson = await jeRes.json()
   179→        setJournalEntries(jeJson.journalEntries || [])
   180→      }
   181→      if (arRes.ok) {
   182→        const arJson = await arRes.json()
   183→        setReceivables(arJson.receivables || [])
   184→      }

   186→        const apJson = await apRes.json()
   187→        setPayables(apJson.payables || [])

   199→
   200→  useEffect(() => {
   201→    fetchData()
   202→  }, [fetchData])
   203→
   204→  const tabs = [
   205→    { key: 'overview' as const, label: 'Overview' },
   206→    { key: 'gl' as const, label: 'GL Accounts' },
   207→    { key: 'ar-ap' as const, label: 'AR / AP' },
   208→    { key: 'tax' as const, label: 'Tax' },
   209→  ]
   210→
   211→  // Build summary cards from finance data
   212→  const summaryCards = financeData ? [
   213→    { label: 'Revenue', value: financeData.summary.totalIncome, icon: TrendingUp, color: 'text-emerald-600 dark:text-emerald-400', bg: 'bg-emerald-500/10', trend: 'up' as const },
   214→    { label: 'Expenses', value: financeData.summary.totalExpenses, icon: TrendingDown, color: 'text-red-500 dark:text-red-400', bg: 'bg-red-500/10', trend: 'down' as const },
   215→    { label: 'Net Profit', value: financeData.summary.netProfit, icon: Wallet, color: 'text-teal-600 dark:text-teal-400', bg: 'bg-teal-500/10', trend: 'up' as const },
   216→    { label: 'Outstanding', value: financeData.supplierInvoices.totalPending, icon: Receipt, color: 'text-rose-500 dark:text-rose-400', bg: 'bg-rose-500/10', trend: 'down' as const },
   217→  ] : []
   218→
   219→  // Build cash flow data from income/expense categories
   220→  const cashFlowData = (() => {
   221→    if (!financeData) return []
   222→    // We'll create a simple single-period view from the data
   223→    return [{
   224→      month: new Date().toLocaleDateString('en-US', { month: 'short' }),
   225→      inflow: financeData.summary.totalIncome,
   226→      outflow: financeData.summary.totalExpenses,
   227→    }]
   228→  })()
   229→
   230→  // ── Loading State ──
   231→  if (loading && !financeData) {
   232→    return (
   233→      <div className="space-y-5">
   234→        <div className="flex items-center gap-1 rounded-xl bg-black/[0.03] dark:bg-white/[0.04] p-1 w-fit">
   235→          {[...Array(4)].map((_, i) => (
   236→            <div key={i} className="px-4 py-1.5 w-24 h-7 bg-muted/20 rounded-lg animate-pulse" />
   237→          ))}
   238→        </div>
   239→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   240→          {[...Array(4)].map((_, i) => (
   241→            <div key={i} className="macos-card p-4 h-[110px] animate-pulse bg-muted/10" />
   242→          ))}
   243→        </div>
   244→      </div>
   245→    )
   246→  }
   247→
   248→  // ── Error State ──
   249→  if (error && !financeData) {
   250→    return (
   251→      <div className="macos-card p-8 flex flex-col items-center justify-center text-center min-h-[300px]">
   252→        <AlertCircle className="size-10 text-teal-500 mb-4" />
   253→        <h3 className="text-[15px] font-semibold text-foreground mb-1">Failed to load finance data</h3>
   254→        <p className="text-[13px] text-muted-foreground mb-4">{error}</p>
   255→        <button onClick={fetchData} className="macos-btn flex items-center gap-1.5">
   256→          <RefreshCw className="size-3.5" /> Retry
   257→        </button>
   258→      </div>
   259→    )
   260→  }
   261→
   262→  return (
   263→    <div className="space-y-5">
   264→      {/* Tab Bar */}
   265→      <div className="flex items-center justify-between">
   266→        <div className="flex items-center gap-1 rounded-xl bg-black/[0.03] dark:bg-white/[0.04] p-1 w-fit">
   267→          {tabs.map((tab) => (
   268→            <button
   269→              key={tab.key}
   270→              onClick={() => setActiveTab(tab.key)}
   271→              className={`px-4 py-1.5 rounded-lg text-[13px] font-medium transition-all duration-150 ${
   272→                activeTab === tab.key
   273→                  ? 'bg-card text-foreground shadow-sm'
   274→                  : 'text-muted-foreground hover:text-foreground'
   275→              }`}
   276→            >
   277→              {tab.label}
   278→            </button>

   307→                key={card.label}
   308→                variants={{ initial: { opacity: 0, y: 8 }, animate: { opacity: 1, y: 0 } }}

   311→              >
   312→                <div className="flex items-start justify-between">
   313→                  <div>
   314→                    <p className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">{card.label}</p>

   316→                    {financeData && card.label === 'Net Profit' && (
   317→                      <div className="flex items-center gap-1 mt-1.5">
   318→                        <span className="text-[11px] text-muted-foreground">Margin: {financeData.summary.profitMargin}%</span>
   319→                      </div>
   320→                    )}
   321→                    {financeData && card.label === 'Expenses' && (
   322→                      <div className="flex items-center gap-1 mt-1.5">
   323→                        <span className="text-[11px] text-muted-foreground">Payroll: {formatPKR(financeData.summary.payrollCosts)}</span>
   324→                      </div>
   325→                    )}
   326→                    {financeData && card.label === 'Outstanding' && (
   327→                      <div className="flex items-center gap-1 mt-1.5">
   328→                        <span className="text-[11px] text-muted-foreground">{financeData.supplierInvoices.pendingCount} pending invoices</span>
   329→                      </div>
   330→                    )}
   331→                    {financeData && card.label === 'Revenue' && (
   332→                      <div className="flex items-center gap-1 mt-1.5">
   333→                        <span className="text-[11px] text-muted-foreground">Fuel: {formatPKR(financeData.summary.totalFuelRevenue)}</span>
   334→                      </div>
   335→                    )}
   336→                  </div>
   337→                  <div className={`flex size-9 items-center justify-center rounded-xl ${card.bg}`}>
   338→                    <card.icon className={`size-[18px] ${card.color}`} />
   339→                  </div>
   340→                </div>
   341→              </motion.div>
   342→            ))}
   343→          </motion.div>
   344→
   345→          {/* Cash Flow Chart */}
   346→          <div className="macos-card p-5">
   347→            <div className="flex items-center justify-between mb-4">
   348→              <div>
   349→                <h3 className="text-[13px] font-semibold">Cash Flow</h3>
   350→                <p className="text-[11px] text-muted-foreground mt-0.5">Income vs expenses for {financeData?.period || 'current period'}</p>
   351→              </div>
   352→              <div className="flex items-center gap-4">
   353→                <div className="flex items-center gap-1.5">
   354→                  <div className="size-2.5 rounded-full bg-emerald-500" />
   355→                  <span className="text-[11px] text-muted-foreground">Income</span>
   356→                </div>
   357→                <div className="flex items-center gap-1.5">
   358→                  <div className="size-2.5 rounded-full bg-rose-400" />
   359→                  <span className="text-[11px] text-muted-foreground">Expenses</span>
   360→                </div>
   361→              </div>
   362→            </div>
   363→            {cashFlowData.length > 0 ? (
   364→              <div className="h-64">
   365→                <ResponsiveContainer width="100%" height="100%">
   366→                  <AreaChart data={cashFlowData} margin={{ top: 4, right: 4, bottom: 0, left: 0 }}>
   367→                    <defs>
   368→                      <linearGradient id="inflowGrad" x1="0" y1="0" x2="0" y2="1">
   369→                        <stop offset="0%" stopColor="#34c759" stopOpacity={0.2} />
   370→                        <stop offset="100%" stopColor="#34c759" stopOpacity={0} />
   371→                      </linearGradient>
   372→                      <linearGradient id="outflowGrad" x1="0" y1="0" x2="0" y2="1">
   373→                        <stop offset="0%" stopColor="#ff6b6b" stopOpacity={0.15} />
   374→                        <stop offset="100%" stopColor="#ff6b6b" stopOpacity={0} />
   375→                      </linearGradient>
   376→                    </defs>
   377→                    <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />
   378→                    <XAxis dataKey="month" tick={{ fontSize: 11, fill: 'var(--muted-foreground)' }} axisLine={false} tickLine={false} />
   379→                    <YAxis tick={{ fontSize: 11, fill: 'var(--muted-foreground)' }} axisLine={false} tickLine={false} tickFormatter={(v: number) => `${(v / 1000000).toFixed(0)}M`} />
   380→                    <Tooltip
   381→                      contentStyle={{ fontSize: 12, borderRadius: 10, border: '0.5px solid var(--border)', boxShadow: '0 4px 12px rgba(0,0,0,0.08)' }}
   382→                      formatter={(value: number, name: string) => [formatFull(value), name === 'inflow' ? 'Income' : 'Expenses']}
   383→                    />
   384→                    <Area type="monotone" dataKey="inflow" stroke="#34c759" strokeWidth={2} fill="url(#inflowGrad)" />
   385→                    <Area type="monotone" dataKey="outflow" stroke="#ff6b6b" strokeWidth={2} fill="url(#outflowGrad)" />
   386→                  </AreaChart>
   387→                </ResponsiveContainer>
   388→              </div>
   389→            ) : (
   390→              <div className="h-64 flex items-center justify-center text-muted-foreground text-[13px]">
   391→                No cash flow data available
   392→              </div>
   393→            )}
   394→          </div>
   395→
   396→          {/* Income & Expense Breakdown */}
   397→          {financeData && (Object.keys(financeData.incomeByCategory).length > 0 || Object.keys(financeData.expenseByCategory).length > 0) && (
   398→            <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   399→              {/* Income Breakdown */}
   400→              {Object.keys(financeData.incomeByCategory).length > 0 && (
   401→                <div className="macos-card p-4">
   402→                  <h3 className="text-[13px] font-semibold mb-3">Income by Category</h3>
   403→                  <div className="macos-group">
   404→                    {Object.entries(financeData.incomeByCategory)
   405→                      .sort(([, a], [, b]) => b - a)
   406→                      .map(([category, amount]) => (
   407→                        <div key={category} className="macos-group-item flex items-center justify-between">
   408→                          <span className="text-[13px] truncate">{category}</span>
   409→                          <span className="text-[13px] font-medium tabular-nums">{formatPKR(amount)}</span>
   410→                        </div>
   411→                      ))}
   412→                  </div>
   413→                </div>
   414→              )}
   415→              {/* Expense Breakdown */}
   416→              {Object.keys(financeData.expenseByCategory).length > 0 && (
   417→                <div className="macos-card p-4">
   418→                  <h3 className="text-[13px] font-semibold mb-3">Expenses by Category</h3>
   419→                  <div className="macos-group">
   420→                    {Object.entries(financeData.expenseByCategory)
   421→                      .sort(([, a], [, b]) => b - a)
   422→                      .map(([category, amount]) => (
   423→                        <div key={category} className="macos-group-item flex items-center justify-between">
   424→                          <span className="text-[13px] truncate">{category}</span>
   425→                          <span className="text-[13px] font-medium tabular-nums">{formatPKR(amount)}</span>
   426→                        </div>
   427→                      ))}
   428→                  </div>
   429→                </div>
   430→              )}
   431→            </div>
   432→          )}
   433→
   434→          {/* Recent Journal Entries */}
   435→          <div className="macos-card overflow-hidden">
   436→            <div className="flex items-center justify-between p-4 pb-0">
   437→              <div className="flex items-center gap-2">
   438→                <FileText className="size-4 text-teal-500" />
   439→                <h3 className="text-[13px] font-semibold">Recent Journal Entries</h3>
   440→              </div>
   441→              <Badge variant="outline" className="text-[11px] px-2 py-0.5">{journalEntries.length} entries</Badge>
   442→            </div>
   443→            <ScrollArea className="max-h-[360px]">
   444→              {journalEntries.length > 0 ? (
   445→                <table className="macos-table">
   446→                  <thead>
   447→                    <tr>
   448→                      <th>Entry ID</th>
   449→                      <th>Date</th>
   450→                      <th>Description</th>
   451→                      <th>Status</th>
   452→                      <th className="text-right">Debit</th>
   453→                      <th className="text-right">Credit</th>
   454→                    </tr>
   455→                  </thead>
   456→                  <tbody>
   457→                    {journalEntries.map((entry) => {
   458→                      const totalDebit = entry.lines?.reduce((s, l) => s + l.debit, 0) || 0
   459→                      const totalCredit = entry.lines?.reduce((s, l) => s + l.credit, 0) || 0

   461→                        <tr key={entry.id}>
   462→                          <td className="font-mono text-[12px] text-teal-600 dark:text-teal-400">{entry.entryNumber}</td>

   465→                          <td>
   466→                            <span className={`macos-pill ${
   467→                              entry.status === 'posted'

   470→                            }`}>
   471→                              {entry.status}
   472→                            </span>
   473→                          </td>
   474→                          <td className="text-right font-medium text-[13px]">{totalDebit > 0 ? formatPKR(totalDebit) : '—'}</td>
   475→                          <td className="text-right font-medium text-[13px] text-emerald-600 dark:text-emerald-400">{totalCredit > 0 ? formatPKR(totalCredit) : '—'}</td>
   476→                        </tr>
   477→                      )
   478→                    })}
   479→                  </tbody>
   480→                </table>
   481→              ) : (
   482→                <div className="p-6 text-center text-muted-foreground text-[13px]">
   483→                  No journal entries found
   484→                </div>
   485→              )}
   486→            </ScrollArea>
   487→          </div>
   488→        </motion.div>
   489→      )}
   490→
   491→      {/* ── GL Accounts Tab ── */}
   492→      {activeTab === 'gl' && (
   493→        <motion.div {...fadeUp} className="space-y-4">
   494→          <div className="flex items-center gap-2">
   495→            <Calculator className="size-4 text-teal-500" />
   496→            <h3 className="text-[13px] font-semibold">General Ledger Accounts</h3>
   497→          </div>
   498→          {glAccounts.length > 0 ? (
   499→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">

   504→                  <div key={type} className="macos-group">
   505→                    <div className="macos-group-item flex items-center justify-between bg-black/[0.02] dark:bg-white/[0.02]">
   506→                      <span className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">{type}</span>
   507→                      <span className="text-[11px] text-muted-foreground">{accounts.length} accounts</span>
