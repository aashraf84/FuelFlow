    83→const fuelIconMap: Record<string, React.ElementType> = {
    84→  petrol: Droplets,
    85→  diesel: Flame,
    86→  hobc: Zap,
    87→  premium: DollarSign,
    88→  default: DollarSign,

    91→const fuelColorMap: Record<string, string> = {
    92→  petrol: '#34c759',
    93→  diesel: '#e8930c',
    94→  hobc: '#ff9500',
    95→  premium: '#ff6b35',
    96→  default: '#e8930c',

    99→function getFuelIcon(fuelType: string): React.ElementType {
   100→  const lower = fuelType.toLowerCase()
   101→  for (const [key, Icon] of Object.entries(fuelIconMap)) {
   102→    if (lower.includes(key)) return Icon
   103→  }
   104→  return fuelIconMap.default
   105→}
   106→
   107→function getFuelColor(fuelType: string): string {
   108→  const lower = fuelType.toLowerCase()
   109→  for (const [key, color] of Object.entries(fuelColorMap)) {
   110→    if (lower.includes(key)) return color
   111→  }
   112→  return fuelColorMap.default
   113→}

   117→function CustomTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ name: string; value: number; color: string }>; label?: string }) {
   118→  if (!active || !payload) return null
   119→  return (
   120→    <div className="macos-card p-3 shadow-lg !rounded-lg">
   121→      <p className="text-[11px] font-semibold text-muted-foreground mb-1.5">{label}</p>
   122→      {payload.map((p) => (
   123→        <div key={p.name} className="flex items-center gap-2 text-[12px]">
   124→          <div className="size-2 rounded-full" style={{ background: p.color }} />
   125→          <span className="text-muted-foreground">{p.name}:</span>
   126→          <span className="font-semibold">PKR {p.value}</span>
   127→        </div>
   128→      ))}
   129→    </div>
   130→  )
   131→}

   133→// ── Component ─────────────────────────────────────────────

   135→export default function PricingModule() {
   136→  const [activeTab, setActiveTab] = useState<TabKey>('current')
   137→  const [prices, setPrices] = useState<FuelPrice[]>([])
   138→  const [priceBoards, setPriceBoards] = useState<PriceBoard[]>([])
   139→  const [loading, setLoading] = useState(true)
   140→  const [error, setError] = useState<string | null>(null)
   141→
   142→  const fetchData = useCallback(async () => {
   143→    setLoading(true)
   144→    setError(null)
   145→    try {
   146→      const [pricesRes, boardsRes] = await Promise.all([
   147→        fetch('/api/prices'),
   148→        fetch('/api/price-boards'),
   149→      ])
   150→      if (!pricesRes.ok) throw new Error('Failed to load fuel prices')
   151→      if (!boardsRes.ok) throw new Error('Failed to load price boards')
   152→      const pricesJson = await pricesRes.json()
   153→      const boardsJson = await boardsRes.json()
   154→      setPrices(pricesJson.prices || [])
   155→      setPriceBoards(boardsJson.boards || [])
   156→    } catch (err) {
   157→      setError(err instanceof Error ? err.message : 'Unknown error')
   158→    } finally {
   159→      setLoading(false)
   160→    }
   161→  }, [])
   162→
   163→  useEffect(() => {
   164→    fetchData()
   165→  }, [fetchData])
   166→
   167→  // Build price history from all prices (including expired ones if they were loaded)
   168→  // Since the API only returns active prices, we'll derive a simple history from current data
   169→  const priceHistoryData = (() => {
   170→    // Group prices by fuel type
   171→    const typeMap: Record<string, { name: string; current: number; previous: number | null; effectiveFrom: string }> = {}
   172→    for (const p of prices) {
   173→      const key = p.fuelType.toLowerCase()
   174→      if (!typeMap[key] || new Date(p.effectiveFrom) > new Date(typeMap[key].effectiveFrom)) {
   175→        typeMap[key] = {
   176→          name: p.fuelType.charAt(0).toUpperCase() + p.fuelType.slice(1),
   177→          current: p.pricePerUnit,
   178→          previous: p.previousPrice,
   179→          effectiveFrom: p.effectiveFrom,
   180→        }
   181→      }
   182→    }
   183→    return typeMap
   184→  })()
   185→
   186→  return (
   187→    <div className="space-y-5">
   188→      {/* ── Tab Bar ── */}
   189→      <div className="flex items-center justify-between">
   190→        <div className="flex items-center gap-1 overflow-x-auto pb-1 -mb-1">
   191→          {tabs.map((tab) => {
   192→            const isActive = activeTab === tab.key
   193→            const Icon = tab.icon
   194→            return (
   195→              <button
   196→                key={tab.key}
   197→                onClick={() => setActiveTab(tab.key)}
   198→                className={`flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[13px] font-medium transition-all duration-150 whitespace-nowrap ${
   199→                  isActive
   200→                    ? 'bg-teal-500/10 text-teal-600 dark:text-teal-400'
   201→                    : 'text-muted-foreground hover:bg-black/[0.03] dark:hover:bg-white/[0.05] hover:text-foreground'
   202→                }`}
   203→              >
   204→                <Icon className="size-3.5" strokeWidth={isActive ? 2 : 1.5} />
   205→                {tab.label}
   206→              </button>
   207→            )
   208→          })}
   209→        </div>
   210→        <button
   211→          onClick={fetchData}
   212→          disabled={loading}
   213→          className="macos-btn-secondary flex items-center gap-1.5 text-[12px] px-3 py-1.5"
   214→        >
   215→          {loading ? (
   216→            <Loader2 className="size-3 animate-spin" />
   217→          ) : (
   218→            <RefreshCw className="size-3" />
   219→          )}
   220→          Refresh
   221→        </button>
   222→      </div>
   223→
   224→      {/* ── Tab Content ── */}
   225→      <AnimatePresence mode="wait">
   226→        <motion.div
   227→          key={activeTab}
   228→          initial={{ opacity: 0, y: 6 }}
   229→          animate={{ opacity: 1, y: 0 }}
   230→          exit={{ opacity: 0, y: -4 }}
   231→          transition={{ duration: 0.2, ease: [0.25, 0.1, 0.25, 1] }}
   232→        >
   233→          {activeTab === 'current' && <CurrentPricesTab prices={prices} loading={loading} error={error} onRetry={fetchData} />}
   234→          {activeTab === 'history' && <PriceHistoryTab priceHistoryData={priceHistoryData} />}
   235→          {activeTab === 'boards' && <PriceBoardsTab boards={priceBoards} onSync={fetchData} />}
   236→          {activeTab === 'schedule' && <ScheduleTab prices={prices} />}
   237→        </motion.div>
   238→      </AnimatePresence>
   239→    </div>
   240→  )
   241→}

   243→// ── Current Prices Tab ────────────────────────────────────
   244→
   245→function CurrentPricesTab({ prices, loading, error, onRetry }: {
   246→  prices: FuelPrice[]
   247→  loading: boolean
   248→  error: string | null
   249→  onRetry: () => void
   250→}) {
   251→  if (loading && prices.length === 0) {
   252→    return (
   253→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   254→        {[...Array(4)].map((_, i) => (
   255→          <div key={i} className="macos-card p-5 h-[180px] animate-pulse bg-muted/10" />
   256→        ))}
   257→      </div>
   258→    )
   259→  }
   260→
   261→  if (error && prices.length === 0) {
   262→    return (
   263→      <div className="macos-card p-8 flex flex-col items-center justify-center text-center min-h-[200px]">
   264→        <AlertTriangle className="size-10 text-teal-500 mb-4" />
   265→        <h3 className="text-[15px] font-semibold text-foreground mb-1">Failed to load prices</h3>
   266→        <p className="text-[13px] text-muted-foreground mb-4">{error}</p>
   267→        <button onClick={onRetry} className="macos-btn flex items-center gap-1.5">
   268→          <RefreshCw className="size-3.5" /> Retry
   269→        </button>
   270→      </div>
   271→    )
   272→  }
   273→
   274→  if (prices.length === 0) {
   275→    return (
   276→      <div className="macos-card p-8 flex flex-col items-center justify-center text-center min-h-[200px]">
   277→        <DollarSign className="size-10 text-muted-foreground mb-4" />
   278→        <h3 className="text-[15px] font-semibold text-foreground mb-1">No active prices</h3>
   279→        <p className="text-[13px] text-muted-foreground">Set up fuel prices to get started.</p>
   280→      </div>
   281→    )
   282→  }
   283→
   284→  // Deduplicate by fuel type (keep latest active price per type)
   285→  const uniquePrices = (() => {
   286→    const map = new Map<string, FuelPrice>()
   287→    for (const p of prices) {
   288→      const key = p.fuelType.toLowerCase()
   289→      const existing = map.get(key)
   290→      if (!existing || new Date(p.effectiveFrom) > new Date(existing.effectiveFrom)) {
   291→        map.set(key, p)
   292→      }
   293→    }
   294→    return Array.from(map.values())
   295→  })()
   296→
   297→  return (
   298→    <div className="space-y-4">
   299→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   300→        {uniquePrices.map((fuel, i) => {
   301→          const change = fuel.previousPrice ? fuel.pricePerUnit - fuel.previousPrice : 0
   302→          const isUp = change > 0
   303→          const Icon = getFuelIcon(fuel.fuelType)
   304→          const color = getFuelColor(fuel.fuelType)
   305→          return (
   306→            <motion.div
   307→              key={fuel.id}
   308→              initial={{ opacity: 0, y: 12 }}
   309→              animate={{ opacity: 1, y: 0 }}
   310→              transition={{ delay: i * 0.07, duration: 0.35 }}
   311→              className="macos-card p-5"
   312→            >
   313→              <div className="flex items-center justify-between mb-4">
   314→                <div className="flex items-center gap-2">
   315→                  <div className="size-8 rounded-lg flex items-center justify-center" style={{ background: `${color}15` }}>
   316→                    <Icon className="size-4" style={{ color }} />
   317→                  </div>
   318→                  <div>
   319→                    <p className="text-[13px] font-semibold">{fuel.fuelType.charAt(0).toUpperCase() + fuel.fuelType.slice(1)}</p>
   320→                    <p className="text-[11px] text-muted-foreground">{fuel.fuelGrade || 'Standard'}</p>
   321→                  </div>
   322→                </div>
   323→                <span className="macos-pill bg-teal-500/10 text-teal-600 dark:text-teal-400">Active</span>
   324→              </div>
   325→
   326→              {/* Big Number */}
   327→              <p className="text-[32px] font-bold tracking-tight text-foreground leading-none">
   328→                {fuel.pricePerUnit.toFixed(2)}
   329→              </p>
   330→              <p className="text-[11px] text-muted-foreground mt-0.5">{fuel.currency}/L</p>
   331→
   332→              {/* Change indicator */}
   333→              {fuel.previousPrice !== null && fuel.previousPrice > 0 && (
   334→                <div className="flex items-center gap-1.5 mt-3 pt-3 border-t border-border">
   335→                  {isUp ? (
   336→                    <ArrowUpRight className="size-3.5 text-red-500" />
   337→                  ) : (
   338→                    <ArrowDownRight className="size-3.5 text-emerald-500" />
   339→                  )}
   340→                  <span className={`text-[12px] font-semibold ${isUp ? 'text-red-500' : 'text-emerald-500'}`}>
   341→                    PKR {Math.abs(change).toFixed(2)}
   342→                  </span>
   343→                  <span className="text-[11px] text-muted-foreground">vs previous</span>
   344→                </div>
   345→              )}
   346→            </motion.div>
   347→          )
   348→        })}
   349→      </div>
   350→
   351→      {/* Last Update Info */}
   352→      <div className="flex items-center justify-between macos-card p-3 px-4">
   353→        <div className="flex items-center gap-2">
   354→          <div className="macos-status-dot macos-status-online" />
   355→          <span className="text-[12px] text-muted-foreground">Prices synced — Last updated: <span className="font-medium text-foreground">{uniquePrices.length > 0 ? new Date(uniquePrices[0].effectiveFrom).toLocaleString() : 'N/A'}</span></span>
   356→        </div>
   357→        <button onClick={onRetry} className="macos-btn-secondary gap-1">
   358→          <RefreshCw className="size-3" /> Sync Now
   359→        </button>
   360→      </div>
   361→    </div>
   362→  )
   363→}

   365→// ── Price History Tab ─────────────────────────────────────

   367→function PriceHistoryTab({ priceHistoryData }: { priceHistoryData: Record<string, { name: string; current: number; previous: number | null; effectiveFrom: string }> }) {
   368→  const entries = Object.values(priceHistoryData)
   369→
   370→  if (entries.length === 0) {
   371→    return (
   372→      <div className="macos-card p-8 flex flex-col items-center justify-center text-center min-h-[200px]">
   373→        <BarChart3 className="size-10 text-muted-foreground mb-4" />
   374→        <h3 className="text-[15px] font-semibold text-foreground mb-1">No price history</h3>
   375→        <p className="text-[13px] text-muted-foreground">Price history will appear once prices are updated.</p>
   376→      </div>
   377→    )
   378→  }
   379→
   380→  // Build chart data - show current vs previous for each fuel type
   381→  const chartData = entries.map(e => ({
   382→    name: e.name,
   383→    current: e.current,
   384→    previous: e.previous || 0,
   385→  }))
   386→
   387→  return (
   388→    <div className="space-y-4">
   389→      <div className="macos-card p-5">
   390→        <div className="flex items-center justify-between mb-4">
   391→          <div>
   392→            <h3 className="text-[13px] font-semibold">Current vs Previous Prices</h3>
   393→            <p className="text-[11px] text-muted-foreground mt-0.5">PKR per litre comparison</p>
   394→          </div>
   395→        </div>
   396→        <div className="h-[320px]">
   397→          <ResponsiveContainer width="100%" height="100%">
   398→            <LineChart data={chartData} margin={{ top: 5, right: 10, left: 0, bottom: 5 }}>
   399→              <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" />
   400→              <XAxis
   401→                dataKey="name"
   402→                tick={{ fontSize: 11, fill: 'var(--muted-foreground)' }}
   403→                axisLine={{ stroke: 'var(--border)' }}
   404→                tickLine={false}
   405→              />
   406→              <YAxis
   407→                tick={{ fontSize: 11, fill: 'var(--muted-foreground)' }}
   408→                axisLine={false}
   409→                tickLine={false}
   410→                tickFormatter={(v) => `${v}`}
   411→              />
   412→              <Tooltip content={<CustomTooltip />} />
   413→              <Legend
   414→                iconType="circle"
   415→                iconSize={8}
   416→                wrapperStyle={{ fontSize: '12px', paddingTop: '12px' }}
   417→              />
   418→              <Line type="monotone" dataKey="current" name="Current" stroke="#e8930c" strokeWidth={2} dot={{ r: 4, fill: '#e8930c' }} activeDot={{ r: 5 }} />
   419→              <Line type="monotone" dataKey="previous" name="Previous" stroke="#e8930c" strokeOpacity={0.4} strokeWidth={2} strokeDasharray="5 5" dot={{ r: 3, fill: '#e8930c', fillOpacity: 0.4 }} activeDot={{ r: 5 }} />
   420→            </LineChart>
   421→          </ResponsiveContainer>
   422→        </div>
   423→      </div>
   424→
   425→      {/* Price Change Table */}
   426→      <div className="macos-card overflow-hidden">
   427→        <table className="macos-table">
   428→          <thead>
   429→            <tr>
   430→              <th>Fuel Type</th>
   431→              <th className="text-right">Previous Price</th>
   432→              <th className="text-right">Current Price</th>
   433→              <th className="text-right">Change</th>
   434→              <th>Effective From</th>
   435→            </tr>
   436→          </thead>
   437→          <tbody>
   438→            {entries.map((entry) => {
   439→              const change = entry.previous ? entry.current - entry.previous : 0
   440→              const isUp = change > 0
   441→              return (
   442→                <tr key={entry.name}>
   443→                  <td className="font-semibold">{entry.name}</td>
   444→                  <td className="text-right font-mono">{entry.previous ? pkr(entry.previous) : '—'}</td>
   445→                  <td className="text-right font-mono font-semibold">{pkr(entry.current)}</td>
   446→                  <td className="text-right">
   447→                    {entry.previous ? (
   448→                      <span className={`inline-flex items-center gap-1 font-semibold text-[12px] ${isUp ? 'text-red-500' : change < 0 ? 'text-emerald-500' : 'text-muted-foreground'}`}>
   449→                        {isUp ? <ArrowUpRight className="size-3" /> : change < 0 ? <ArrowDownRight className="size-3" /> : null}
   450→                        PKR {Math.abs(change).toFixed(2)}
   451→                      </span>
   452→                    ) : '—'}
   453→                  </td>
   454→                  <td className="text-[12px] text-muted-foreground">{new Date(entry.effectiveFrom).toLocaleDateString()}</td>
   455→                </tr>
   456→              )
   457→            })}
   458→          </tbody>
   459→        </table>
   460→      </div>
   461→    </div>
   462→  )
   463→}

   467→function PriceBoardsTab({ boards, onSync }: { boards: PriceBoard[]; onSync: () => void }) {
   468→  const onlineCount = boards.filter(b => b.status === 'online' || b.status === 'active').length
   469→
   470→  return (
   471→    <div className="space-y-4">
   472→      <div className="grid grid-cols-3 gap-3">
   473→        <div className="macos-card p-4">
   474→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Boards Online</p>

   476→        </div>
   477→        <div className="macos-card p-4">
   478→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Board Types</p>
   479→          <p className="text-[20px] font-bold text-foreground mt-1">{new Set(boards.map(b => b.boardType)).size}</p>
   480→        </div>
   481→        <div className="macos-card p-4">
   482→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Last Sync</p>
   483→          <p className="text-[16px] font-bold text-foreground mt-1">
   484→            {boards.length > 0 && boards[0].lastSyncAt
   485→              ? new Date(boards[0].lastSyncAt).toLocaleTimeString()
   486→              : 'Never'}
   487→          </p>
   488→        </div>
   489→      </div>
   490→
   491→      {boards.length > 0 ? (
   492→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   493→          {boards.map((board) => {
   494→            const isOnline = board.status === 'online' || board.status === 'active'
   495→            const isWarning = board.status === 'warning' || board.status === 'maintenance'
   496→            return (
   497→              <motion.div
   498→                key={board.id}
   499→                initial={{ opacity: 0, scale: 0.97 }}
   500→                animate={{ opacity: 1, scale: 1 }}
   501→                className="macos-card p-4"
   502→              >
   503→                <div className="flex items-start justify-between mb-3">
   504→                  <div className="flex items-center gap-2.5">
   505→                    <div className={`size-9 rounded-lg flex items-center justify-center ${
   506→                      isOnline ? 'bg-emerald-500/10' : isWarning ? 'bg-teal-500/10' : 'bg-red-500/10'
   507→                    }`}>
   508→                      <Monitor className={`size-4 ${isOnline ? 'text-emerald-500' : isWarning ? 'text-teal-500' : 'text-red-500'}`} />
   509→                    </div>
   510→                    <div>
   511→                      <h4 className="text-[13px] font-semibold">{board.name}</h4>
   512→                      <p className="text-[11px] text-muted-foreground">{board.boardType}</p>
   513→                    </div>
   514→                  </div>
   515→                  <div className="flex items-center gap-1.5">
