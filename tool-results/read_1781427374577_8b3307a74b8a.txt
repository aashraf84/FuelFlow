   113→export default function ReportBuilderModule() {
   114→  const [activeTab, setActiveTab] = useState('templates')
   115→  const [templates, setTemplates] = useState<ReportTemplate[]>([])
   116→  const [scheduledReports, setScheduledReports] = useState<ScheduledReport[]>([])
   117→  const [recentReports, setRecentReports] = useState<DailyReport[]>([])
   118→  const [loading, setLoading] = useState(true)
   119→  const [error, setError] = useState<string | null>(null)
   120→  const [generating, setGenerating] = useState<string | null>(null)
   121→
   122→  const fetchData = useCallback(async () => {
   123→    setLoading(true)
   124→    setError(null)
   125→    try {
   126→      const [templatesRes, scheduledRes, dailyRes] = await Promise.all([
   127→        fetch('/api/reports?type=templates'),
   128→        fetch('/api/reports?type=scheduled'),
   129→        fetch('/api/daily-reports?days=30'),
   130→      ])
   131→
   132→      if (!templatesRes.ok) throw new Error('Failed to fetch report templates')
   133→
   134→      const templatesData = await templatesRes.json()
   135→      const scheduledData = scheduledRes.ok ? await scheduledRes.json() : { scheduled: [] }
   136→      const dailyData = dailyRes.ok ? await dailyRes.json() : { reports: [] }
   137→
   138→      setTemplates(templatesData.templates || [])
   139→      setScheduledReports(scheduledData.scheduled || [])
   140→      setRecentReports(dailyData.reports || [])
   141→    } catch (err) {
   142→      setError(err instanceof Error ? err.message : 'Failed to load report data')
   143→    } finally {
   144→      setLoading(false)
   145→    }
   146→  }, [])
   147→
   148→  useEffect(() => {
   149→    fetchData()
   150→  }, [fetchData])
   151→
   152→  const handleGenerate = async (templateId: string, templateName: string) => {
   153→    setGenerating(templateId)
   154→    try {
   155→      const res = await fetch('/api/reports', {
   156→        method: 'POST',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify({
   159→          action: 'run_report',
   160→          templateId,
   161→        }),
   162→      })
   163→
   164→      if (!res.ok) {
   165→        const errData = await res.json().catch(() => ({}))
   166→        throw new Error(errData.error || 'Report generation failed')
   167→      }
   168→
   169→      // Refresh recent reports after generation
   170→      const dailyRes = await fetch('/api/daily-reports?days=30')
   171→      if (dailyRes.ok) {
   172→        const dailyData = await dailyRes.json()
   173→        setRecentReports(dailyData.reports || [])
   174→      }
   175→    } catch (err) {
   176→      // Silent fail - could add toast notification
   177→    } finally {
   178→      setGenerating(null)
   179→    }
   180→  }
   181→
   182→  // Loading state
   183→  if (loading) {
   184→    return (
   185→      <div className="space-y-4">
   186→        <div className="flex items-center justify-between">
   187→          <div className="h-5 bg-muted rounded w-32 animate-pulse" />
   188→          <div className="h-7 bg-muted rounded w-28 animate-pulse" />
   189→        </div>
   190→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-3">
   191→          {[1, 2, 3, 4, 5, 6].map((i) => (
   192→            <div key={i} className="macos-card p-4 animate-pulse">
   193→              <div className="flex items-start gap-2.5 mb-2">
   194→                <div className="size-9 rounded-[10px] bg-muted" />
   195→                <div className="flex-1">
   196→                  <div className="h-4 bg-muted rounded w-32 mb-1" />
   197→                  <div className="h-3 bg-muted rounded w-16" />
   198→                </div>
   199→              </div>
   200→              <div className="h-3 bg-muted rounded w-full mb-1" />
   201→              <div className="h-3 bg-muted rounded w-2/3" />
   202→            </div>
   203→          ))}
   204→        </div>
   205→      </div>
   206→    )
   207→  }
   208→
   209→  // Error state
   210→  if (error) {
   211→    return (
   212→      <div className="macos-card p-12 flex flex-col items-center justify-center text-muted-foreground">
   213→        <AlertTriangle className="size-10 mb-3 text-teal-500" />
   214→        <p className="text-[13px] font-medium">Failed to load report data</p>
   215→        <p className="text-[11px] mt-1">{error}</p>
   216→        <button onClick={fetchData} className="macos-btn mt-4 text-[12px] flex items-center gap-1.5">
   217→          <RefreshCw className="size-3" /> Retry
   218→        </button>
   219→      </div>
   220→    )
   221→  }
   222→

   241→        {/* ── Templates ── */}

   289→                          {template.category}
   290→                        </Badge>
   291→                      )}
   292→                      {template.chartType && (
   293→                        <Badge variant="outline" className="text-[9px] px-1.5 py-0 bg-teal-500/5 text-teal-600 border-teal-500/15">

   296→                      )}
   297→                      {template.isPublic && (
   298→                        <Badge variant="outline" className="text-[9px] px-1.5 py-0 bg-emerald-500/10 text-emerald-600 border-emerald-500/20">Public</Badge>
   299→                      )}
   300→                    </div>
   301→                    <div className="flex items-center justify-between mt-3 pt-3 border-t border-border">
   302→                      <span className="text-[10px] text-muted-foreground">Created: {new Date(template.createdAt).toLocaleDateString()}</span>

   305→                          <Eye className="size-3 text-muted-foreground" />
   306→                        </button>
   307→                        <button

   309→                          disabled={generating === template.id}
   310→                          onClick={() => handleGenerate(template.id, template.name)}
   311→                        >
   312→                          {generating === template.id ? (
   313→                            <Loader2 className="size-2.5 animate-spin" />

   315→                            <Activity className="size-2.5" />

   317→                          Generate
   318→                        </button>
   319→                      </div>
   320→                    </div>
   321→                  </motion.div>
   322→                )
   323→              })}
   324→            </div>
   325→          )}
   326→        </TabsContent>
   327→
   328→        {/* ── Scheduled Reports ── */}
   329→        <TabsContent value="scheduled" className="mt-4 space-y-3">
   330→          <div className="flex items-center justify-between">
   331→            <div className="macos-section-header">Scheduled Reports</div>
   332→            <button className="macos-btn text-[11px] px-2.5 py-1 flex items-center gap-1">
   333→              <Plus className="size-3" /> New Schedule
   334→            </button>
   335→          </div>
   336→          {scheduledReports.length === 0 ? (
   337→            <div className="macos-card p-12 flex flex-col items-center justify-center text-muted-foreground">
   338→              <Calendar className="size-10 mb-3 opacity-20" />
   339→              <p className="text-[13px] font-medium">No scheduled reports</p>

   367→                              {sched.scheduleType}
   368→                            </Badge>

   373→                              {sched.format}
