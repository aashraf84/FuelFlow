    99→// ── Formatting Helpers ──────────────────────────────────────
   100→
   101→function formatPKR(value: number): string {
   102→  return `PKR ${value.toLocaleString('en-PK')}`
   103→}
   104→
   105→function formatCompactPKR(value: number): string {
   106→  if (value >= 1000000) return `PKR ${(value / 1000000).toFixed(1)}M`
   107→  if (value >= 1000) return `PKR ${(value / 1000).toFixed(0)}K`
   108→  return `PKR ${value}`
   109→}

   113→function RevenueTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number }>; label?: string }) {

   125→function StatCard({
   126→  label,
   127→  value,
   128→  change,
   129→  icon: Icon,
   130→  accent = 'amber',
   131→}: {
   132→  label: string
   133→  value: string
   134→  change?: { value: number; label: string }
   135→  icon: React.ElementType
   136→  accent?: 'amber' | 'green' | 'red'
   137→}) {
   138→  const isPositive = change ? change.value >= 0 : true
   139→  const iconBg = accent === 'amber'
   140→    ? 'bg-teal-500/10'
   141→    : accent === 'green'
   142→      ? 'bg-emerald-500/10'
   143→      : 'bg-red-500/10'
   144→  const iconColor = accent === 'amber'
   145→    ? 'text-teal-500'
   146→    : accent === 'green'
   147→      ? 'text-emerald-500'
   148→      : 'text-red-500'
   149→
   150→  return (
   151→    <div className="macos-card p-4 flex flex-col gap-3">
   152→      <div className="flex items-start justify-between">
   153→        <div className="flex flex-col gap-0.5">
   154→          <span className="text-[11px] font-medium text-muted-foreground uppercase tracking-wide">{label}</span>
   155→          <span className="text-[22px] font-bold text-foreground leading-tight tracking-tight">{value}</span>
   156→        </div>
   157→        <div className={`flex size-9 items-center justify-center rounded-lg ${iconBg}`}>
   158→          <Icon className={`size-[18px] ${iconColor}`} strokeWidth={1.8} />
   159→        </div>
   160→      </div>
   161→      {change && (
   162→        <div className="flex items-center gap-1.5">
   163→          {isPositive ? (
   164→            <ArrowUpRight className="size-3 text-emerald-500" strokeWidth={2.5} />
   165→          ) : (
   166→            <ArrowDownRight className="size-3 text-red-500" strokeWidth={2.5} />
   167→          )}
   168→          <span className={`text-[11px] font-semibold ${isPositive ? 'text-emerald-500' : 'text-red-500'}`}>
   169→            {Math.abs(change.value)}%
   170→          </span>
   171→          <span className="text-[11px] text-muted-foreground">{change.label}</span>
   172→        </div>
   173→      )}
   174→    </div>
   175→  )
   176→}

   189→    dispensing: { dot: 'macos-status-online', label: 'Dispensing', badgeBg: 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400' },
   190→    active: { dot: 'macos-status-online', label: 'Active', badgeBg: 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400' },
   191→    idle: { dot: 'macos-status-warning', label: 'Idle', badgeBg: 'bg-teal-500/10 text-teal-600 dark:text-teal-400' },
   192→    maintenance: { dot: 'macos-status-error', label: 'Maintenance', badgeBg: 'bg-red-500/10 text-red-600 dark:text-red-400' },

   207→        </span>

   275→// ── AI Insight derived from dashboard data ──────────────────
   276→
   277→interface AIInsight {
   278→  id: string
   279→  type: 'critical' | 'warning' | 'info'
   280→  icon: React.ElementType
   281→  title: string
   282→  description: string
   283→}
   284→
   285→function generateInsights(data: DashboardData): AIInsight[] {
   286→  const insights: AIInsight[] = []
   287→
   288→  // Tank level alerts
   289→  data.tankLevels.forEach((tank) => {
   290→    if (tank.level < 20) {
   291→      insights.push({
   292→        id: `tank-critical-${tank.id}`,
   293→        type: 'critical',
   294→        icon: Droplets,
   295→        title: `${tank.product} tank below 20% — schedule delivery`,
   296→        description: `${tank.product} tank at ${tank.level}%. ${tank.currentLitres.toLocaleString()} L remaining. Recommend scheduling delivery immediately.`,
   297→      })
   298→    } else if (tank.level < 30) {
   299→      insights.push({
   300→        id: `tank-warning-${tank.id}`,
   301→        type: 'warning',
   302→        icon: Droplets,
   303→        title: `${tank.product} tank at ${tank.level}% — monitor closely`,
   304→        description: `${tank.currentLitres.toLocaleString()} L remaining out of ${tank.capacity.toLocaleString()} L capacity. Consider scheduling a top-up.`,
   305→      })
   306→    }
   307→  })
   308→
   309→  // Revenue trend insight
   310→  if (data.kpis.revenue.today > 0 && data.kpis.revenue.week > 0) {
   311→    const dailyAvg = data.kpis.revenue.week / 7
   312→    const pctChange = ((data.kpis.revenue.today - dailyAvg) / dailyAvg) * 100
   313→    if (Math.abs(pctChange) > 5) {
   314→      insights.push({
   315→        id: 'revenue-trend',
   316→        type: pctChange > 0 ? 'info' : 'warning',
   317→        icon: TrendingUp,
   318→        title: `Today's revenue ${pctChange > 0 ? 'up' : 'down'} ${Math.abs(pctChange).toFixed(1)}% vs weekly avg`,
   319→        description: `Today: ${formatPKR(data.kpis.revenue.today)}. Weekly average: ${formatPKR(Math.round(dailyAvg))}. ${pctChange > 0 ? 'Strong performance, ensure adequate staffing.' : 'Consider promotional activities to boost sales.'}`,
   320→      })
   321→    }
   322→  }
   323→
   324→  // Alerts from KPIs
   325→  if (data.kpis.alerts.critical > 0) {
   326→    insights.push({
   327→      id: 'alerts-critical',
   328→      type: 'critical',
   329→      icon: Zap,
   330→      title: `${data.kpis.alerts.critical} critical alert${data.kpis.alerts.critical > 1 ? 's' : ''} require attention`,
   331→      description: `${data.kpis.alerts.critical} critical and ${data.kpis.alerts.warning} warning alerts are active. Review and resolve immediately.`,
   332→    })
   333→  }
   334→
   335→  // Always add at least one info insight if we have data
   336→  if (insights.length === 0) {
   337→    insights.push({
   338→      id: 'all-clear',
   339→      type: 'info',
   340→      icon: TrendingUp,
   341→      title: 'All systems operating normally',
   342→      description: `Station ${data.stationInfo.name} is operational. Shift ${data.stationInfo.shift} is active with ${data.employeeOnDuty} employees on duty.`,
   343→    })
   344→  }
   345→
   346→  return insights
   347→}
