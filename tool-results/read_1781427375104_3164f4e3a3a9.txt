   136→// ── Component ─────────────────────────────────────────────
   137→
   138→export default function RetailModule() {
   139→  const [activeTab, setActiveTab] = useState<TabKey>('overview')
   140→
   141→  // Data states
   142→  const [products, setProducts] = useState<RetailProduct[]>([])
   143→  const [carWashSessions, setCarWashSessions] = useState<CarWashSessionData[]>([])
   144→  const [lotterySales, setLotterySales] = useState<LotterySaleData[]>([])
   145→  const [carWashSummary, setCarWashSummary] = useState<CarWashSummary | null>(null)
   146→  const [lotterySummary, setLotterySummary] = useState<LotterySummary | null>(null)
   147→  const [loading, setLoading] = useState(true)
   148→  const [error, setError] = useState<string | null>(null)
   149→
   150→  const fetchData = useCallback(async () => {
   151→    setLoading(true)
   152→    setError(null)
   153→    try {
   154→      const res = await fetch('/api/retail')
   155→      if (!res.ok) throw new Error('Failed to load retail data')
   156→      const data = await res.json()
   157→
   158→      setProducts((data.products ?? []) as RetailProduct[])
   159→      setCarWashSessions((data.carWash?.sessions ?? []) as CarWashSessionData[])
   160→      setLotterySales((data.lottery?.sales ?? []) as LotterySaleData[])
   161→      setCarWashSummary((data.carWash?.summary ?? null) as CarWashSummary | null)
   162→      setLotterySummary((data.lottery?.summary ?? null) as LotterySummary | null)
   163→    } catch (err) {
   164→      setError(err instanceof Error ? err.message : 'Unknown error')
   165→    } finally {
   166→      setLoading(false)
   167→    }
   168→  }, [])
   169→
   170→  useEffect(() => { fetchData() }, [fetchData])
   171→
   172→  // Compute stats from real product data
   173→  const retailStats = (() => {
   174→    const revenue = products.reduce((s, p) => s + p.price * p.stockQty, 0)
   175→    const itemsSold = products.reduce((s, p) => s + p.stockQty, 0)
   176→    const avgBasket = products.length > 0 ? Math.round(revenue / products.length) : 0
   177→
   178→    // Find top category by revenue
   179→    const catRevenue = new Map<string, number>()
   180→    for (const p of products) {
   181→      const cat = p.category ?? 'Other'
   182→      catRevenue.set(cat, (catRevenue.get(cat) ?? 0) + p.price * p.stockQty)
   183→    }
   184→    let topCategory = '—'
   185→    let topRev = 0
   186→    for (const [cat, rev] of catRevenue) {
   187→      if (rev > topRev) { topCategory = cat; topRev = rev }
   188→    }
   189→
   190→    return { revenue, itemsSold, avgBasket, topCategory }
   191→  })()
   192→
   193→  // Compute categories from real product data
   194→  const categories = (() => {
   195→    const catMap = new Map<string, { items: number; revenue: number }>()
   196→    for (const p of products) {
   197→      const cat = p.category ?? 'Other'
   198→      const existing = catMap.get(cat) ?? { items: 0, revenue: 0 }
   199→      existing.items++
   200→      existing.revenue += p.price * p.stockQty
   201→      catMap.set(cat, existing)
   202→    }
   203→    return Array.from(catMap.entries()).map(([name, data]) => ({
   204→      name,
   205→      icon: CATEGORY_ICONS[name] ?? Package,
   206→      items: data.items,
   207→      revenue: data.revenue,
   208→      color: CATEGORY_COLORS[name] ?? '#8e8e93',
   209→    }))
   210→  })()
   211→
   212→  // Compute top products by revenue
   213→  const topProducts = [...products]
   214→    .sort((a, b) => (b.price * b.stockQty) - (a.price * a.stockQty))
   215→    .slice(0, 8)
   216→    .map((p, i) => ({
   217→      rank: i + 1,
   218→      name: p.name,
   219→      sku: p.sku ?? '—',
   220→      category: p.category ?? 'Other',
   221→      sold: p.stockQty,
   222→      revenue: p.price * p.stockQty,
   223→      inStock: p.stockQty,
   224→    }))
   225→
   226→  // Compute recent transactions from products (as proxy for sales)
   227→  const recentTransactions = products.slice(0, 8).map((p, i) => ({
   228→    id: `RT-${1000 + i}`,
   229→    time: p.createdAt ?? new Date().toLocaleTimeString('en-PK', { hour: '2-digit', minute: '2-digit' }),
   230→    items: 1,
   231→    total: p.price,
   232→    payment: i % 3 === 0 ? 'Cash' : i % 3 === 1 ? 'Card' : 'JazzCash',
   233→    attendant: 'System',
   234→  }))
   235→
   236→  if (error) return <ErrorBanner message={error} onRetry={fetchData} />
   237→
   238→  return (
   239→    <div className="space-y-5" style={{ fontStyle: 'normal !important' }}>
   240→      {/* ── Tab Bar ── */}
   241→      <div className="flex items-center gap-1 overflow-x-auto pb-1 -mb-1">
   242→        {tabs.map((tab) => {
   243→          const isActive = activeTab === tab.key
   244→          const Icon = tab.icon
   245→          return (
   246→            <button
   247→              key={tab.key}
   248→              onClick={() => setActiveTab(tab.key)}
   249→              className={`flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[13px] font-medium transition-all duration-150 whitespace-nowrap ${
   250→                isActive
   251→                  ? 'bg-teal-500/10 text-teal-600 dark:text-teal-400'
   252→                  : 'text-muted-foreground hover:bg-black/[0.03] dark:hover:bg-white/[0.05] hover:text-foreground'
   253→              }`}
   254→            >
   255→              <Icon className="size-3.5" strokeWidth={isActive ? 2 : 1.5} />
   256→              {tab.label}
   257→            </button>
   258→          )
   259→        })}
   260→      </div>
   261→
   262→      {/* ── Tab Content ── */}
   263→      <AnimatePresence mode="wait">
   264→        <motion.div
   265→          key={activeTab}
   266→          initial={{ opacity: 0, y: 6 }}
   267→          animate={{ opacity: 1, y: 0 }}
   268→          exit={{ opacity: 0, y: -4 }}
   269→          transition={{ duration: 0.2, ease: [0.25, 0.1, 0.25, 1] }}
   270→        >
   271→          {activeTab === 'overview' && <OverviewTab loading={loading} retailStats={retailStats} categories={categories} recentTransactions={recentTransactions} topProducts={topProducts} />}
   272→          {activeTab === 'products' && <ProductsTab loading={loading} topProducts={topProducts} />}
   273→          {activeTab === 'transactions' && <TransactionsTab loading={loading} recentTransactions={recentTransactions} />}
   274→          {activeTab === 'carwash' && <CarWashTab loading={loading} sessions={carWashSessions} summary={carWashSummary} />}
   275→          {activeTab === 'lottery' && <LotteryTab loading={loading} sales={lotterySales} summary={lotterySummary} />}
   276→        </motion.div>
   277→      </AnimatePresence>
   278→    </div>
   279→  )
   280→}
