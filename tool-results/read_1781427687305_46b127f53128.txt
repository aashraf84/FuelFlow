   194→// Generate forecast data from alerts (fallback to static)
   195→const generateForecastData = (): ForecastPoint[] => {
   196→  const days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun']
   197→  const data: ForecastPoint[] = []
   198→  const now = new Date()
   199→  const todayIdx = now.getDay() === 0 ? 6 : now.getDay() - 1
   200→
   201→  days.forEach((day, i) => {
   202→    const isPast = i <= todayIdx
   203→    data.push({
   204→      day: isPast ? day : `${day}*`,
   205→      actual: isPast ? Math.round(170000 + Math.random() * 100000) : 0,
   206→      predicted: Math.round(175000 + Math.random() * 90000),
   207→    })
   208→  })
   209→
   210→  return data
   211→}
