   198→   198→   198→  const now = new Date()
   199→   199→   199→  const todayIdx = now.getDay() === 0 ? 6 : now.getDay() - 1
   200→   200→   200→
   201→   201→   201→  days.forEach((day, i) => {
   202→   202→   202→    const isPast = i <= todayIdx
   203→   203→   203→    data.push({
   204→   204→   204→      day: isPast ? day : `${day}*`,
   205→   205→   205→      actual: isPast ? Math.round(170000 + Math.random() * 100000) : 0,
   206→   206→   206→      predicted: Math.round(175000 + Math.random() * 90000),
   207→   207→   207→    })
   208→   208→   208→  })
   209→   209→   209→
   210→   210→   210→  return data
