   255→function timeAgo(date: Date | string | null): string {
   256→  if (!date) return 'N/A'
   257→  const d = typeof date === 'string' ? new Date(date) : date
   258→  const diff = Date.now() - d.getTime()
   259→  if (diff < 60000) return 'just now'
   260→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`
   261→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`
   262→  return `${Math.floor(diff / 86400000)}d ago`
   263→}
