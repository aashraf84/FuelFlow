   123→// ── Helpers ──────────────────────────────────────
   124→
   125→function formatPKR(amount: number): string {
   126→  return new Intl.NumberFormat('en-PK', { style: 'currency', currency: 'PKR', minimumFractionDigits: 0, maximumFractionDigits: 2 }).format(amount)
   127→}
   128→
   129→function formatNumber(n: number, decimals = 2): string {
   130→  return new Intl.NumberFormat('en-PK', { minimumFractionDigits: decimals, maximumFractionDigits: decimals }).format(n)
   131→}
   132→
   133→function formatDate(d: string | Date): string {
   134→  return new Date(d).toLocaleDateString('en-PK', { year: 'numeric', month: 'short', day: 'numeric' })
   135→}
   136→
   137→function formatDateTime(d: string | Date): string {
   138→  return new Date(d).toLocaleString('en-PK', { year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
   139→}
