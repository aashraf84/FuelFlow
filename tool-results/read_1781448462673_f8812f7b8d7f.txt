     4→// Helper to parse date params safely
     5→function parseDate(value: string | null): Date | undefined {
     6→  if (!value) return undefined
     7→  const d = new Date(value)
     8→  return isNaN(d.getTime()) ? undefined : d
     9→}
    10→
    11→// Build common where clause for station/date filtering
    12→function buildWhere(stationId: string | null, dateFrom?: Date, dateTo?: Date, dateField = 'createdAt') {
    13→  const where: Record<string, unknown> = {}
    14→  if (stationId) where.stationId = stationId
    15→  if (dateFrom || dateTo) {
    16→    where[dateField] = {
    17→      ...(dateFrom ? { gte: dateFrom } : {}),
    18→      ...(dateTo ? { lte: dateTo } : {}),
    19→    }
    20→  }
    21→  return where
    22→}

    24→// Standard report response structure
    25→function reportResponse(reportType: string, columns: string[], rows: Record<string, unknown>[], summary: Record<string, unknown>, dateFrom?: Date, dateTo?: Date) {
    26→  return NextResponse.json({
    27→    report: {
    28→      reportType,
    29→      generatedAt: new Date().toISOString(),
    30→      period: { from: (dateFrom || new Date()).toISOString(), to: (dateTo || new Date()).toISOString() },
    31→      columns,
    32→      rows,
    33→      summary,
    34→    },
    35→  })
    36→}

    38→export async function GET(request: Request) {
    39→  try {
    40→    const { searchParams } = new URL(request.url)
    41→    const type = searchParams.get('type') || 'overview'
    42→    const reportType = searchParams.get('reportType') || ''
    43→    const stationId = searchParams.get('stationId')
    44→    const tenantId = searchParams.get('tenantId')
    45→    const dateFrom = parseDate(searchParams.get('dateFrom'))
    46→    const dateTo = parseDate(searchParams.get('dateTo'))
    47→
    48→    switch (type) {
    49→      case 'report': {
    50→        return await handleReport(reportType, stationId, tenantId, dateFrom, dateTo)
    51→      }
    52→      case 'templates': {
    53→        const where = tenantId ? { tenantId } : {}
    54→        const templates = await db.reportTemplate.findMany({ where, orderBy: { createdAt: 'desc' } })
    55→        return NextResponse.json({ templates })
    56→      }
    57→      case 'scheduled': {
    58→        const where = tenantId ? { tenantId } : {}
    59→        const scheduled = await db.scheduledReport.findMany({ where, orderBy: { nextRunAt: 'asc' } })
    60→        return NextResponse.json({ scheduled })
    61→      }
    62→      case 'overview': {
    63→        const tWhere = tenantId ? { tenantId } : {}
    64→        const [templateCount, systemCount, customCount, scheduledCount, activeCount] = await Promise.all([
    65→          db.reportTemplate.count({ where: tWhere }),
    66→          db.reportTemplate.count({ where: { ...tWhere, isSystem: true } }),
    67→          db.reportTemplate.count({ where: { ...tWhere, isSystem: false } }),
    68→          db.scheduledReport.count({ where: tenantId ? { tenantId } : {} }),
    69→          db.scheduledReport.count({ where: { ...tWhere, isActive: true } }),
    70→        ])
    71→        return NextResponse.json({
    72→          overview: {
    73→            templates: { total: templateCount, system: systemCount, custom: customCount },
    74→            scheduled: { total: scheduledCount, active: activeCount },
    75→          },
    76→        })
    77→      }
    78→      default:
    79→        return NextResponse.json({ error: `Unknown type: ${type}` }, { status: 400 })
    80→    }
    81→  } catch (error) {
    82→    console.error('Reports GET error:', error)
    83→    return NextResponse.json({ error: 'Failed to fetch report data' }, { status: 500 })
    84→  }
    85→}

    87→async function handleReport(reportType: string, stationId: string | null, tenantId: string | null, dateFrom?: Date, dateTo?: Date) {
    88→  try {
    89→    switch (reportType) {
    90→      case 'daily_sales': return await dailySalesReport(stationId, dateFrom, dateTo)
    91→      case 'shift_summary': return await shiftSummaryReport(stationId, dateFrom, dateTo)
    92→      case 'tank_reconciliation': return await tankReconciliationReport(stationId, dateFrom, dateTo)
    93→      case 'fuel_inventory': return await fuelInventoryReport(stationId, dateFrom, dateTo)
    94→      case 'financial_summary': return await financialSummaryReport(stationId, tenantId, dateFrom, dateTo)
    95→      case 'price_history': return await priceHistoryReport(tenantId, dateFrom, dateTo)
    96→      case 'fleet_summary': return await fleetSummaryReport(tenantId, dateFrom, dateTo)
    97→      case 'employee_performance': return await employeePerformanceReport(stationId, dateFrom, dateTo)
    98→      case 'delivery_log': return await deliveryLogReport(stationId, dateFrom, dateTo)
    99→      case 'retail_sales': return await retailSalesReport(stationId, dateFrom, dateTo)
   100→      case 'loyalty_summary': return await loyaltySummaryReport(tenantId, dateFrom, dateTo)
   101→      case 'ev_cng_sessions': return await evCngSessionsReport(stationId, dateFrom, dateTo)
   102→      case 'expense_report': return await expenseReport(stationId, dateFrom, dateTo)
   103→      case 'attendance_report': return await attendanceReport(stationId, dateFrom, dateTo)
   104→      case 'reconciliation_report': return await reconciliationReport(stationId, dateFrom, dateTo)
   105→      case 'cash_reconciliation': return await cashReconciliationReport(stationId, dateFrom, dateTo)
   106→      case 'supplier_statement': return await supplierStatementReport(stationId, tenantId, dateFrom, dateTo)
   107→      case 'environmental_report': return await environmentalReport(stationId, dateFrom, dateTo)
   108→      case 'security_audit': return await securityAuditReport(tenantId, dateFrom, dateTo)
   109→      case 'station_performance': return await stationPerformanceReport(tenantId, dateFrom, dateTo)
   110→      default: return NextResponse.json({ error: `Unknown reportType: ${reportType}` }, { status: 400 })
   111→    }
   112→  } catch (error) {
   113→    console.error(`Report error [${reportType}]:`, error)
   114→    return reportResponse(reportType, [], [], { error: 'Failed to generate report' }, dateFrom, dateTo)
   115→  }
   116→}

   119→// 1. DAILY SALES REPORT
   120→// =====================================================
   121→async function dailySalesReport(stationId: string | null, dateFrom?: Date, dateTo?: Date) {
   122→  try {
   123→    const where = buildWhere(stationId, dateFrom, dateTo, 'startedAt') as Record<string, unknown>
   124→    where.status = 'completed'
   125→    const fuelings = await db.fueling.findMany({ where, orderBy: { startedAt: 'desc' }, take: 5000 })
   126→
   127→    // Group by date
   128→    const byDate: Record<string, { date: string; totalVolume: number; totalRevenue: number; transactionCount: number }> = {}
   129→    for (const f of fuelings) {
   130→      const dk = new Date(f.startedAt).toISOString().split('T')[0]
   131→      if (!byDate[dk]) byDate[dk] = { date: dk, totalVolume: 0, totalRevenue: 0, transactionCount: 0 }
   132→      byDate[dk].totalVolume += f.volume
   133→      byDate[dk].totalRevenue += f.totalAmount
   134→      byDate[dk].transactionCount += 1
   135→    }
   136→
   137→    const totalVolume = fuelings.reduce((s, f) => s + f.volume, 0)
   138→    const totalRevenue = fuelings.reduce((s, f) => s + f.totalAmount, 0)
   139→    const columns = ['date', 'transactionCount', 'totalVolume', 'totalRevenue']
   140→    const rows = Object.values(byDate).sort((a, b) => b.date.localeCompare(a.date)).map((d) => ({
   141→      date: d.date, transactionCount: d.transactionCount, totalVolume: Math.round(d.totalVolume * 100) / 100, totalRevenue: Math.round(d.totalRevenue * 100) / 100,
   142→    }))
   143→    return reportResponse('daily_sales', columns, rows, {
   144→      totalDays: rows.length, totalTransactions: fuelings.length,
   145→      totalVolume: Math.round(totalVolume * 100) / 100, totalRevenue: Math.round(totalRevenue * 100) / 100,
   146→      averageTicket: fuelings.length ? Math.round((totalRevenue / fuelings.length) * 100) / 100 : 0,
   147→    }, dateFrom, dateTo)
   148→  } catch { return reportResponse('daily_sales', [], [], { totalDays: 0, totalTransactions: 0, totalVolume: 0, totalRevenue: 0 }, dateFrom, dateTo) }
   149→}

   214→      totalTanks: tanks.length, totalCapacity, totalCurrentLevel: totalCurrent,
   215→      avgUtilization: totalCapacity > 0 ? Math.round((totalCurrent / totalCapacity) * 10000) / 100 : 0,
   216→      lowTanks: tanks.filter((t) => t.capacity > 0 && (t.currentLevel / t.capacity) < 0.2).length,
   217→    }, _dateFrom, _dateTo)

   222→// 5. FINANCIAL SUMMARY
   223→// =====================================================
   224→async function financialSummaryReport(stationId: string | null, tenantId: string | null, dateFrom?: Date, dateTo?: Date) {
   225→  try {
   226→    const tWhere = buildWhere(stationId, dateFrom, dateTo, 'date') as Record<string, unknown>
   227→    tWhere.status = 'completed'
   228→    if (tenantId) tWhere.tenantId = tenantId
   229→    const transactions = await db.transaction.findMany({ where: tWhere, take: 5000 })
   230→
   231→    const eWhere = buildWhere(stationId, dateFrom, dateTo, 'date') as Record<string, unknown>
   232→    eWhere.status = { in: ['approved', 'paid'] }
   233→    const expenses = await db.expense.findMany({ where: eWhere, take: 5000 })
   234→
   235→    const totalIncome = transactions.filter((t) => t.type === 'income').reduce((s, t) => s + t.amount, 0)
   236→    const totalExpenses = expenses.reduce((s, e) => s + e.amount, 0)
   237→
   238→    const columns = ['type', 'category', 'amount', 'paymentMethod', 'date', 'status']
   239→    const incomeRows = transactions.filter((t) => t.type === 'income').slice(0, 200).map((t) => ({
   240→      type: 'Income', category: t.category, amount: t.amount, paymentMethod: t.paymentMethod,
   241→      date: t.date.toISOString(), status: t.status,
   242→    }))
   243→    const expenseRows = expenses.slice(0, 200).map((e) => ({
   244→      type: 'Expense', category: e.category, amount: e.amount, paymentMethod: e.paymentMethod || 'N/A',
   245→      date: e.date.toISOString(), status: e.status,
   246→    }))
   247→    const rows = [...incomeRows, ...expenseRows]
   248→    return reportResponse('financial_summary', columns, rows, {
   249→      totalIncome: Math.round(totalIncome * 100) / 100, totalExpenses: Math.round(totalExpenses * 100) / 100,
   250→      netProfit: Math.round((totalIncome - totalExpenses) * 100) / 100,
   251→      profitMargin: totalIncome > 0 ? Math.round(((totalIncome - totalExpenses) / totalIncome) * 10000) / 100 : 0,
   252→    }, dateFrom, dateTo)
   253→  } catch { return reportResponse('financial_summary', [], [], { totalIncome: 0, totalExpenses: 0, netProfit: 0 }, dateFrom, dateTo) }
   254→}

   569→async function securityAuditReport(tenantId: string | null, dateFrom?: Date, dateTo?: Date) {
   570→  try {
   571→    const aWhere: Record<string, unknown> = {}
   572→    if (tenantId) aWhere.tenantId = tenantId
   573→    if (dateFrom || dateTo) aWhere.createdAt = { ...(dateFrom ? { gte: dateFrom } : {}), ...(dateTo ? { lte: dateTo } : {}) }
   574→    const [alerts, sessions, logs] = await Promise.all([
   575→      db.securityAlert.findMany({ where: aWhere, orderBy: { createdAt: 'desc' }, take: 200 }),
   576→      db.securitySession.findMany({ where: aWhere, orderBy: { createdAt: 'desc' }, take: 200 }),
   577→      db.auditLog.findMany({ where: aWhere, orderBy: { createdAt: 'desc' }, take: 200 }),
   578→    ])
   579→    const columns = ['source', 'action', 'severity', 'details', 'userId', 'createdAt']
   580→    const rows = [
   581→      ...alerts.map((a) => ({ source: 'Alert', action: a.alertType, severity: a.severity, details: a.message, userId: a.userId?.slice(-8) || 'System', createdAt: a.createdAt.toISOString() })),
   582→      ...sessions.map((s) => ({ source: 'Session', action: s.action, severity: 'info', details: `IP: ${s.ipAddress || 'N/A'}`, userId: s.userId?.slice(-8) || 'N/A', createdAt: s.createdAt.toISOString() })),
   583→      ...logs.map((l) => ({ source: 'Audit', action: l.action, severity: 'info', details: l.details || 'N/A', userId: l.userId?.slice(-8) || 'N/A', createdAt: l.createdAt.toISOString() })),
   584→    ].sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()).slice(0, 500)
   585→    return reportResponse('security_audit', columns, rows, {
   586→      totalAlerts: alerts.length, totalSessions: sessions.length, totalAuditLogs: logs.length,
   587→      criticalAlerts: alerts.filter((a) => a.severity === 'critical').length,
   588→    }, dateFrom, dateTo)
   589→  } catch { return reportResponse('security_audit', [], [], { totalAlerts: 0, totalSessions: 0, totalAuditLogs: 0 }, dateFrom, dateTo) }
   590→}

   614→// Keep existing POST handler for templates and scheduled reports
   615→export async function POST(request: Request) {
   616→  try {
   617→    const body = await request.json()
   618→    const { action } = body
   619→    switch (action) {
   620→      case 'create_template': {
   621→        const { tenantId, name, description, reportType, category, dataSource, columns, filters, groupBy, sortBy, chartType, isPublic, createdBy } = body
   622→        if (!tenantId || !name || !reportType || !dataSource || !columns) {
   623→          return NextResponse.json({ error: 'Missing required fields' }, { status: 400 })
   624→        }
   625→        const template = await db.reportTemplate.create({
   626→          data: { tenantId, name, description: description || null, reportType, category: category || null, dataSource, columns, filters: filters || null, groupBy: groupBy || null, sortBy: sortBy || null, chartType: chartType || null, isSystem: false, isPublic: isPublic || false, createdBy: createdBy || null },
   627→        })
   628→        return NextResponse.json({ template }, { status: 201 })
   629→      }
   630→      case 'schedule_report': {
   631→        const { tenantId, reportTemplateId, scheduleType, scheduleTime, scheduleDay, recipients, format } = body
   632→        if (!tenantId || !reportTemplateId || !scheduleType || !recipients) {
   633→          return NextResponse.json({ error: 'Missing required fields' }, { status: 400 })
   634→        }
   635→        const scheduled = await db.scheduledReport.create({
   636→          data: { tenantId, reportTemplateId, scheduleType, scheduleTime: scheduleTime || null, scheduleDay: scheduleDay || null, recipients, format: format || 'pdf' },
   637→        })
   638→        return NextResponse.json({ scheduled }, { status: 201 })
   639→      }
   640→      default:
   641→        return NextResponse.json({ error: `Unknown action: ${action}` }, { status: 400 })
   642→    }
   643→  } catch (error) {
   644→    console.error('Reports POST error:', error)
   645→    return NextResponse.json({ error: 'Failed to process report action' }, { status: 500 })
   646→  }
   647→}
